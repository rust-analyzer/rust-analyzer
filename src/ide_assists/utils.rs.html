41pub(crate) fn unwrap_trivial_block(block_expr: ast::BlockExpr) -> ast::Expr {
42    extract_trivial_expression(&block_expr)
43        .filter(|expr| !expr.syntax().text().contains_char('\n'))
44        .unwrap_or_else(|| block_expr.into())
45}

47pub fn extract_trivial_expression(block_expr: &ast::BlockExpr) -> Option<ast::Expr> {
48    if block_expr.modifier().is_some() {
49        return None;
50    }
51    let stmt_list = block_expr.stmt_list()?;
52    let has_anything_else = |thing: &SyntaxNode| -> bool {
53        let mut non_trivial_children =
54            stmt_list.syntax().children_with_tokens().filter(|it| match it.kind() {
55                WHITESPACE | T!['{'] | T!['}'] => false,
56                _ => it.as_node() != Some(thing),
57            });
58        non_trivial_children.next().is_some()
59    };
60    if stmt_list.syntax().children_with_tokens().any(|it| ast::AnyComment::can_cast(it.kind())) {
61        return None;
62    }
63
64    if let Some(expr) = stmt_list.tail_expr() {
65        if has_anything_else(expr.syntax()) {
66            return None;
67        }
68        return Some(expr);
69    }
70    // Unwrap `{ continue; }`
71    let stmt = stmt_list.statements().next()?;
72    if let ast::Stmt::ExprStmt(expr_stmt) = stmt {
73        if has_anything_else(expr_stmt.syntax()) {
74            return None;
75        }
76        let expr = expr_stmt.expr()?;
77        if matches!(expr.syntax().kind(), CONTINUE_EXPR | BREAK_EXPR | RETURN_EXPR) {
78            return Some(expr);
79        }
80    }
81    None
82}

84pub(crate) fn wrap_block(expr: &ast::Expr, make: &SyntaxFactory) -> ast::BlockExpr {
85    if let ast::Expr::BlockExpr(block) = expr
86        && let Some(first) = block.syntax().first_token()
87        && first.kind() == T!['{']
88    {
89        block.reset_indent()
90    } else {
91        make.block_expr(None, Some(expr.reset_indent().indent(1.into())))
92    }
93}

103fn needs_parens_in_call(make: &SyntaxFactory, param: &ast::Expr) -> bool {
104    let call = make.expr_call(make.expr_unit(), make.arg_list(Vec::new()));
105    let callable = call.expr().expect("invalid make call");
106    param.needs_parens_in_place_of(call.syntax(), callable.syntax())

109pub(crate) fn wrap_paren_in_guard_chain(guard: ast::Expr, make: &SyntaxFactory) -> ast::Expr {

113fn needs_parens_in_guard_chain(make: &SyntaxFactory, guard: &ast::Expr) -> bool {
114    let ast::Expr::BinExpr(if_let_and_guard) = make.expr_bin_op(
115        make.expr_unit(),
116        ast::BinaryOp::LogicOp(ast::LogicOp::And),
117        make.expr_unit(),
118    ) else {
119        stdx::never!("`SyntaxFactory::expr_bin_op` returns a `BinExpr`");
120        return false;
121    };
122    let Some(fake_guard) = if_let_and_guard.rhs() else {
123        stdx::never!("invalid make call");
124        return false;
125    };
126    guard.needs_parens_in_place_of(if_let_and_guard.syntax(), fake_guard.syntax())
127}

129fn needs_parens_in_bin_expr(make: &SyntaxFactory, expr: &ast::Expr, kind: ast::BinaryOp) -> bool {
130    let ast::Expr::BinExpr(bin_expr) = make.expr_bin_op(make.expr_unit(), kind, make.expr_unit())
131    else {
132        stdx::never!("invalid make bin_expr");
133        return false;
134    };
135    let lhs = bin_expr.lhs().expect("invalid make");
136    expr.needs_parens_in_place_of(bin_expr.syntax(), lhs.syntax())
137}

139/// This is a method with a heuristics to support test methods annotated with custom test annotations, such as
140/// `#[test_case(...)]`, `#[tokio::test]` and similar.
141/// Also a regular `#[test]` annotation is supported.
142///
143/// It may produce false positives, for example, `#[wasm_bindgen_test]` requires a different command to run the test,
144/// but it's better than not to have the runnables for the tests at all.
145pub fn test_related_attribute_syn(fn_def: &ast::Fn) -> Option<ast::Attr> {
146    fn_def.attrs().find_map(|attr| {
147        let path = attr.path()?;
148        let text = path.syntax().text().to_string();
149        if text.starts_with("test") || text.ends_with("test") { Some(attr) } else { None }
150    })
151}

169pub fn filter_assoc_items(
170    sema: &Semantics<'_, RootDatabase>,
171    trait_: hir::Trait,
172    items: &[(hir::AssocItem, IsRequiredAssocItem)],
173    default_methods: DefaultMethods,
174    ignore_items: IgnoreAssocItems,
175) -> Vec<InFile<ast::AssocItem>> {
176    let mut result = items
177        .iter()
178        .copied()
179        .filter(|(assoc_item, is_required)| {
180            if ignore_items == IgnoreAssocItems::DocHiddenAttrPresent
181                && assoc_item.attrs(sema.db).is_doc_hidden()
182            {
183                if let hir::AssocItem::Function(f) = assoc_item
184                    && !f.has_body(sema.db)
185                {
186                    return true;
187                }
188                return false;
189            }
190
191            is_required.0 == (default_methods == DefaultMethods::No)
192        })
193        .map(|(item, _)| (item, item.attrs(sema.db).unstable_feature(sema.db)))
194        // Note: This throws away items with no source.
195        .filter_map(|(assoc_item, unstable_feature)| {
196            let item = match assoc_item {
197                hir::AssocItem::Function(it) => sema.source(it)?.map(ast::AssocItem::Fn),
198                hir::AssocItem::TypeAlias(it) => sema.source(it)?.map(ast::AssocItem::TypeAlias),
199                hir::AssocItem::Const(it) => sema.source(it)?.map(ast::AssocItem::Const),
200            };
201            Some((item, unstable_feature))
202        })
203        .collect::<Vec<_>>();
204
205    // Now, we want to filter unstable assoc items whose feature is not enabled, unless:
206    //  - it's required, or
207    //  - the trait has the same feature, so the user probably intends to enable it.
208    if default_methods == DefaultMethods::Only {
209        let trait_unstable_feature = trait_.attrs(sema.db).unstable_feature(sema.db);
210        let krate = trait_.krate(sema.db);
211        result.retain(|(_, item_unstable_feature)| {
212            *item_unstable_feature == trait_unstable_feature
213                || item_unstable_feature
214                    .as_ref()
215                    .is_none_or(|feature| krate.is_unstable_feature_enabled(sema.db, feature))
216        });
217    }
218
219    result.into_iter().map(|(item, _)| item).collect()
220}

222/// Given `original_items` retrieved from the trait definition (usually by
223/// [`filter_assoc_items()`]), clones each item for update and applies path transformation to it,
224/// then inserts into `impl_`. Returns the modified `impl_` and the first associated item that got
225/// inserted.
226#[must_use]
227pub fn add_trait_assoc_items_to_impl(
228    make: &SyntaxFactory,
229    sema: &Semantics<'_, RootDatabase>,
230    config: &AssistConfig,
231    original_items: &[InFile<ast::AssocItem>],
232    trait_: hir::Trait,
233    impl_: &ast::Impl,
234    target_scope: &hir::SemanticsScope<'_>,
235    default_mode: DefaultMethods,
236) -> Vec<ast::AssocItem> {
237    let new_indent_level = IndentLevel::from_node(impl_.syntax()) + 1;
238    original_items
239        .iter()
240        .map(|InFile { file_id, value: original_item }| {
241            let mut cloned_item = {
242                if let Some(macro_file) = file_id.macro_file() {
243                    let span_map = macro_file.expansion_span_map(sema.db);
244                    let item_prettified = prettify_macro_expansion(
245                        sema.db,
246                        original_item.syntax().clone(),
247                        span_map,
248                        target_scope.krate().into(),
249                    );
250                    if let Some(formatted) = ast::AssocItem::cast(item_prettified) {
251                        return formatted;
252                    } else {
253                        stdx::never!("formatted `AssocItem` could not be cast back to `AssocItem`");
254                    }
255                }
256                original_item
257            }
258            .reset_indent();
259
260            if let Some(source_scope) = sema.scope(original_item.syntax()) {
261                // FIXME: Paths in nested macros are not handled well. See
262                // `add_missing_impl_members::paths_in_nested_macro_should_get_transformed` test.
263                let transform =
264                    PathTransform::trait_impl(target_scope, &source_scope, trait_, impl_.clone());
265                cloned_item = ast::AssocItem::cast(transform.apply(cloned_item.syntax())).unwrap();
266            }
267            let (editor, cloned_item) = SyntaxEditor::with_ast_node(&cloned_item);
268            cloned_item.remove_attrs_and_docs(&editor);
269            ast::AssocItem::cast(editor.finish().new_root().clone()).unwrap()
270        })
271        .filter_map(|item| match item {
272            // We can check `fn_.body().is_none()`, but this is actually not what we want to check: some functions (`Drop::drop()`
273            // or `#[rustc_must_implement_one_of]`) have a default body that should be ignored. So the criteria is whether
274            // we requested required or defaulted methods, and not whether the method actually has a body.
275            ast::AssocItem::Fn(fn_) if default_mode == DefaultMethods::No => {
276                let (fn_editor, fn_) = SyntaxEditor::with_ast_node(&fn_);
277                let fill_expr: ast::Expr = match config.expr_fill_default {
278                    ExprFillDefaultMode::Todo | ExprFillDefaultMode::Default => make.expr_todo(),
279                    ExprFillDefaultMode::Underscore => make.expr_underscore().into(),
280                };
281                let new_body = make.block_expr(None::<ast::Stmt>, Some(fill_expr));
282                fn_.replace_or_insert_body(&fn_editor, new_body);
283                let new_fn_ = fn_editor.finish().new_root().clone();
284                ast::AssocItem::cast(new_fn_)
285            }
286            ast::AssocItem::TypeAlias(type_alias) => {
287                let (type_alias_editor, type_alias) = SyntaxEditor::with_ast_node(&type_alias);
288                if let Some(type_bound_list) = type_alias.type_bound_list() {
289                    type_bound_list.remove(&type_alias_editor);
290                };
291                let type_alias = type_alias_editor.finish().new_root().clone();
292                ast::AssocItem::cast(type_alias)
293            }
294            item => Some(item),
295        })
296        .map(|item| AstNodeEdit::indent(&item, new_indent_level))
297        .collect()
298}

313fn invert_special_case(make: &SyntaxFactory, expr: &ast::Expr) -> Option<ast::Expr> {
314    match expr {
315        ast::Expr::BinExpr(bin) => {
316            let op_kind = bin.op_kind()?;
317            let rev_kind = match op_kind {
318                ast::BinaryOp::CmpOp(ast::CmpOp::Eq { negated }) => {
319                    ast::BinaryOp::CmpOp(ast::CmpOp::Eq { negated: !negated })
320                }
321                ast::BinaryOp::CmpOp(ast::CmpOp::Ord { ordering: ast::Ordering::Less, strict }) => {
322                    ast::BinaryOp::CmpOp(ast::CmpOp::Ord {
323                        ordering: ast::Ordering::Greater,
324                        strict: !strict,
325                    })
326                }
327                ast::BinaryOp::CmpOp(ast::CmpOp::Ord {
328                    ordering: ast::Ordering::Greater,
329                    strict,
330                }) => ast::BinaryOp::CmpOp(ast::CmpOp::Ord {
331                    ordering: ast::Ordering::Less,
332                    strict: !strict,
333                }),
334                // Parenthesize other expressions before prefixing `!`
335                _ => {
336                    return Some(
337                        make.expr_prefix(T![!], make.expr_paren(expr.clone()).into()).into(),
338                    );
339                }
340            };
341            let mut lhs = bin.lhs()?;
342            if needs_parens_in_bin_expr(make, &lhs, rev_kind) {
343                lhs = make.expr_paren(lhs).into();
344            }
345
346            Some(make.expr_bin(lhs, rev_kind, bin.rhs()?).into())
347        }
348        ast::Expr::MethodCallExpr(mce) => {
349            let receiver = mce.receiver()?;
350            let method = mce.name_ref()?;
351            let arg_list = mce.arg_list()?;
352
353            let method = match method.text() {
354                "is_some" => "is_none",
355                "is_none" => "is_some",
356                "is_ok" => "is_err",
357                "is_err" => "is_ok",
358                _ => return None,
359            };
360
361            Some(make.expr_method_call(receiver, make.name_ref(method), arg_list).into())
362        }
363        ast::Expr::PrefixExpr(pe) if pe.op_kind()? == ast::UnaryOp::Not => match pe.expr()? {
364            ast::Expr::ParenExpr(parexpr) => parexpr.expr(),
365            _ => pe.expr(),
366        },
367        ast::Expr::Literal(lit) => match lit.kind() {
368            ast::LiteralKind::Bool(b) => match b {
369                true => Some(ast::Expr::Literal(make.expr_literal("false"))),
370                false => Some(ast::Expr::Literal(make.expr_literal("true"))),
371            },
372            _ => None,
373        },
374        _ => None,
375    }
376}

378pub(crate) fn insert_attributes(
379    before: impl Element,
380    editor: &SyntaxEditor,
381    attrs: impl IntoIterator<Item = ast::Attr>,
382) {
383    let make = editor.make();
384    let mut attrs = attrs.into_iter().peekable();
385    if attrs.peek().is_none() {
386        return;
387    }
388    let elem = before.syntax_element();
389    let indent = IndentLevel::from_element(&elem);
390    let whitespace = format!("\n{indent}");
391    let elements: Vec<syntax::SyntaxElement> = attrs
392        .flat_map(|attr| [attr.syntax().clone().into(), make.whitespace(&whitespace).into()])
393        .collect();
394    editor.insert_all(syntax::syntax_editor::Position::before(elem), elements);
395}

401pub(crate) fn does_pat_match_variant(pat: &ast::Pat, var: &ast::Pat) -> bool {
402    let first_node_text = |pat: &ast::Pat| pat.syntax().first_child().map(|node| node.text());
403
404    let pat_head = match pat {
405        ast::Pat::IdentPat(bind_pat) => match bind_pat.pat() {
406            Some(p) => first_node_text(&p),
407            None => return pat.syntax().text() == var.syntax().text(),
408        },
409        pat => first_node_text(pat),
410    };
411
412    let var_head = first_node_text(var);
413
414    pat_head == var_head
415}

430fn check_pat_variant_nested_or_literal_with_depth(
431    ctx: &AssistContext<'_, '_>,
432    pat: &ast::Pat,
433    depth_after_refutable: usize,
434) -> bool {
435    if depth_after_refutable > 1 {
436        return true;
437    }
438
439    match pat {
440        ast::Pat::RestPat(_) | ast::Pat::WildcardPat(_) | ast::Pat::RefPat(_) => false,
441
442        ast::Pat::LiteralPat(_)
443        | ast::Pat::RangePat(_)
444        | ast::Pat::MacroPat(_)
445        | ast::Pat::PathPat(_)
446        | ast::Pat::BoxPat(_)
447        | ast::Pat::DerefPat(_)
448        | ast::Pat::NotNull(_)
449        | ast::Pat::CfgPredPat(_)
450        | ast::Pat::ConstBlockPat(_) => true,
451
452        ast::Pat::IdentPat(ident_pat) => ident_pat.pat().is_some_and(|pat| {
453            check_pat_variant_nested_or_literal_with_depth(ctx, &pat, depth_after_refutable)
454        }),
455        ast::Pat::ParenPat(paren_pat) => paren_pat.pat().is_none_or(|pat| {
456            check_pat_variant_nested_or_literal_with_depth(ctx, &pat, depth_after_refutable)
457        }),
458        ast::Pat::TuplePat(tuple_pat) => tuple_pat.fields().any(|pat| {
459            check_pat_variant_nested_or_literal_with_depth(ctx, &pat, depth_after_refutable)
460        }),
461        ast::Pat::RecordPat(record_pat) => {
462            let adjusted_next_depth =
463                depth_after_refutable + if check_pat_variant_from_enum(ctx, pat) { 1 } else { 0 };
464            record_pat.record_pat_field_list().is_none_or(|pat| {
465                pat.fields().any(|pat| {
466                    pat.pat().is_none_or(|pat| {
467                        check_pat_variant_nested_or_literal_with_depth(
468                            ctx,
469                            &pat,
470                            adjusted_next_depth,
471                        )
472                    })
473                })
474            })
475        }
476        ast::Pat::OrPat(or_pat) => or_pat.pats().any(|pat| {
477            check_pat_variant_nested_or_literal_with_depth(ctx, &pat, depth_after_refutable)
478        }),
479        ast::Pat::TupleStructPat(tuple_struct_pat) => {
480            let adjusted_next_depth =
481                depth_after_refutable + if check_pat_variant_from_enum(ctx, pat) { 1 } else { 0 };
482            tuple_struct_pat.fields().any(|pat| {
483                check_pat_variant_nested_or_literal_with_depth(ctx, &pat, adjusted_next_depth)
484            })
485        }
486        ast::Pat::SlicePat(slice_pat) => {
487            let mut pats = slice_pat.pats();
488            pats.next()
489                .is_none_or(|pat| !matches!(pat, ast::Pat::RestPat(_)) || pats.next().is_some())
490        }
491    }
492}

503// Uses a syntax-driven approach to find any impl blocks for the struct that
504// exist within the module/file
505//
506// Returns `None` if we've found an existing fn
507//
508// FIXME: change the new fn checking to a more semantic approach when that's more
509// viable (e.g. we process proc macros, etc)
510// FIXME: this partially overlaps with `find_impl_block_*`
511
512/// `find_struct_impl` looks for impl of a struct, but this also has additional feature
513/// where it takes a list of function names and check if they exist inside impl_, if
514/// even one match is found, it returns None.
515///
516/// That means this function can have 3 potential return values:
517///  - `None`: an impl exists, but one of the function names within the impl matches one of the provided names.
518///  - `Some(None)`: no impl exists.
519///  - `Some(Some(_))`: an impl exists, with no matching function names.
520pub(crate) fn find_struct_impl(
521    ctx: &AssistContext<'_, '_>,
522    adt: &ast::Adt,
523    names: &[String],
524) -> Option<Option<ast::Impl>> {
525    let db = ctx.db();
526    let module = adt.syntax().parent()?;
527
528    let struct_def = ctx.sema.to_def(adt)?;
529
530    let block = module.descendants().filter_map(ast::Impl::cast).find_map(|impl_blk| {
531        let blk = ctx.sema.to_def(&impl_blk)?;
532
533        // FIXME: handle e.g. `struct S<T>; impl<U> S<U> {}`
534        // (we currently use the wrong type parameter)
535        // also we wouldn't want to use e.g. `impl S<u32>`
536
537        let same_ty = match blk.self_ty(db).as_adt() {
538            Some(def) => def == struct_def,
539            None => false,
540        };
541        let not_trait_impl = blk.trait_(db).is_none();
542
543        if !(same_ty && not_trait_impl) { None } else { Some(impl_blk) }
544    });
545
546    if let Some(ref impl_blk) = block
547        && has_any_fn(impl_blk, names)
548    {
549        return None;
550    }
551
552    Some(block)
553}
554
555fn has_any_fn(imp: &ast::Impl, names: &[String]) -> bool {
556    if let Some(il) = imp.assoc_item_list() {
557        for item in il.assoc_items() {
558            if let ast::AssocItem::Fn(f) = item
559                && let Some(name) = f.name()
560                && names.iter().any(|n| n.eq_ignore_ascii_case(name.text()))
561            {
562                return true;
563            }
564        }
565    }
566
567    false
568}

570/// Generates the corresponding `impl Type {}` including type and lifetime
571/// parameters.
572pub(crate) fn generate_impl_with_item(
573    make: &SyntaxFactory,
574    adt: &ast::Adt,

628fn generate_impl_inner(
629    make: &SyntaxFactory,
630    is_unsafe: bool,
631    adt: &ast::Adt,
632    trait_: Option<ast::Type>,
633    trait_is_transitive: bool,
634    body: Option<ast::AssocItemList>,
635) -> ast::Impl {
636    // Ensure lifetime params are before type & const params
637    let generic_params = adt.generic_param_list().map(|generic_params| {
638        let lifetime_params =
639            generic_params.lifetime_params().map(ast::GenericParam::LifetimeParam);
640        let ty_or_const_params = generic_params.type_or_const_params().filter_map(|param| {
641            let param = match param {
642                ast::TypeOrConstParam::Type(param) => {
643                    // remove defaults since they can't be specified in impls
644                    let mut bounds =
645                        param.type_bound_list().map_or_else(Vec::new, |it| it.bounds().collect());
646                    if let Some(trait_) = &trait_ {
647                        // Add the current trait to `bounds` if the trait is transitive,
648                        // meaning `impl<T> Trait for U<T>` requires `T: Trait`.
649                        if trait_is_transitive {
650                            bounds.push(make.type_bound(trait_.clone()));
651                        }
652                    };
653                    // `{ty_param}: {bounds}`
654                    let param = make.type_param(param.name()?, make.type_bound_list(bounds));
655                    ast::GenericParam::TypeParam(param)
656                }
657                ast::TypeOrConstParam::Const(param) => {
658                    // remove defaults since they can't be specified in impls
659                    let param = make.const_param(param.name()?, param.ty()?);
660                    ast::GenericParam::ConstParam(param)
661                }
662            };
663            Some(param)
664        });
665
666        make.generic_param_list(itertools::chain(lifetime_params, ty_or_const_params))
667    });
668    let generic_args = generic_params.as_ref().map(|params| params.to_generic_args(make));
669    let adt_assoc_bounds = trait_
670        .as_ref()
671        .zip(generic_params.as_ref())
672        .and_then(|(trait_, params)| generic_param_associated_bounds(make, adt, trait_, params));
673
674    let ty: ast::Type = make.ty_path(make.ident_path(adt.name().unwrap().text())).into();
675
676    let cfg_attrs = adt.attrs().filter(|attr| matches!(attr.meta(), Some(ast::Meta::CfgMeta(_))));
677    match trait_ {
678        Some(trait_) => make.impl_trait(
679            cfg_attrs,
680            is_unsafe,
681            None,
682            None,
683            generic_params,
684            generic_args,
685            false,
686            trait_,
687            ty,
688            adt_assoc_bounds,
689            adt.where_clause(),
690            body,
691        ),
692        None => make.impl_(cfg_attrs, generic_params, generic_args, ty, adt.where_clause(), body),
693    }
694}
695
696fn generic_param_associated_bounds(
697    make: &SyntaxFactory,
698    adt: &ast::Adt,
699    trait_: &ast::Type,
700    generic_params: &ast::GenericParamList,
701) -> Option<ast::WhereClause> {
702    let in_type_params = |name: &ast::NameRef| {
703        generic_params
704            .generic_params()
705            .filter_map(|param| match param {
706                ast::GenericParam::TypeParam(type_param) => type_param.name(),
707                _ => None,
708            })
709            .any(|param| param.text() == name.text())
710    };
711    let adt_body = match adt {
712        ast::Adt::Enum(e) => e.variant_list().map(|it| it.syntax().clone()),
713        ast::Adt::Struct(s) => s.field_list().map(|it| it.syntax().clone()),
714        ast::Adt::Union(u) => u.record_field_list().map(|it| it.syntax().clone()),
715    };
716    let mut trait_where_clause = adt_body
717        .into_iter()
718        .flat_map(|it| it.descendants())
719        .filter_map(ast::Path::cast)
720        .filter_map(|path| {
721            let qualifier = path.qualifier()?.as_single_segment()?;
722            let qualifier = qualifier
723                .name_ref()
724                .or_else(|| match qualifier.type_anchor()?.self_ty()? {
725                    ast::Type::PathType(path_type) => path_type.path()?.as_single_name_ref(),
726                    _ => None,
727                })
728                .filter(in_type_params)?;
729            Some((qualifier, path.segment()?.name_ref()?))
730        })
731        .map(|(qualifier, assoc_name)| {
732            let segments = [qualifier, assoc_name].map(|nr| make.path_segment(nr));
733            let path = make.path_from_segments(segments, false);
734            let bounds = [make.type_bound(trait_.clone())];
735            make.where_pred(either::Either::Right(make.ty_path(path).into()), bounds)
736        })
737        .unique_by(|it| it.syntax().to_string())
738        .peekable();
739    trait_where_clause.peek().is_some().then(|| make.where_clause(trait_where_clause))
740}

742#[derive(Debug)]
743pub(crate) struct ReferenceConversion<'db> {
744    conversion: ReferenceConversionType,
745    ty: hir::Type<'db>,
746    impls_deref: bool,
747}
748
749#[derive(Debug)]
750enum ReferenceConversionType {
751    // reference can be stripped if the type is Copy
752    Copy,
753    // &String -> &str
754    AsRefStr,
755    // &Vec<T> -> &[T]
756    AsRefSlice,
757    // &Box<T> -> &T
758    Dereferenced,
759    // &Option<T> -> Option<&T>
760    Option,
761    // &Result<T, E> -> Result<&T, &E>
762    Result,
763}
764
765impl<'db> ReferenceConversion<'db> {
766    fn type_to_string(&self, db: &'db dyn HirDatabase, module: hir::Module) -> String {
767        match self.conversion {
768            ReferenceConversionType::Copy => self
769                .ty
770                .display_source_code(db, module.into(), true)
771                .unwrap_or_else(|_| "_".to_owned()),
772            ReferenceConversionType::AsRefStr => "&str".to_owned(),
773            ReferenceConversionType::AsRefSlice => {
774                let type_argument_name = self
775                    .ty
776                    .type_arguments()
777                    .next()
778                    .unwrap()
779                    .display_source_code(db, module.into(), true)
780                    .unwrap_or_else(|_| "_".to_owned());
781                format!("&[{type_argument_name}]")
782            }
783            ReferenceConversionType::Dereferenced => {
784                let type_argument_name = self
785                    .ty
786                    .type_arguments()
787                    .next()
788                    .unwrap()
789                    .display_source_code(db, module.into(), true)
790                    .unwrap_or_else(|_| "_".to_owned());
791                format!("&{type_argument_name}")
792            }
793            ReferenceConversionType::Option => {
794                let type_argument_name = self
795                    .ty
796                    .type_arguments()
797                    .next()
798                    .unwrap()
799                    .display_source_code(db, module.into(), true)
800                    .unwrap_or_else(|_| "_".to_owned());
801                format!("Option<&{type_argument_name}>")
802            }
803            ReferenceConversionType::Result => {
804                let mut type_arguments = self.ty.type_arguments();
805                let first_type_argument_name = type_arguments
806                    .next()
807                    .unwrap()
808                    .display_source_code(db, module.into(), true)
809                    .unwrap_or_else(|_| "_".to_owned());
810                let second_type_argument_name = type_arguments
811                    .next()
812                    .unwrap()
813                    .display_source_code(db, module.into(), true)
814                    .unwrap_or_else(|_| "_".to_owned());
815                format!("Result<&{first_type_argument_name}, &{second_type_argument_name}>")
816            }
817        }
818    }
819
820    pub(crate) fn convert_type(&self, db: &'db dyn HirDatabase, module: hir::Module) -> ast::Type {
821        let ty = self.type_to_string(db, module);
822        make::ty(&ty)
823    }
824
825    pub(crate) fn convert_type_with_factory(
826        &self,
827        make: &SyntaxFactory,
828        db: &'db dyn HirDatabase,
829        module: hir::Module,
830    ) -> ast::Type {
831        let ty = self.type_to_string(db, module);
832        make.ty(&ty)
833    }
834
835    pub(crate) fn getter(&self, make: &SyntaxFactory, field_name: String) -> ast::Expr {
836        let expr = make.expr_field(make.expr_self(), &field_name);
837
838        match self.conversion {
839            ReferenceConversionType::Copy => expr.into(),
840            ReferenceConversionType::AsRefStr
841            | ReferenceConversionType::AsRefSlice
842            | ReferenceConversionType::Dereferenced
843            | ReferenceConversionType::Option
844            | ReferenceConversionType::Result => {
845                if self.impls_deref {
846                    make.expr_ref(expr.into(), false)
847                } else {
848                    make.expr_method_call(expr.into(), make.name_ref("as_ref"), make.arg_list([]))
849                        .into()
850                }
851            }
852        }
853    }
854}

857//        and all users of this function operate on string type names, so they can do the conversion

859pub(crate) fn convert_reference_type<'db>(
860    ty: hir::Type<'db>,
861    db: &'db RootDatabase,
862    famous_defs: &FamousDefs<'_, 'db>,
863) -> Option<ReferenceConversion<'db>> {

865        .or_else(|| handle_as_ref_str(&ty, db, famous_defs))
866        .or_else(|| handle_as_ref_slice(&ty, db, famous_defs))

869        .or_else(|| handle_result_as_ref(&ty, db, famous_defs))
870        .map(|(conversion, impls_deref)| ReferenceConversion { ty, conversion, impls_deref })
871}

960/// Trim(remove leading and trailing whitespace) `initial_range` in `source_file`, return the trimmed range.
961pub(crate) fn trimmed_text_range(source_file: &SourceFile, initial_range: TextRange) -> TextRange {
962    let mut trimmed_range = initial_range;
963    while source_file
964        .syntax()
965        .token_at_offset(trimmed_range.start())
966        .find_map(Whitespace::cast)
967        .is_some()
968        && trimmed_range.start() < trimmed_range.end()
969    {
970        let start = trimmed_range.start() + TextSize::from(1);
971        trimmed_range = TextRange::new(start, trimmed_range.end());
972    }
973    while source_file
974        .syntax()
975        .token_at_offset(trimmed_range.end())
976        .find_map(Whitespace::cast)
977        .is_some()
978        && trimmed_range.start() < trimmed_range.end()
979    {
980        let end = trimmed_range.end() - TextSize::from(1);
981        trimmed_range = TextRange::new(trimmed_range.start(), end);
982    }
983    trimmed_range
984}

1005/// Calculate the number of hashes required for a raw string containing `s`
1006pub(crate) fn required_hashes(s: &str) -> usize {
1007    let mut res = 0usize;
1008    for idx in s.match_indices('"').map(|(i, _)| i) {
1009        let (_, sub) = s.split_at(idx + 1);
1010        let n_hashes = sub.chars().take_while(|c| *c == '#').count();
1011        res = res.max(n_hashes + 1)
1012    }
1013    res
1014}

1016fn test_required_hashes() {
1017    assert_eq!(0, required_hashes("abc"));
1018    assert_eq!(0, required_hashes("###"));
1019    assert_eq!(1, required_hashes("\""));
1020    assert_eq!(2, required_hashes("\"#abc"));
1021    assert_eq!(0, required_hashes("#abc"));
1022    assert_eq!(3, required_hashes("#ab\"##c"));
1023    assert_eq!(5, required_hashes("#ab\"##\"####c"));
1024}

1085/// Creates a token tree list from a syntax node, creating the needed delimited sub token trees.
1086/// Assumes that the input syntax node is a valid syntax tree.
1087pub(crate) fn tt_from_syntax(
1088    node: SyntaxNode,
1089    make: &SyntaxFactory,
1090) -> Vec<NodeOrToken<ast::TokenTree, SyntaxToken>> {
1091    let mut tt_stack = vec![(None, vec![])];
1092
1093    for element in node.descendants_with_tokens() {
1094        let NodeOrToken::Token(token) = element else { continue };
1095
1096        match token.kind() {
1097            T!['('] | T!['{'] | T!['['] => {
1098                // Found an opening delimiter, start a new sub token tree
1099                tt_stack.push((Some(token.kind()), vec![]));
1100            }
1101            T![')'] | T!['}'] | T![']'] => {
1102                // Closing a subtree
1103                let (delimiter, tt) = tt_stack.pop().expect("unbalanced delimiters");
1104                let (_, parent_tt) = tt_stack
1105                    .last_mut()
1106                    .expect("parent token tree was closed before it was completed");
1107                let closing_delimiter = delimiter.map(|it| match it {
1108                    T!['('] => T![')'],
1109                    T!['{'] => T!['}'],
1110                    T!['['] => T![']'],
1111                    _ => unreachable!(),
1112                });
1113                stdx::always!(
1114                    closing_delimiter == Some(token.kind()),
1115                    "mismatched opening and closing delimiters"
1116                );
1117
1118                let sub_tt = make.token_tree(delimiter.expect("unbalanced delimiters"), tt);
1119                parent_tt.push(NodeOrToken::Node(sub_tt));
1120            }
1121            _ => {
1122                let (_, current_tt) = tt_stack.last_mut().expect("unmatched delimiters");
1123                current_tt.push(NodeOrToken::Token(token))
1124            }
1125        }
1126    }
1127
1128    tt_stack.pop().expect("parent token tree was closed before it was completed").1
1129}

1131pub(crate) fn cover_let_chain(mut expr: ast::Expr, range: TextRange) -> Option<ast::Expr> {
1132    if !expr.syntax().text_range().contains_range(range) {
1133        return None;
1134    }
1135    loop {
1136        let (chain_expr, rest) = if let ast::Expr::BinExpr(bin_expr) = &expr
1137            && bin_expr.op_kind() == Some(ast::BinaryOp::LogicOp(ast::LogicOp::And))
1138        {
1139            (bin_expr.rhs(), bin_expr.lhs())
1140        } else {
1141            (Some(expr), None)
1142        };
1143
1144        if let Some(chain_expr) = chain_expr
1145            && chain_expr.syntax().text_range().contains_range(range)
1146        {
1147            break Some(chain_expr);
1148        }
1149        expr = rest?;
1150    }
1151}

1155    range: TextRange,
1156) -> std::ops::RangeInclusive<syntax::SyntaxElement> {
1157    let node = match source.covering_element(range) {
1158        NodeOrToken::Node(node) => node,
1159        NodeOrToken::Token(t) => t.parent().unwrap(),
1160    };
1161    let mut iter = node.children_with_tokens().filter(|it| range.contains_range(it.text_range()));
1162    let first = iter.next().unwrap_or(node.into());
1163    let last = iter.last().unwrap_or_else(|| first.clone());
1164    first..=last
1165}

1176pub fn is_body_const(sema: &Semantics<'_, RootDatabase>, expr: &ast::Expr) -> bool {
1177    let mut is_const = true;
1178    preorder_expr(expr, &mut |ev| {
1179        let expr = match ev {
1180            WalkEvent::Enter(_) if !is_const => return true,
1181            WalkEvent::Enter(expr) => expr,
1182            WalkEvent::Leave(_) => return false,
1183        };
1184        match expr {
1185            ast::Expr::CallExpr(call) => {
1186                if let Some(ast::Expr::PathExpr(path_expr)) = call.expr()
1187                    && let Some(PathResolution::Def(ModuleDef::Function(func))) =
1188                        path_expr.path().and_then(|path| sema.resolve_path(&path))
1189                {
1190                    is_const &= func.is_const(sema.db);
1191                }
1192            }
1193            ast::Expr::MethodCallExpr(call) => {
1194                is_const &=
1195                    sema.resolve_method_call(&call).map(|it| it.is_const(sema.db)).unwrap_or(true)
1196            }
1197            ast::Expr::ForExpr(_)
1198            | ast::Expr::ReturnExpr(_)
1199            | ast::Expr::TryExpr(_)
1200            | ast::Expr::YieldExpr(_)
1201            | ast::Expr::AwaitExpr(_) => is_const = false,
1202            _ => (),
1203        }
1204        !is_const
1205    });
1206    is_const
1207}
