28// Assist: extract_struct_from_enum_variant
29//
30// Extracts a struct from enum variant.
31//
32// ```
33// enum A { $0One(u32, u32) }
34// ```
35// ->
36// ```
37// struct One(u32, u32);
38//
39// enum A { One(One) }
40// ```
41pub(crate) fn extract_struct_from_enum_variant(
42    acc: &mut Assists,
43    ctx: &AssistContext<'_, '_>,
44) -> Option<()> {
45    let variant = ctx.find_node_at_offset::<ast::Variant>()?;
46    let field_list = extract_field_list_if_applicable(&variant)?;
47
48    let variant_name = variant.name()?;
49    let variant_hir = ctx.sema.to_def(&variant)?;
50    if existing_definition(ctx.db(), &variant_name, &variant_hir) {
51        cov_mark::hit!(test_extract_enum_not_applicable_if_struct_exists);
52        return None;
53    }
54
55    let enum_ast = variant.parent_enum();
56    let enum_hir = ctx.sema.to_def(&enum_ast)?;
57    let target = variant.syntax().text_range();
58    acc.add(
59        AssistId::refactor_rewrite("extract_struct_from_enum_variant"),
60        "Extract struct from enum variant",
61        target,
62        |builder| {
63            let editor = builder.make_editor(variant.syntax());
64            let make = editor.make();
65            let edition = enum_hir.krate(ctx.db()).edition(ctx.db());
66            let variant_hir_name = variant_hir.name(ctx.db());
67            let enum_module_def = ModuleDef::from(enum_hir);
68            let usages = Definition::EnumVariant(variant_hir).usages(&ctx.sema).all();
69
70            let mut visited_modules_set = FxHashSet::default();
71            let current_module = enum_hir.module(ctx.db());
72            visited_modules_set.insert(current_module);
73            // record file references of the file the def resides in, we only want to swap to the edited file in the builder once
74            let mut def_file_references = None;
75            for (file_id, references) in usages {
76                if file_id == ctx.file_id() {
77                    def_file_references = Some(references);
78                    continue;
79                }
80                let processed = process_references(
81                    ctx,
82                    &mut visited_modules_set,
83                    &enum_module_def,
84                    &variant_hir_name,
85                    references,
86                );
87                if processed.is_empty() {
88                    continue;
89                }
90                let file_editor = builder.make_editor(processed[0].0.syntax());
91                processed.into_iter().for_each(|(path, node, import)| {
92                    apply_references(
93                        ctx.config.insert_use,
94                        path,
95                        node,
96                        import,
97                        edition,
98                        &file_editor,
99                    )
100                });
101                builder.add_file_edits(file_id.file_id(ctx.db()), file_editor);
102            }
103
104            if let Some(references) = def_file_references {
105                let processed = process_references(
106                    ctx,
107                    &mut visited_modules_set,
108                    &enum_module_def,
109                    &variant_hir_name,
110                    references,
111                );
112                processed.into_iter().for_each(|(path, node, import)| {
113                    apply_references(ctx.config.insert_use, path, node, import, edition, &editor)
114                });
115            }
116
117            let generic_params = enum_ast.generic_param_list().and_then(|known_generics| {
118                extract_generic_params(make, &known_generics, &field_list)
119            });
120
121            // resolve GenericArg in field_list to actual type
122            let field_list = if let Some((target_scope, source_scope)) =
123                ctx.sema.scope(enum_ast.syntax()).zip(ctx.sema.scope(field_list.syntax()))
124            {
125                let field_list = field_list.reset_indent();
126                let field_list =
127                    PathTransform::generic_transformation(&target_scope, &source_scope)
128                        .apply(field_list.syntax());
129                match_ast! {
130                    match field_list {
131                        ast::RecordFieldList(field_list) => Either::Left(field_list),
132                        ast::TupleFieldList(field_list) => Either::Right(field_list),
133                        _ => unreachable!(),
134                    }
135                }
136            } else {
137                field_list.clone()
138            };
139
140            let (comments_for_struct, comments_to_delete) =
141                collect_variant_comments(make, variant.syntax());
142            for element in &comments_to_delete {
143                editor.delete(element.clone());
144            }
145
146            let def = create_struct_def(
147                make,
148                variant_name.clone(),
149                &field_list,
150                generic_params.clone(),
151                &enum_ast,
152            );
153
154            let enum_ast = variant.parent_enum();
155            let indent = enum_ast.indent_level();
156            let def = def.indent(indent);
157
158            let mut insert_items: Vec<SyntaxElement> = Vec::new();
159            for attr in enum_ast.attrs() {
160                insert_items.push(attr.syntax().clone().into());
161                insert_items.push(make.whitespace("\n").into());
162            }
163            insert_items.extend(comments_for_struct);
164            insert_items.push(def.syntax().clone().into());
165            insert_items.push(make.whitespace(&format!("\n\n{indent}")).into());
166            editor.insert_all_with_whitespace(Position::before(enum_ast.syntax()), insert_items);
167
168            update_variant(&editor, &variant, generic_params);
169
170            builder.add_file_edits(ctx.vfs_file_id(), editor);
171        },
172    )
173}

175fn extract_field_list_if_applicable(
176    variant: &ast::Variant,
177) -> Option<Either<ast::RecordFieldList, ast::TupleFieldList>> {
178    match variant.kind() {
179        ast::StructKind::Record(field_list) if field_list.fields().next().is_some() => {
180            Some(Either::Left(field_list))
181        }
182        ast::StructKind::Tuple(field_list) if field_list.fields().count() > 1 => {
183            Some(Either::Right(field_list))
184        }
185        _ => None,
186    }
187}

189fn existing_definition(db: &RootDatabase, variant_name: &ast::Name, variant: &EnumVariant) -> bool {
190    variant
191        .parent_enum(db)
192        .module(db)
193        .scope(db, None)
194        .into_iter()
195        .filter(|(_, def)| match def {
196            // only check type-namespace
197            hir::ScopeDef::ModuleDef(def) => matches!(
198                def,
199                ModuleDef::Module(_)
200                    | ModuleDef::Adt(_)
201                    | ModuleDef::EnumVariant(_)
202                    | ModuleDef::Trait(_)
203                    | ModuleDef::TypeAlias(_)
204                    | ModuleDef::BuiltinType(_)
205            ),
206            _ => false,
207        })
208        .any(|(name, _)| name.as_str() == variant_name.text().trim_start_matches("r#"))
209}

211fn extract_generic_params(
212    make: &SyntaxFactory,
213    known_generics: &ast::GenericParamList,
214    field_list: &Either<ast::RecordFieldList, ast::TupleFieldList>,
215) -> Option<ast::GenericParamList> {
216    let mut generics = known_generics.generic_params().map(|param| (param, false)).collect_vec();
217
218    #[expect(clippy::unnecessary_fold, reason = "this function has side effects")]
219    let tagged_one = match field_list {
220        Either::Left(field_list) => field_list
221            .fields()
222            .filter_map(|f| f.ty())
223            .fold(false, |tagged, ty| tag_generics_in_variant(&ty, &mut generics) || tagged),
224        Either::Right(field_list) => field_list
225            .fields()
226            .filter_map(|f| f.ty())
227            .fold(false, |tagged, ty| tag_generics_in_variant(&ty, &mut generics) || tagged),
228    };
229
230    let generics = generics.into_iter().filter_map(|(param, tag)| tag.then_some(param));
231    tagged_one.then(|| make.generic_param_list(generics))
232}
233
234fn tag_generics_in_variant(ty: &ast::Type, generics: &mut [(ast::GenericParam, bool)]) -> bool {
235    let mut tagged_one = false;
236
237    for token in ty.syntax().descendants_with_tokens().filter_map(SyntaxElement::into_token) {
238        for (param, tag) in generics.iter_mut().filter(|(_, tag)| !tag) {
239            match param {
240                ast::GenericParam::LifetimeParam(lt)
241                    if matches!(token.kind(), T![lifetime_ident]) =>
242                {
243                    if let Some(lt) = lt.lifetime()
244                        && lt.text() == token.text()
245                    {
246                        *tag = true;
247                        tagged_one = true;
248                        break;
249                    }
250                }
251                param if matches!(token.kind(), T![ident]) => {
252                    if match param {
253                        ast::GenericParam::ConstParam(konst) => {
254                            konst.name().map(|name| name.text() == token.text()).unwrap_or_default()
255                        }
256                        ast::GenericParam::TypeParam(ty) => {
257                            ty.name().map(|name| name.text() == token.text()).unwrap_or_default()
258                        }
259                        ast::GenericParam::LifetimeParam(lt) => {
260                            lt.lifetime().map(|lt| lt.text() == token.text()).unwrap_or_default()
261                        }
262                    } {
263                        *tag = true;
264                        tagged_one = true;
265                        break;
266                    }
267                }
268                _ => (),
269            }
270        }
271    }
272
273    tagged_one
274}

276fn create_struct_def(
277    make: &SyntaxFactory,
278    name: ast::Name,
279    field_list: &Either<ast::RecordFieldList, ast::TupleFieldList>,
280    generics: Option<ast::GenericParamList>,
281    enum_: &ast::Enum,
282) -> ast::Struct {
283    let enum_vis = enum_.visibility();
284
285    // for fields without any existing visibility, use visibility of enum
286    let field_list: ast::FieldList = match field_list {
287        Either::Left(field_list) => {
288            if let Some(vis) = &enum_vis {
289                let new_fields = field_list.fields().map(|field| {
290                    if field.visibility().is_none()
291                        && let Some(name) = field.name()
292                        && let Some(ty) = field.ty()
293                    {
294                        make.record_field(Some(vis.clone()), name, ty)
295                    } else {
296                        field
297                    }
298                });
299                make.record_field_list(new_fields).into()
300            } else {
301                field_list.clone().into()
302            }
303        }
304        Either::Right(field_list) => {
305            if let Some(vis) = &enum_vis {
306                let new_fields = field_list.fields().map(|field| {
307                    if field.visibility().is_none()
308                        && let Some(ty) = field.ty()
309                    {
310                        make.tuple_field(Some(vis.clone()), ty)
311                    } else {
312                        field
313                    }
314                });
315                make.tuple_field_list(new_fields).into()
316            } else {
317                field_list.clone().into()
318            }
319        }
320    };
321
322    make.struct_(enum_vis, name, generics, field_list)
323}

325fn update_variant(
326    editor: &SyntaxEditor,
327    variant: &ast::Variant,
328    generics: Option<ast::GenericParamList>,
329) -> Option<()> {
330    let make = editor.make();
331    let name = variant.name()?;
332    let generic_args = generics
333        .filter(|generics| generics.generic_params().count() > 0)
334        .map(|generics| generics.to_generic_args(make));
335    // FIXME: replace with a `ast::make` constructor
336    let ty = match generic_args {
337        Some(generic_args) => make.ty(&format!("{name}{generic_args}")),
338        None => make.ty(name.text()),
339    };
340
341    // change from a record to a tuple field list
342    let tuple_field = make.tuple_field(None, ty);
343    let field_list = make.tuple_field_list(iter::once(tuple_field));
344    editor.replace(variant.field_list()?.syntax(), field_list.syntax());
345
346    // remove any ws after the name
347    if let Some(ws) = name
348        .syntax()
349        .siblings_with_tokens(syntax::Direction::Next)
350        .find_map(|tok| tok.into_token().filter(|tok| tok.kind() == WHITESPACE))
351    {
352        editor.delete(ws);
353    }
354
355    Some(())
356}

358fn collect_variant_comments(
359    make: &SyntaxFactory,
360    node: &SyntaxNode,
361) -> (Vec<SyntaxElement>, Vec<SyntaxElement>) {
362    let mut to_insert: Vec<SyntaxElement> = Vec::new();
363    let mut to_delete: Vec<SyntaxElement> = Vec::new();
364    let mut after_comment = false;
365
366    for child in node.children_with_tokens() {
367        match child.kind() {
368            COMMENT | DOC_COMMENT => {
369                after_comment = true;
370                to_insert.push(child.clone());
371                to_delete.push(child);
372            }
373            WHITESPACE if after_comment => {
374                after_comment = false;
375                to_insert.push(make.whitespace("\n").into());
376                to_delete.push(child);
377            }
378            _ => {
379                after_comment = false;
380            }
381        }
382    }
383
384    (to_insert, to_delete)
385}

387fn apply_references(
388    insert_use_cfg: InsertUseConfig,
389    segment: ast::PathSegment,
390    node: SyntaxNode,
391    import: Option<(ImportScope, hir::ModPath)>,
392    edition: Edition,
393    editor: &SyntaxEditor,
394) {
395    let make = editor.make();
396    if let Some((scope, path)) = import {
397        insert_use_with_editor(
398            &scope,
399            mod_path_to_ast_with_factory(make, &path, edition),
400            &insert_use_cfg,
401            editor,
402        );
403    }
404    // deep clone to prevent cycle
405    let path = make.path_from_segments(iter::once(segment.clone()), false);
406    editor.insert(Position::before(segment.syntax()), make.token(T!['(']));
407    editor.insert(Position::before(segment.syntax()), path.syntax());
408    editor.insert(Position::after(&node), make.token(T![')']));
409}

411fn process_references(
412    ctx: &AssistContext<'_, '_>,
413    visited_modules: &mut FxHashSet<Module>,
414    enum_module_def: &ModuleDef,
415    variant_hir_name: &Name,
416    refs: Vec<FileReference>,
417) -> Vec<(ast::PathSegment, SyntaxNode, Option<(ImportScope, hir::ModPath)>)> {
418    // we have to recollect here eagerly as we are about to edit the tree we need to calculate the changes
419    // and corresponding nodes up front
420    refs.into_iter()
421        .flat_map(|reference| {
422            let (segment, scope_node, module) = reference_to_node(&ctx.sema, reference)?;
423            if !visited_modules.contains(&module) {
424                let cfg =
425                    ctx.config.find_path_config(ctx.sema.is_nightly(module.krate(ctx.sema.db)));
426                let mod_path = module.find_use_path(
427                    ctx.sema.db,
428                    *enum_module_def,
429                    ctx.config.insert_use.prefix_kind,
430                    cfg,
431                );
432                if let Some(mut mod_path) = mod_path {
433                    mod_path.pop_segment();
434                    mod_path.push_segment(variant_hir_name.clone());
435                    let scope = ImportScope::find_insert_use_container(&scope_node, &ctx.sema)?;
436                    visited_modules.insert(module);
437                    return Some((segment, scope_node, Some((scope, mod_path))));
438                }
439            }
440            Some((segment, scope_node, None))
441        })
442        .collect()
443}

445fn reference_to_node(
446    sema: &hir::Semantics<'_, RootDatabase>,
447    reference: FileReference,
448) -> Option<(ast::PathSegment, SyntaxNode, hir::Module)> {
449    let segment =
450        reference.name.as_name_ref()?.syntax().parent().and_then(ast::PathSegment::cast)?;
451
452    // filter out the reference in marco
453    let segment_range = segment.syntax().text_range();
454    if segment_range != reference.range {
455        return None;
456    }
457
458    let parent = segment.parent_path().syntax().parent()?;
459    let expr_or_pat = match_ast! {
460        match parent {
461            ast::PathExpr(_it) => parent.parent()?,
462            ast::RecordExpr(_it) => parent,
463            ast::TupleStructPat(_it) => parent,
464            ast::RecordPat(_it) => parent,
465            _ => return None,
466        }
467    };
468    let module = sema.scope(&expr_or_pat)?.module();
469    Some((segment, expr_or_pat, module))
470}
