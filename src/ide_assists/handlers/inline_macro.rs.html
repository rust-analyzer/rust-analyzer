9// Assist: inline_macro
10//
11// Takes a macro and inlines it one step.
12//
13// ```
14// macro_rules! num {
15//     (+$($t:tt)+) => (1 + num!($($t )+));
16//     (-$($t:tt)+) => (-1 + num!($($t )+));
17//     (+) => (1);
18//     (-) => (-1);
19// }
20//
21// fn main() {
22//     let number = num$0!(+ + + - + +);
23//     println!("{number}");
24// }
25// ```
26// ->
27// ```
28// macro_rules! num {
29//     (+$($t:tt)+) => (1 + num!($($t )+));
30//     (-$($t:tt)+) => (-1 + num!($($t )+));
31//     (+) => (1);
32//     (-) => (-1);
33// }
34//
35// fn main() {
36//     let number = 1+num!(+ + - + +);
37//     println!("{number}");
38// }
39// ```
40pub(crate) fn inline_macro(acc: &mut Assists, ctx: &AssistContext<'_, '_>) -> Option<()> {
41    let source = ctx.source_file().syntax();
42    let sel = ctx.selection_trimmed();
43    let (macro_call, text_range) = ctx
44        .sema
45        .find_nodes_at_offset_with_descend::<ast::MacroCall>(source, ctx.offset())
46        .find_map(|macro_call_node| {
47            let macro_call = ctx.sema.to_def(&macro_call_node)?;
48            let original_range =
49                original_range_in(ctx.file_id(), &ctx.sema, macro_call_node.syntax())?;
50            original_range.contains_range(sel).then_some((macro_call, original_range))
51        })?;
52    let target_crate_id = ctx.sema.file_to_module_def(ctx.vfs_file_id())?.krate(ctx.db()).into();
53
54    acc.add(
55        AssistId::refactor_inline("inline_macro"),
56        "Inline macro".to_owned(),
57        text_range,
58        |builder| {
59            let expanded = ctx.sema.parse_or_expand(macro_call.into());
60            let span_map = macro_call.expansion_span_map(ctx.sema.db);
61            // Don't call `prettify_macro_expansion()` outside the actual assist action; it does some heavy rowan tree manipulation,
62            // which can be very costly for big macros when it is done *even without the assist being invoked*.
63            let expanded = prettify_macro_expansion(ctx.db(), expanded, span_map, target_crate_id);
64
65            // macro_call is from an expansion, use source position for indent
66            let indent = source
67                .token_at_offset(text_range.start())
68                .right_biased()
69                .map_or_else(IndentLevel::zero, |t| IndentLevel::from_token(&t));
70            let expanded = ast::edit::indent(&expanded, indent);
71
72            let editor = builder.make_editor(source);
73            let place = cover_edit_range(source, text_range);
74            editor.replace_all(place, vec![expanded.into()]);
75            builder.add_file_edits(ctx.vfs_file_id(), editor);
76        },
77    )
78}
