8// Assist: add_lifetime_to_type
9//
10// Adds a new lifetime to a struct, enum or union.
11//
12// ```
13// struct Point {
14//     x: &$0u32,
15//     y: u32,
16// }
17// ```
18// ->
19// ```
20// struct Point<'a> {
21//     x: &'a u32,
22//     y: u32,
23// }
24// ```
25pub(crate) fn add_lifetime_to_type(acc: &mut Assists, ctx: &AssistContext<'_, '_>) -> Option<()> {
26    let ref_type_focused = ctx.find_node_at_offset::<ast::RefType>()?;
27    if ref_type_focused.lifetime().is_some_and(|lifetime| lifetime.text() != "'_") {
28        return None;
29    }
30
31    let node = ctx.find_node_at_offset::<ast::Adt>()?;
32    let has_lifetime = node
33        .generic_param_list()
34        .is_some_and(|gen_list| gen_list.lifetime_params().next().is_some());
35
36    if has_lifetime {
37        return None;
38    }
39
40    let changes = fetch_borrowed_types(&node)?;
41    let target = node.syntax().text_range();
42
43    acc.add(AssistId::quick_fix("add_lifetime_to_type"), "Add lifetime", target, |builder| {
44        match node.generic_param_list() {
45            Some(gen_param) => {
46                if let Some(left_angle) = gen_param.l_angle_token() {
47                    builder.insert(left_angle.text_range().end(), "'a, ");
48                }
49            }
50            None => {
51                if let Some(name) = node.name() {
52                    builder.insert(name.syntax().text_range().end(), "<'a>");
53                }
54            }
55        }
56
57        for change in changes {
58            match change {
59                Change::Replace(it) => {
60                    builder.replace(it.text_range(), "'a");
61                }
62                Change::Insert(it) => {
63                    builder.insert(it.text_range().end(), "'a ");
64                }
65            }
66        }
67    })
68}
69
70fn fetch_borrowed_types(node: &ast::Adt) -> Option<Vec<Change>> {
71    let ref_types: Vec<_> = match node {
72        ast::Adt::Enum(enum_) => {
73            let variant_list = enum_.variant_list()?;
74            variant_list
75                .variants()
76                .filter_map(|variant| {
77                    let field_list = variant.field_list()?;
78
79                    find_ref_types_from_field_list(&field_list)
80                })
81                .flatten()
82                .collect()
83        }
84        ast::Adt::Struct(strukt) => {
85            let field_list = strukt.field_list()?;
86            find_ref_types_from_field_list(&field_list)?
87        }
88        ast::Adt::Union(un) => {
89            let record_field_list = un.record_field_list()?;
90            find_ref_types_from_field_list(&record_field_list.into())?
91        }
92    };
93
94    if ref_types.is_empty() { None } else { Some(ref_types) }
95}
96
97fn find_ref_types_from_field_list(field_list: &ast::FieldList) -> Option<Vec<Change>> {
98    let ref_types: Vec<_> = match field_list {
99        ast::FieldList::RecordFieldList(record_list) => {
100            record_list.fields().flat_map(|f| infer_lifetimes(f.syntax())).collect()
101        }
102        ast::FieldList::TupleFieldList(tuple_field_list) => {
103            tuple_field_list.fields().flat_map(|f| infer_lifetimes(f.syntax())).collect()
104        }
105    };
106
107    if ref_types.is_empty() { None } else { Some(ref_types) }
108}
109
110enum Change {
111    Replace(SyntaxToken),
112    Insert(SyntaxToken),
113}
114
115fn infer_lifetimes(node: &SyntaxNode) -> Vec<Change> {
116    node.children()
117        .filter(|it| !matches!(it.kind(), SyntaxKind::FN_PTR_TYPE | SyntaxKind::TYPE_BOUND_LIST))
118        .flat_map(|it| {
119            infer_lifetimes(&it)
120                .into_iter()
121                .chain(ast::Lifetime::cast(it.clone()).and_then(|lt| {
122                    lt.lifetime_ident_token().filter(|lt| lt.text() == "'_").map(Change::Replace)
123                }))
124                .chain(
125                    ast::RefType::cast(it)
126                        .filter(|ty| ty.lifetime().is_none())
127                        .and_then(|ty| ty.amp_token())
128                        .map(Change::Insert),
129                )
130        })
131        .collect()
132}

140    #[test]
141    fn add_lifetime_to_struct() {
142        check_assist(
143            add_lifetime_to_type,
144            r#"struct Foo { a: &$0i32 }"#,
145            r#"struct Foo<'a> { a: &'a i32 }"#,
146        );
147
148        check_assist(
149            add_lifetime_to_type,
150            r#"struct Foo { a: &$0i32, b: &usize }"#,
151            r#"struct Foo<'a> { a: &'a i32, b: &'a usize }"#,
152        );
153
154        check_assist(
155            add_lifetime_to_type,
156            r#"struct Foo { a: &$0i32, b: usize }"#,
157            r#"struct Foo<'a> { a: &'a i32, b: usize }"#,
158        );
159
160        check_assist(
161            add_lifetime_to_type,
162            r#"struct Foo<T> { a: &$0T, b: usize }"#,
163            r#"struct Foo<'a, T> { a: &'a T, b: usize }"#,
164        );
165
166        check_assist_not_applicable(add_lifetime_to_type, r#"struct Foo<'a> { a: &$0'a i32 }"#);
167        check_assist_not_applicable(add_lifetime_to_type, r#"struct Foo { a: &'a$0 i32 }"#);
168    }
