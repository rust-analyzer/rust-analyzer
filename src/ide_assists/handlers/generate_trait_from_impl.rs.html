17// Assist: generate_trait_from_impl
18//
19// Generate trait for an already defined inherent impl and convert impl to a trait impl.
20//
21// ```
22// struct Foo<const N: usize>([i32; N]);
23//
24// macro_rules! const_maker {
25//     ($t:ty, $v:tt) => {
26//         const CONST: $t = $v;
27//     };
28// }
29//
30// impl<const N: usize> Fo$0o<N> {
31//     // Used as an associated constant.
32//     const CONST_ASSOC: usize = N * 4;
33//
34//     fn create() -> Option<()> {
35//         Some(())
36//     }
37//
38//     const_maker! {i32, 7}
39// }
40// ```
41// ->
42// ```
43// struct Foo<const N: usize>([i32; N]);
44//
45// macro_rules! const_maker {
46//     ($t:ty, $v:tt) => {
47//         const CONST: $t = $v;
48//     };
49// }
50//
51// trait ${0:Create}<const N: usize> {
52//     // Used as an associated constant.
53//     const CONST_ASSOC: usize = N * 4;
54//
55//     fn create() -> Option<()>;
56//
57//     const_maker! {i32, 7}
58// }
59//
60// impl<const N: usize> ${0:Create}<N> for Foo<N> {
61//     // Used as an associated constant.
62//     const CONST_ASSOC: usize = N * 4;
63//
64//     fn create() -> Option<()> {
65//         Some(())
66//     }
67//
68//     const_maker! {i32, 7}
69// }
70// ```
71pub(crate) fn generate_trait_from_impl(
72    acc: &mut Assists,
73    ctx: &AssistContext<'_, '_>,
74) -> Option<()> {
75    // Get AST Node
76    let impl_ast = ctx.find_node_at_offset::<ast::Impl>()?;
77
78    // Check if cursor is to the left of assoc item list's L_CURLY.
79    // if no L_CURLY then return.
80    let l_curly = impl_ast.assoc_item_list()?.l_curly_token()?;
81
82    let cursor_offset = ctx.offset();
83    let l_curly_offset = l_curly.text_range();
84    if cursor_offset >= l_curly_offset.start() {
85        return None;
86    }
87
88    // If impl is not inherent then we don't really need to go any further.
89    if impl_ast.for_token().is_some() {
90        return None;
91    }
92
93    let impl_assoc_items = impl_ast.assoc_item_list()?;
94    let first_element = impl_assoc_items.assoc_items().next();
95    first_element.as_ref()?;
96
97    let impl_name = impl_ast.self_ty()?;
98
99    acc.add(
100        AssistId::generate("generate_trait_from_impl"),
101        "Generate trait from impl",
102        impl_ast.syntax().text_range(),
103        |builder| {
104            let trait_items: ast::AssocItemList = {
105                let (trait_items_editor, trait_items) =
106                    SyntaxEditor::with_ast_node(&impl_assoc_items);
107
108                trait_items.assoc_items().for_each(|item| {
109                    strip_body(&trait_items_editor, &item);
110                    remove_items_visibility(&trait_items_editor, &item);
111                });
112                ast::AssocItemList::cast(trait_items_editor.finish().new_root().clone()).unwrap()
113            };
114
115            let editor = builder.make_editor(impl_ast.syntax());
116            let make = editor.make();
117            let params = used_params(&impl_ast, make, ctx);
118            let trait_ast = make.trait_(
119                false,
120                trait_name(&impl_assoc_items, make).text(),
121                params.clone(),
122                impl_ast.where_clause(),
123                trait_items,
124            );
125
126            let trait_name = trait_ast.name().expect("new trait should have a name");
127            let trait_name_ref = make.name_ref(&trait_name.to_string());
128
129            // Change `impl Foo` to `impl NewTrait for Foo`
130            let mut elements = vec![
131                trait_name_ref.syntax().clone().into(),
132                make.whitespace(" ").into(),
133                make.token(T![for]).into(),
134                make.whitespace(" ").into(),
135            ];
136
137            if let Some(params) = params {
138                let gen_args = &params.to_generic_args(make);
139                elements.insert(1, gen_args.syntax().clone().into());
140            }
141
142            impl_assoc_items.assoc_items().for_each(|item| {
143                remove_items_visibility(&editor, &item);
144                remove_doc_comments(&editor, &item);
145            });
146
147            editor.insert_all(Position::before(impl_name.syntax()), elements);
148
149            // Insert trait before TraitImpl
150            editor.insert_all(
151                Position::before(impl_ast.syntax()),
152                vec![
153                    trait_ast.syntax().clone().into(),
154                    make.whitespace(&format!("\n\n{}", impl_ast.indent_level())).into(),
155                ],
156            );
157
158            // Link the trait name & trait ref names together as a placeholder snippet group
159            if let Some(cap) = ctx.config.snippet_cap {
160                let placeholder = builder.make_placeholder_snippet(cap);
161                editor.add_annotation(trait_name.syntax(), placeholder);
162                editor.add_annotation(trait_name_ref.syntax(), placeholder);
163            }
164            builder.add_file_edits(ctx.vfs_file_id(), editor);
165        },
166    );
167
168    Some(())
169}

171fn used_params(
172    impl_ast: &ast::Impl,
173    make: &SyntaxFactory,
174    ctx: &AssistContext<'_, '_>,
175) -> Option<ast::GenericParamList> {
176    let impl_only_ranges = impl_ast
177        .assoc_item_list()
178        .into_iter()
179        .flat_map(|list| list.assoc_items())
180        .filter_map(|item| match item {
181            ast::AssocItem::Fn(f) => Some(f.body()?.syntax().text_range()),
182            _ => None,
183        })
184        .chain(impl_ast.self_ty().map(|it| it.syntax().text_range()))
185        .collect::<Vec<_>>();
186    let used_in_impl = |param: &ast::GenericParam| {
187        let Some(def) = ctx.sema.to_def(param) else { return true };
188        Definition::GenericParam(def)
189            .usages(&ctx.sema)
190            .in_scope(&SearchScope::single_file(ctx.file_id()))
191            .all()
192            .file_ranges()
193            .any(|it| !impl_only_ranges.iter().any(|range| range.contains_range(it.range)))
194    };
195    let params = impl_ast.generic_param_list()?;
196    let mut params = params.generic_params().filter(used_in_impl).peekable();
197    params.peek().is_some().then(|| make.generic_param_list(params))
198}

200fn trait_name(items: &ast::AssocItemList, make: &SyntaxFactory) -> ast::Name {
201    let mut fn_names = items
202        .assoc_items()
203        .filter_map(|x| if let ast::AssocItem::Fn(f) = x { f.name() } else { None });
204    fn_names
205        .next()
206        .and_then(|name| {
207            fn_names.next().is_none().then(|| make.name(&stdx::to_camel_case(name.text())))
208        })
209        .unwrap_or_else(|| make.name("NewTrait"))
210}

212/// `E0449` Trait items always share the visibility of their trait
213fn remove_items_visibility(editor: &SyntaxEditor, item: &ast::AssocItem) {
214    if let Some(has_vis) = ast::AnyHasVisibility::cast(item.syntax().clone()) {
215        if let Some(vis) = has_vis.visibility()
216            && let Some(token) = vis.syntax().next_sibling_or_token()
217            && token.kind() == SyntaxKind::WHITESPACE
218        {
219            editor.delete(token);
220        }
221        if let Some(vis) = has_vis.visibility() {
222            editor.delete(vis.syntax());
223        }
224    }
225}
226
227fn remove_doc_comments(editor: &SyntaxEditor, item: &ast::AssocItem) {
228    for doc in item.doc_comments() {
229        if let Some(next) = doc.syntax().last_token().and_then(|it| it.next_token())
230            && next.kind() == SyntaxKind::WHITESPACE
231        {
232            editor.delete(next);
233        }
234        editor.delete(doc.syntax());
235    }
236}
237
238fn strip_body(editor: &SyntaxEditor, item: &ast::AssocItem) {
239    let make = editor.make();
240    if let ast::AssocItem::Fn(f) = item
241        && let Some(body) = f.body()
242    {
243        // In contrast to function bodies, we want to see no ws before a semicolon.
244        // So let's remove them if we see any.
245        if let Some(prev) = body.syntax().prev_sibling_or_token()
246            && prev.kind() == SyntaxKind::WHITESPACE
247        {
248            editor.delete(prev);
249        }
250
251        editor.replace(body.syntax(), make.token(T![;]));
252    };
253}
