17// Assist: add_missing_match_arms
18//
19// Adds missing clauses to a `match` expression.
20//
21// ```
22// enum Action { Move { distance: u32 }, Stop }
23//
24// fn handle(action: Action) {
25//     match action {
26//         $0
27//     }
28// }
29// ```
30// ->
31// ```
32// enum Action { Move { distance: u32 }, Stop }
33//
34// fn handle(action: Action) {
35//     match action {
36//         Action::Move { distance } => ${1:todo!()},
37//         Action::Stop => ${2:todo!()},$0
38//     }
39// }
40// ```
41pub(crate) fn add_missing_match_arms(acc: &mut Assists, ctx: &AssistContext<'_, '_>) -> Option<()> {
42    let match_expr = ctx.find_node_at_offset_with_descend::<ast::MatchExpr>()?;
43    let match_arm_list = match_expr.match_arm_list()?;
44    let arm_list_range = ctx.sema.original_range_opt(match_arm_list.syntax())?;
45
46    if cursor_at_trivial_match_arm_list(ctx, &match_expr, &match_arm_list).is_none() {
47        let cursor_in_range = arm_list_range.range.contains_range(ctx.selection_trimmed());
48        if cursor_in_range {
49            cov_mark::hit!(not_applicable_outside_of_range_right);
50            return None;
51        }
52    }
53
54    let expr = match_expr.expr()?;
55
56    let mut has_catch_all_arm = false;
57
58    let top_lvl_pats: Vec<_> = match_arm_list
59        .arms()
60        .filter_map(|arm| Some((arm.pat()?, arm.guard().is_some())))
61        .flat_map(|(pat, has_guard)| {
62            match pat {
63                // Special case OrPat as separate top-level pats
64                Pat::OrPat(or_pat) => Either::Left(or_pat.pats()),
65                _ => Either::Right(iter::once(pat)),
66            }
67            .map(move |pat| (pat, has_guard))
68        })
69        .filter_map(|(pat, has_guard)| {
70            has_catch_all_arm |= !has_guard && matches!(pat, Pat::WildcardPat(_));
71            (!has_guard).then_some(pat)
72        })
73        // Exclude top level wildcards so that they are expanded by this assist, retains status quo in #8129.
74        .filter(|pat| !matches!(pat, Pat::WildcardPat(_)))
75        .collect();
76
77    let make = SyntaxFactory::without_mappings();
78
79    let scope = ctx.sema.scope(expr.syntax())?;
80    let module = scope.module();
81    let cfg = ctx.config.find_path_config(ctx.sema.is_nightly(scope.krate()));
82    let self_ty = if ctx.config.prefer_self_ty {
83        scope
84            .expression_store_owner()
85            .and_then(|def| {
86                match def {
87                    hir::ExpressionStoreOwner::Body(def_with_body) => {
88                        def_with_body.as_assoc_item(ctx.db())
89                    }
90                    hir::ExpressionStoreOwner::Signature(def) => def.as_assoc_item(ctx.db()),
91                    hir::ExpressionStoreOwner::VariantFields(_) => None,
92                }?
93                .implementing_ty(ctx.db())
94            })
95            .map(|self_ty| {
96                if let Some(owner) = scope.generic_def() {
97                    self_ty.try_rebase_into_owner(ctx.db(), owner).unwrap()
98                } else {
99                    self_ty
100                }
101            })
102    } else {
103        None
104    };
105    let (missing_pats, is_non_exhaustive, has_hidden_variants): (
106        Vec<(ast::Pat, bool)>,
107        bool,
108        bool,
109    ) = if let Some(enum_def) = resolve_enum_def(&ctx.sema, &expr, self_ty.as_ref()) {
110        let is_non_exhaustive = enum_def.is_non_exhaustive(ctx.db(), module.krate(ctx.db()));
111
112        let variants = enum_def.variants(ctx.db());
113
114        let has_hidden_variants = variants
115            .iter()
116            .any(|variant| variant.should_be_hidden(ctx.db(), module.krate(ctx.db())));
117
118        let missing_pats = variants
119            .into_iter()
120            .filter_map(|variant| {
121                Some((
122                    build_pat(ctx, &make, module, variant, cfg)?,
123                    variant.should_be_hidden(ctx.db(), module.krate(ctx.db())),
124                ))
125            })
126            .filter(|(variant_pat, _)| is_variant_missing(&top_lvl_pats, variant_pat));
127
128        let option_enum = FamousDefs(&ctx.sema, module.krate(ctx.db())).core_option_Option();
129        let missing_pats: Vec<_> = if matches!(enum_def, ExtendedEnum::Enum { enum_: e, .. } if Some(e) == option_enum)
130        {
131            // Match `Some` variant first.
132            cov_mark::hit!(option_order);
133            missing_pats.rev().collect()
134        } else {
135            missing_pats.collect()
136        };
137        (missing_pats, is_non_exhaustive, has_hidden_variants)
138    } else if let Some(enum_defs) = resolve_tuple_of_enum_def(&ctx.sema, &expr, self_ty.as_ref()) {
139        let is_non_exhaustive = enum_defs
140            .iter()
141            .any(|enum_def| enum_def.is_non_exhaustive(ctx.db(), module.krate(ctx.db())));
142
143        let mut n_arms = 1;
144        let variants_of_enums: Vec<Vec<ExtendedVariant>> = enum_defs
145            .into_iter()
146            .map(|enum_def| enum_def.variants(ctx.db()))
147            .inspect(|variants| n_arms *= variants.len())
148            .collect();
149
150        // When calculating the match arms for a tuple of enums, we want
151        // to create a match arm for each possible combination of enum
152        // values. The `multi_cartesian_product` method transforms
153        // Vec<Vec<EnumVariant>> into Vec<(EnumVariant, .., EnumVariant)>
154        // where each tuple represents a proposed match arm.
155
156        // A number of arms grows very fast on even a small tuple of large enums.
157        // We skip the assist beyond an arbitrary threshold.
158        if n_arms > 256 {
159            return None;
160        }
161
162        let has_hidden_variants = variants_of_enums
163            .iter()
164            .flatten()
165            .any(|variant| variant.should_be_hidden(ctx.db(), module.krate(ctx.db())));
166
167        let missing_pats = variants_of_enums
168            .into_iter()
169            .multi_cartesian_product()
170            .inspect(|_| cov_mark::hit!(add_missing_match_arms_lazy_computation))
171            .map(|variants| {
172                let is_hidden = variants
173                    .iter()
174                    .any(|variant| variant.should_be_hidden(ctx.db(), module.krate(ctx.db())));
175                let patterns = variants
176                    .into_iter()
177                    .filter_map(|variant| build_pat(ctx, &make, module, variant, cfg));
178
179                (ast::Pat::from(make.tuple_pat(patterns)), is_hidden)
180            })
181            .filter(|(variant_pat, _)| is_variant_missing(&top_lvl_pats, variant_pat))
182            .collect();
183        (missing_pats, is_non_exhaustive, has_hidden_variants)
184    } else if let Some((enum_def, len)) =
185        resolve_array_of_enum_def(&ctx.sema, &expr, self_ty.as_ref())
186    {
187        let is_non_exhaustive = enum_def.is_non_exhaustive(ctx.db(), module.krate(ctx.db()));
188        let variants = enum_def.variants(ctx.db());
189
190        if len.pow(variants.len() as u32) > 256 {
191            return None;
192        }
193
194        let has_hidden_variants = variants
195            .iter()
196            .any(|variant| variant.should_be_hidden(ctx.db(), module.krate(ctx.db())));
197
198        let variants_of_enums = vec![variants; len];
199
200        let missing_pats = variants_of_enums
201            .into_iter()
202            .multi_cartesian_product()
203            .inspect(|_| cov_mark::hit!(add_missing_match_arms_lazy_computation))
204            .map(|variants| {
205                let is_hidden = variants
206                    .iter()
207                    .any(|variant| variant.should_be_hidden(ctx.db(), module.krate(ctx.db())));
208                let patterns = variants
209                    .into_iter()
210                    .filter_map(|variant| build_pat(ctx, &make, module, variant, cfg));
211
212                (ast::Pat::from(make.slice_pat(patterns)), is_hidden)
213            })
214            .filter(|(variant_pat, _)| is_variant_missing(&top_lvl_pats, variant_pat))
215            .collect();
216        (missing_pats, is_non_exhaustive, has_hidden_variants)
217    } else {
218        return None;
219    };
220
221    let mut needs_catch_all_arm = is_non_exhaustive && !has_catch_all_arm;
222
223    if !needs_catch_all_arm
224        && ((has_hidden_variants && has_catch_all_arm) || missing_pats.is_empty())
225    {
226        return None;
227    }
228
229    let visible_count = missing_pats.iter().filter(|(_, hidden)| !hidden).count();
230    let label = if visible_count == 0 {
231        "Add missing catch-all match arm `_`".to_owned()
232    } else if visible_count == 1 {
233        let pat = &missing_pats.iter().find(|(_, hidden)| !hidden).unwrap().0;
234        format!("Add missing match arm `{pat}`")
235    } else {
236        format!("Add {visible_count} missing match arms")
237    };
238
239    acc.add(
240        AssistId::quick_fix("add_missing_match_arms"),
241        label,
242        ctx.sema.original_range(match_expr.syntax()).range,
243        |builder| {
244            // having any hidden variants means that we need a catch-all arm
245            needs_catch_all_arm |= has_hidden_variants;
246
247            let mut missing_arms = missing_pats
248                .into_iter()
249                .filter(|(_, hidden)| {
250                    // filter out hidden patterns because they're handled by the catch-all arm
251                    !hidden
252                })
253                .map(|(pat, _)| make.match_arm(pat, None, utils::expr_fill_default(ctx.config)))
254                .collect::<Vec<_>>();
255
256            if needs_catch_all_arm && !has_catch_all_arm {
257                cov_mark::hit!(added_wildcard_pattern);
258                let arm = make.match_arm(
259                    make.wildcard_pat().into(),
260                    None,
261                    utils::expr_fill_default(ctx.config),
262                );
263                missing_arms.push(arm);
264            }
265
266            // FIXME: Hack for syntax trees not having great support for macros
267            // Just edit the element that the original range came from
268            let old_place = {
269                // Find the original element
270                let file = ctx.sema.parse(arm_list_range.file_id);
271                let old_place = file.syntax().covering_element(arm_list_range.range);
272
273                match old_place {
274                    syntax::SyntaxElement::Node(it) => it,
275                    syntax::SyntaxElement::Token(it) => {
276                        // If a token is found, it is '{' or '}'
277                        // The parent is `{ ... }`
278                        it.parent().expect("Token must have a parent.")
279                    }
280                }
281            };
282
283            let editor = builder.make_editor(&old_place);
284            let mut arms_edit = ArmsEdit { match_arm_list, place: old_place, last_arm: None };
285
286            arms_edit.remove_wildcard_arms(ctx, &editor);
287            arms_edit.add_comma_after_last_arm(ctx, &make, &editor);
288            arms_edit.append_arms(&missing_arms, &make, &editor);
289
290            if let Some(cap) = ctx.config.snippet_cap {
291                if let Some(it) = missing_arms
292                    .first()
293                    .and_then(|arm| arm.syntax().descendants().find_map(ast::WildcardPat::cast))
294                {
295                    editor.add_annotation(it.syntax(), builder.make_placeholder_snippet(cap));
296                }
297
298                for arm in &missing_arms {
299                    if let Some(expr) = arm.expr() {
300                        editor.add_annotation(expr.syntax(), builder.make_placeholder_snippet(cap));
301                    }
302                }
303
304                if let Some(arm) = missing_arms.last() {
305                    editor.add_annotation(arm.syntax(), builder.make_tabstop_after(cap));
306                }
307            }
308
309            builder.add_file_edits(ctx.vfs_file_id(), editor);
310        },
311    )
312}

314fn cursor_at_trivial_match_arm_list(
315    ctx: &AssistContext<'_, '_>,
316    match_expr: &MatchExpr,
317    match_arm_list: &MatchArmList,
318) -> Option<()> {
319    // match x { $0 }
320    if match_arm_list.arms().next().is_none() {
321        cov_mark::hit!(add_missing_match_arms_empty_body);
322        return Some(());
323    }
324
325    // match x {
326    //     bar => baz,
327    //     $0
328    // }
329    if let Some(last_arm) = match_arm_list.arms().last() {
330        let last_node = match last_arm.expr() {
331            Some(expr) => expr.syntax().clone(),
332            None => last_arm.syntax().clone(),
333        };
334        let last_node_range = ctx.sema.original_range_opt(&last_node)?.range;
335        let match_expr_range = ctx.sema.original_range_opt(match_expr.syntax())?.range;
336        if last_node_range.end() <= ctx.offset() && ctx.offset() < match_expr_range.end() {
337            cov_mark::hit!(add_missing_match_arms_end_of_last_arm);
338            return Some(());
339        }
340
341        if ast::Expr::cast(last_node.clone()).is_some_and(is_empty_expr)
342            && last_node_range.contains(ctx.offset())
343            && !last_node.text().contains_char('\n')
344        {
345            cov_mark::hit!(add_missing_match_arms_end_of_last_empty_arm);
346            return Some(());
347        }
348    }
349
350    // match { _$0 => {...} }
351    let wild_pat = ctx.find_node_at_offset_with_descend::<ast::WildcardPat>()?;
352    let arm = wild_pat.syntax().parent().and_then(ast::MatchArm::cast)?;
353    let arm_match_expr = arm.syntax().ancestors().nth(2).and_then(ast::MatchExpr::cast)?;
354    if arm_match_expr == *match_expr {
355        cov_mark::hit!(add_missing_match_arms_trivial_arm);
356        return Some(());
357    }
358
359    None
360}

362struct ArmsEdit {
363    match_arm_list: MatchArmList,
364    place: SyntaxNode,
365    last_arm: Option<ast::MatchArm>,
366}
367
368impl ArmsEdit {
369    fn remove_wildcard_arms(&mut self, ctx: &AssistContext<'_, '_>, editor: &SyntaxEditor) {
370        for arm in self.match_arm_list.arms() {
371            if !matches!(arm.pat(), Some(Pat::WildcardPat(_))) {
372                self.last_arm = Some(arm);
373                continue;
374            }
375            if !arm.expr().is_none_or(is_empty_expr) {
376                cov_mark::hit!(add_missing_match_arms_empty_expr);
377                self.last_arm = Some(arm);
378                continue;
379            }
380            let Some(range) = self.cover_edit_range(ctx, &arm) else { continue };
381
382            let prev = match range.start() {
383                syntax::NodeOrToken::Node(node) => {
384                    node.first_token().and_then(|it| it.prev_token())
385                }
386                syntax::NodeOrToken::Token(tok) => tok.prev_token(),
387            };
388            if let Some(prev) = prev
389                && prev.kind() == SyntaxKind::WHITESPACE
390            {
391                editor.delete(prev);
392            }
393
394            editor.delete_all(range);
395        }
396    }
397
398    fn append_arms(&self, arms: &[ast::MatchArm], make: &SyntaxFactory, editor: &SyntaxEditor) {
399        let Some(mut before) = self.place.last_token() else {
400            stdx::never!("match arm list not contain any token");
401            return;
402        };
403        if let Some(prev) = before.prev_token()
404            && prev.kind() == SyntaxKind::WHITESPACE
405        {
406            before = prev;
407        }
408        let open_curly =
409            !self.place.text().contains_char('\n') || before.kind() == SyntaxKind::WHITESPACE;
410        let indent = IndentLevel::from_node(&self.place);
411        let arm_indent = indent + 1;
412        let indent = make.whitespace(&format!("\n{indent}"));
413        let arm_indent = make.whitespace(&format!("\n{arm_indent}"));
414        let elements = arms
415            .iter()
416            .flat_map(|arm| [arm_indent.clone().into(), arm.syntax().clone().into()])
417            .chain(open_curly.then(|| indent.clone().into()))
418            .collect();
419
420        if before.kind() == SyntaxKind::WHITESPACE {
421            editor.replace_with_many(before, elements);
422        } else {
423            editor.insert_all(Position::before(before), elements);
424        }
425    }
426
427    fn add_comma_after_last_arm(
428        &self,
429        ctx: &AssistContext<'_, '_>,
430        make: &SyntaxFactory,
431        editor: &SyntaxEditor,
432    ) {
433        if let Some(last_arm) = &self.last_arm
434            && last_arm.comma_token().is_none()
435            && last_arm.expr().is_none_or(|it| !it.is_block_like())
436            && let Some(range) = self.cover_edit_range(ctx, last_arm)
437        {
438            editor.insert(Position::after(range.end()), make.token(syntax::T![,]));
439        }
440    }
441
442    fn cover_edit_range(
443        &self,
444        ctx: &AssistContext<'_, '_>,
445        node: &impl AstNode,
446    ) -> Option<std::ops::RangeInclusive<syntax::SyntaxElement>> {
447        let range = ctx.sema.original_range_opt(node.syntax())?;
448
449        if !self.place.text_range().contains_range(range.range) {
450            return None;
451        }
452
453        Some(utils::cover_edit_range(&self.place, range.range))
454    }
455}

457fn is_variant_missing(existing_pats: &[Pat], var: &Pat) -> bool {
458    !existing_pats.iter().any(|pat| does_pat_match_variant(pat, var))
459}

461fn is_empty_expr(e: ast::Expr) -> bool {
462    match e {
463        ast::Expr::BlockExpr(b) => b.statements().next().is_none() && b.tail_expr().is_none(),
464        ast::Expr::TupleExpr(t) => t.fields().next().is_none(),
465        _ => false,
466    }
467}

469// Fixme: this is still somewhat limited, use hir_ty::diagnostics::match_check?
470fn does_pat_match_variant(pat: &Pat, var: &Pat) -> bool {
471    match (pat, var) {
472        (Pat::WildcardPat(_), _) => true,
473        (Pat::SlicePat(spat), Pat::SlicePat(svar)) => {
474            spat.pats().zip(svar.pats()).all(|(p, v)| does_pat_match_variant(&p, &v))
475        }
476        (Pat::TuplePat(tpat), Pat::TuplePat(tvar)) => {
477            tpat.fields().zip(tvar.fields()).all(|(p, v)| does_pat_match_variant(&p, &v))
478        }
479        (Pat::OrPat(opat), _) => opat.pats().any(|p| does_pat_match_variant(&p, var)),
480        _ => utils::does_pat_match_variant(pat, var),
481    }
482}

485enum ExtendedEnum {
486    Bool,
487    Enum { enum_: hir::Enum, use_self: bool },
488}

490#[derive(Eq, PartialEq, Clone, Copy, Debug)]
491enum ExtendedVariant {
492    True,
493    False,
494    Variant { variant: hir::EnumVariant, use_self: bool },
495}
496
497impl ExtendedVariant {
498    fn should_be_hidden(self, db: &RootDatabase, krate: Crate) -> bool {
499        match self {
500            ExtendedVariant::Variant { variant: var, .. } => {
501                var.attrs(db).is_doc_hidden() && var.module(db).krate(db) != krate
502            }
503            _ => false,
504        }
505    }
506}
507
508impl ExtendedEnum {
509    fn enum_(
510        db: &RootDatabase,
511        enum_: hir::Enum,
512        enum_ty: &hir::Type<'_>,
513        self_ty: Option<&hir::Type<'_>>,
514    ) -> Self {
515        ExtendedEnum::Enum {
516            enum_,
517            use_self: self_ty.is_some_and(|self_ty| self_ty.could_unify_with_deeply(db, enum_ty)),
518        }
519    }
520
521    fn is_non_exhaustive(&self, db: &RootDatabase, krate: Crate) -> bool {
522        match self {
523            ExtendedEnum::Enum { enum_: e, .. } => {
524                e.attrs(db).is_non_exhaustive() && e.module(db).krate(db) != krate
525            }
526            _ => false,
527        }
528    }
529
530    fn variants(&self, db: &RootDatabase) -> Vec<ExtendedVariant> {
531        match *self {
532            ExtendedEnum::Enum { enum_: e, use_self } => e
533                .variants(db)
534                .into_iter()
535                .map(|variant| ExtendedVariant::Variant { variant, use_self })
536                .collect::<Vec<_>>(),
537            ExtendedEnum::Bool => {
538                Vec::<ExtendedVariant>::from([ExtendedVariant::True, ExtendedVariant::False])
539            }
540        }
541    }
542}

544fn resolve_enum_def(
545    sema: &Semantics<'_, RootDatabase>,
546    expr: &ast::Expr,
547    self_ty: Option<&hir::Type<'_>>,
548) -> Option<ExtendedEnum> {
549    sema.type_of_expr(expr)?.adjusted().autoderef(sema.db).find_map(|ty| match ty.as_adt() {
550        Some(Adt::Enum(e)) => Some(ExtendedEnum::enum_(sema.db, e, &ty, self_ty)),
551        _ => ty.is_bool().then_some(ExtendedEnum::Bool),
552    })
553}
554
555fn resolve_tuple_of_enum_def(
556    sema: &Semantics<'_, RootDatabase>,
557    expr: &ast::Expr,
558    self_ty: Option<&hir::Type<'_>>,
559) -> Option<Vec<ExtendedEnum>> {
560    sema.type_of_expr(expr)?
561        .adjusted()
562        .tuple_fields(sema.db)
563        .iter()
564        .map(|ty| {
565            ty.autoderef(sema.db).find_map(|ty| {
566                match ty.as_adt() {
567                    Some(Adt::Enum(e)) => Some(ExtendedEnum::enum_(sema.db, e, &ty, self_ty)),
568                    // For now we only handle expansion for a tuple of enums. Here
569                    // we map non-enum items to None and rely on `collect` to
570                    // convert Vec<Option<hir::Enum>> into Option<Vec<hir::Enum>>.
571                    _ => ty.is_bool().then_some(ExtendedEnum::Bool),
572                }
573            })
574        })
575        .collect::<Option<Vec<ExtendedEnum>>>()
576        .filter(|list| !list.is_empty())
577}
578
579fn resolve_array_of_enum_def(
580    sema: &Semantics<'_, RootDatabase>,
581    expr: &ast::Expr,
582    self_ty: Option<&hir::Type<'_>>,
583) -> Option<(ExtendedEnum, usize)> {
584    sema.type_of_expr(expr)?.adjusted().as_array(sema.db).and_then(|(ty, len)| {
585        ty.autoderef(sema.db).find_map(|ty| match ty.as_adt() {
586            Some(Adt::Enum(e)) => Some((ExtendedEnum::enum_(sema.db, e, &ty, self_ty), len)),
587            _ => ty.is_bool().then_some((ExtendedEnum::Bool, len)),
588        })
589    })
590}

592fn build_pat(
593    ctx: &AssistContext<'_, '_>,
594    make: &SyntaxFactory,
595    module: hir::Module,
596    var: ExtendedVariant,
597    cfg: FindPathConfig,
598) -> Option<ast::Pat> {
599    let db = ctx.db();
600    match var {
601        ExtendedVariant::Variant { variant: var, use_self } => {
602            let edition = module.krate(db).edition(db);
603            let path = if use_self {
604                make.path_from_segments(
605                    [
606                        make.path_segment(make.name_ref_self_ty()),
607                        make.path_segment(
608                            make.name_ref(&var.name(db).display(db, edition).to_smolstr()),
609                        ),
610                    ],
611                    false,
612                )
613            } else {
614                mod_path_to_ast_with_factory(
615                    make,
616                    &module.find_path(db, ModuleDef::from(var), cfg)?,
617                    edition,
618                )
619            };
620            let fields = var.fields(db);
621            let pat: ast::Pat = match var.kind(db) {
622                hir::StructKind::Tuple => {
623                    let mut name_generator = suggest_name::NameGenerator::default();
624                    let pats = fields.into_iter().map(|f| {
625                        let name = name_generator.for_type(&f.ty(db), db, edition);
626                        match name {
627                            Some(name) => make.ident_pat(false, false, make.name(&name)).into(),
628                            None => make.wildcard_pat().into(),
629                        }
630                    });
631                    make.tuple_struct_pat(path, pats).into()
632                }
633                hir::StructKind::Record => {
634                    let fields = fields
635                        .into_iter()
636                        .map(|f| make.ident_pat(false, false, make.name(f.name(db).as_str())))
637                        .map(|ident| make.record_pat_field_shorthand(ident.into()));
638                    let fields = make.record_pat_field_list(fields, None);
639                    make.record_pat_with_fields(path, fields).into()
640                }
641                hir::StructKind::Unit => make.path_pat(path),
642            };
643            Some(pat)
644        }
645        ExtendedVariant::True => Some(ast::Pat::from(make.literal_pat("true"))),
646        ExtendedVariant::False => Some(ast::Pat::from(make.literal_pat("false"))),
647    }
648}
