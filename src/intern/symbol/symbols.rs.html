15macro_rules! last {
16    ( $($elems:literal,)+ ) => {
17        *[ $($elems,)* ].last().unwrap()
18    };
19}

21impl Integer {
22    #[inline]
23    pub fn as_uint(sym: &Symbol) -> Option<usize> {
24        if !sym.repr.is_arc() {
25            let elem_ref = sym.repr.pointer();
26            // SAFETY: The types have the same layout.
27            let elem_ref = unsafe { std::mem::transmute::<NonNull<*const str>, &&str>(elem_ref) };
28            Self::LIST.element_offset(elem_ref)
29        } else {
30            Self::as_uint_cold(sym)
31        }
32    }
33
34    #[cold]
35    fn as_uint_cold(sym: &Symbol) -> Option<usize> {
36        sym.as_str().parse().ok()
37    }
38}

40macro_rules! define_symbols {
41    (
42        @LISTS: $($list_type_name:ident = $list_prefix:literal + [ $($list_idx:literal,)+ ],)*
43        @WITH_NAME: $($alias:ident = $value:literal,)*
44        @PLAIN: $($name:ident,)*
45    ) => {
46        $(
47            pub enum $list_type_name {}
48            impl $list_type_name {
49                // Ensure we covered all numbers.
50                const LIST: &[&str; last!($($list_idx,)+) + 1] = {
51                    static LIST: [&str; last!($($list_idx,)+) + 1] = [ $( concat!($list_prefix, $list_idx), )* ];
52                    &LIST
53                };
54
55                #[cold]
56                #[inline(never)]
57                fn create(idx: usize) -> Symbol {
58                    use std::fmt::Write;
59                    const MAX_LEN: usize = $list_prefix.len() + u64::MAX.ilog10() as usize + 1;
60                    let mut s = ArrayString::<MAX_LEN>::new();
61                    s.push_str($list_prefix);
62                    _ = write!(s, "{idx}");
63                    Symbol::intern(&s)
64                }
65
66                #[inline]
67                pub fn get(idx: usize) -> Symbol {
68                    match Self::LIST.get(idx) {
69                        Option::Some(s) => Symbol { repr: TaggedArcPtr::non_arc(s) },
70                        Option::None => Self::create(idx),
71                    }
72                }
73            }
74        )*
75
76        // The strings should be in `static`s so that symbol equality holds.
77        $(
78            pub const $name: Symbol = {
79                static SYMBOL_STR: &str = stringify!($name);
80                Symbol { repr: TaggedArcPtr::non_arc(&SYMBOL_STR) }
81            };
82        )*
83        $(
84            pub const $alias: Symbol = {
85                static SYMBOL_STR: &str = $value;
86                Symbol { repr: TaggedArcPtr::non_arc(&SYMBOL_STR) }
87            };
88        )*
89
90
91        pub(super) fn prefill() -> DashMap<Symbol, (), BuildHasherDefault<FxHasher>> {
92            let mut dashmap_ = <DashMap<Symbol, (), BuildHasherDefault<FxHasher>>>::with_hasher(BuildHasherDefault::default());
93
94            let hasher_ = dashmap_.hasher().clone();
95            let hash_one = |it_: &str| hasher_.hash_one(it_);
96            {
97                $(
98                    for s in $list_type_name::LIST {
99                        let hash_ = hash_one(s);
100                        let shard_idx_ = dashmap_.determine_shard(hash_ as usize);
101                        let symbol = Symbol { repr: TaggedArcPtr::non_arc(s) };
102                        dashmap_.shards_mut()[shard_idx_].get_mut().insert(hash_, (symbol, SharedValue::new(())), |(x, _)| hash_one(x.as_str()));
103                    }
104                )*
105                $(
106                    let s = stringify!($name);
107                    let hash_ = hash_one(s);
108                    let shard_idx_ = dashmap_.determine_shard(hash_ as usize);
109                    dashmap_.shards_mut()[shard_idx_].get_mut().insert(hash_, ($name, SharedValue::new(())), |(x, _)| hash_one(x.as_str()));
110                )*
111                $(
112                    let s = $value;
113                    let hash_ = hash_one(s);
114                    let shard_idx_ = dashmap_.determine_shard(hash_ as usize);
115                    dashmap_.shards_mut()[shard_idx_].get_mut().insert(hash_, ($alias, SharedValue::new(())), |(x, _)| hash_one(x.as_str()));
116                )*
117            }
118            dashmap_
119        }
120    };
121}
