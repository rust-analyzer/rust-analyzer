10pub trait GetOrCreateWhereClause: ast::HasGenericParams {
11    fn where_clause_position(&self) -> Option<Position>;
12
13    fn get_or_create_where_clause(
14        &self,
15        editor: &SyntaxEditor,
16        new_preds: impl Iterator<Item = ast::WherePred>,
17    ) {
18        let make = editor.make();
19        let existing = self.where_clause();
20        let all_preds: Vec<_> =
21            existing.iter().flat_map(|wc| wc.predicates()).chain(new_preds).collect();
22        let new_where = make.where_clause(all_preds);
23
24        if let Some(existing) = &existing {
25            editor.replace(existing.syntax(), new_where.syntax());
26        } else if let Some(pos) = self.where_clause_position() {
27            editor.insert_all(
28                pos,
29                vec![make.whitespace(" ").into(), new_where.syntax().clone().into()],
30            );
31        }
32    }
33}

115impl SyntaxEditor {
116    /// Adds a new generic param to the node using `SyntaxEditor`
117    pub fn add_generic_param(
118        &self,
119        node: &impl ast::HasGenericParams,
120        new_param: ast::GenericParam,
121    ) {
122        let make = self.make();
123        match node.generic_param_list() {
124            Some(generic_param_list) => {
125                let is_lifetime = matches!(new_param, ast::GenericParam::LifetimeParam(_));
126
127                if let Some(first_param) = generic_param_list.generic_params().next() {
128                    let last_lifetime = generic_param_list
129                        .generic_params()
130                        .filter(|p| matches!(p, ast::GenericParam::LifetimeParam(_)))
131                        .last();
132
133                    if is_lifetime {
134                        if let Some(last_lt) = last_lifetime {
135                            let elements = vec![
136                                make.token(SyntaxKind::COMMA).into(),
137                                make.token(SyntaxKind::WHITESPACE).into(),
138                                new_param.syntax().clone().into(),
139                            ];
140                            self.insert_all(Position::after(last_lt.syntax()), elements);
141                        } else {
142                            // Insert before the first parameter
143                            let elements = vec![
144                                new_param.syntax().clone().into(),
145                                make.token(SyntaxKind::COMMA).into(),
146                                make.token(SyntaxKind::WHITESPACE).into(),
147                            ];
148                            self.insert_all(Position::before(first_param.syntax()), elements);
149                        }
150                    } else {
151                        let last_param = generic_param_list.generic_params().last().unwrap();
152                        let elements = vec![
153                            make.token(SyntaxKind::COMMA).into(),
154                            make.token(SyntaxKind::WHITESPACE).into(),
155                            new_param.syntax().clone().into(),
156                        ];
157                        self.insert_all(Position::after(last_param.syntax()), elements);
158                    }
159                } else {
160                    if let Some(l_angle) = generic_param_list.l_angle_token() {
161                        self.insert(Position::after(l_angle), new_param.syntax().clone());
162                    }
163                }
164            }
165            None => {
166                let position =
167                    if let Some(name) = node.syntax().children().find_map(ast::Name::cast) {
168                        Position::after(name.syntax())
169                    } else if let Some(impl_node) = ast::Impl::cast(node.syntax().clone()) {
170                        impl_node
171                            .impl_token()
172                            .map_or_else(|| Position::last_child_of(node.syntax()), Position::after)
173                    } else if let Some(fn_node) = ast::Fn::cast(node.syntax().clone()) {
174                        if let Some(fn_token) = fn_node.fn_token() {
175                            Position::after(fn_token)
176                        } else if let Some(param_list) = fn_node.param_list() {
177                            Position::before(param_list.syntax())
178                        } else {
179                            Position::last_child_of(node.syntax())
180                        }
181                    } else {
182                        Position::last_child_of(node.syntax())
183                    };
184
185                let elements = vec![
186                    make.token(SyntaxKind::L_ANGLE).into(),
187                    new_param.syntax().clone().into(),
188                    make.token(SyntaxKind::R_ANGLE).into(),
189                ];
190                self.insert_all(position, elements);
191            }
192        }
193    }
194}

196fn get_or_insert_comma_after(editor: &SyntaxEditor, syntax: &SyntaxNode) -> SyntaxToken {
197    let make = editor.make();
198    match comma_after(syntax) {
199        Some(it) => it,
200        None => {
201            let comma = make.token(T![,]);
202            editor.insert(Position::after(syntax), &comma);
203            comma
204        }
205    }
206}

208impl ast::AssocItemList {
209    /// Adds a new associated item after all of the existing associated items.
210    ///
211    /// Attention! This function does align the first line of `item` with respect to `self`,
212    /// but it does _not_ change indentation of other lines (if any).
213    pub fn add_items(&self, editor: &SyntaxEditor, items: Vec<ast::AssocItem>) {
214        let make = editor.make();
215        let (indent, position, whitespace) = match self.assoc_items().last() {
216            Some(last_item) => (
217                IndentLevel::from_node(last_item.syntax()),
218                Position::after(last_item.syntax()),
219                "\n\n",
220            ),
221            None => match self.l_curly_token() {
222                Some(l_curly) => {
223                    normalize_ws_between_braces(editor, self.syntax());
224                    (IndentLevel::from_token(&l_curly) + 1, Position::after(&l_curly), "\n")
225                }
226                None => (IndentLevel::zero(), Position::last_child_of(self.syntax()), "\n"),
227            },
228        };
229
230        let elements: Vec<SyntaxElement> = items
231            .into_iter()
232            .enumerate()
233            .flat_map(|(i, item)| {
234                let whitespace = if i != 0 { "\n\n" } else { whitespace };
235                vec![
236                    make.whitespace(&format!("{whitespace}{indent}")).into(),
237                    item.syntax().clone().into(),
238                ]
239            })
240            .collect();
241        editor.insert_all(position, elements);
242    }
243}

277fn add_record_fields(
278    editor: &SyntaxEditor,
279    field_list: &SyntaxNode,
280    last_field: Option<SyntaxNode>,
281    l_curly: Option<SyntaxToken>,
282    fields: impl Iterator<Item = SyntaxElement>,
283) {
284    let fields = fields.collect::<Vec<_>>();
285    if fields.is_empty() {
286        return;
287    }
288
289    let make = editor.make();
290    let is_multiline = field_list.text().contains_char('\n');
291    let whitespace = || {
292        if is_multiline {
293            let indent = IndentLevel::from_node(field_list) + 1;
294            make.whitespace(&format!("\n{indent}"))
295        } else {
296            make.whitespace(" ")
297        }
298    };
299
300    if is_multiline {
301        normalize_ws_between_braces(editor, field_list);
302    }
303
304    let mut elements = Vec::new();
305    let next_after_insert;
306    let position = match last_field {
307        Some(last_field) => match comma_after(&last_field) {
308            Some(comma) => {
309                next_after_insert = comma.next_sibling_or_token();
310                Position::after(comma)
311            }
312            None => {
313                next_after_insert = last_field.next_sibling_or_token();
314                elements.push(make.token(T![,]).into());
315                Position::after(last_field)
316            }
317        },
318        None => match l_curly {
319            Some(it) => {
320                next_after_insert = it.next_sibling_or_token();
321                Position::after(it)
322            }
323            None => {
324                next_after_insert = None;
325                Position::last_child_of(field_list)
326            }
327        },
328    };
329
330    let fields_len = fields.len();
331    for (idx, field) in fields.into_iter().enumerate() {
332        elements.push(whitespace().into());
333        elements.push(field);
334        if is_multiline || idx + 1 != fields_len {
335            elements.push(make.token(T![,]).into());
336        }
337    }
338    if !is_multiline && next_after_insert.is_some_and(|it| it.kind() != SyntaxKind::WHITESPACE) {
339        elements.push(make.whitespace(" ").into());
340    }
341
342    editor.insert_all(position, elements);
343}

345fn comma_after(syntax: &SyntaxNode) -> Option<SyntaxToken> {
346    syntax
347        .siblings_with_tokens(Direction::Next)
348        .filter_map(|it| it.into_token())
349        .find(|it| it.kind() == T![,])
350}

371impl ast::VariantList {
372    pub fn add_variant(&self, editor: &SyntaxEditor, variant: &ast::Variant) {
373        let make = editor.make();
374        let (indent, position) = match self.variants().last() {
375            Some(last_item) => (
376                IndentLevel::from_node(last_item.syntax()),
377                Position::after(get_or_insert_comma_after(editor, last_item.syntax())),
378            ),
379            None => match self.l_curly_token() {
380                Some(l_curly) => {
381                    normalize_ws_between_braces(editor, self.syntax());
382                    (IndentLevel::from_token(&l_curly) + 1, Position::after(&l_curly))
383                }
384                None => (IndentLevel::zero(), Position::last_child_of(self.syntax())),
385            },
386        };
387        let elements: Vec<SyntaxElement> = vec![
388            make.whitespace(&format!("{}{indent}", "\n")).into(),
389            variant.syntax().clone().into(),
390            make.token(T![,]).into(),
391        ];
392        editor.insert_all(position, elements);
393    }
394}

396impl ast::Fn {
397    pub fn replace_or_insert_body(&self, editor: &SyntaxEditor, body: ast::BlockExpr) {
398        let make = editor.make();
399        if let Some(old_body) = self.body() {
400            editor.replace(old_body.syntax(), body.syntax());
401        } else {
402            let single_space = make.whitespace(" ");
403            let elements = vec![single_space.into(), body.syntax().clone().into()];
404
405            if let Some(semicolon) = self.semicolon_token() {
406                editor.replace_with_many(semicolon, elements);
407            } else {
408                editor.insert_all(Position::last_child_of(self.syntax()), elements);
409            }
410        }
411    }
412}

414fn normalize_ws_between_braces(editor: &SyntaxEditor, node: &SyntaxNode) -> Option<()> {
415    let make = editor.make();
416    let l = node
417        .children_with_tokens()
418        .filter_map(|it| it.into_token())
419        .find(|it| it.kind() == T!['{'])?;
420    let r = node
421        .children_with_tokens()
422        .filter_map(|it| it.into_token())
423        .find(|it| it.kind() == T!['}'])?;
424
425    let indent = IndentLevel::from_node(node);
426
427    match l.next_sibling_or_token() {
428        Some(ws)
429            if ws.kind() == SyntaxKind::WHITESPACE
430                && ws.next_sibling_or_token()?.into_token()? == r =>
431        {
432            editor.replace(ws, make.whitespace(&format!("\n{indent}")));
433        }
434        Some(ws) if ws.kind() == T!['}'] => {
435            editor.insert(Position::after(l), make.whitespace(&format!("\n{indent}")));
436        }
437        _ => (),
438    }
439    Some(())
440}

442pub trait Removable: AstNode {
443    fn remove(&self, editor: &SyntaxEditor);
444}

446impl Removable for ast::TypeBoundList {
447    fn remove(&self, editor: &SyntaxEditor) {
448        match self.syntax().siblings_with_tokens(Direction::Prev).find(|it| it.kind() == T![:]) {
449            Some(colon) => editor.delete_all(colon..=self.syntax().clone().into()),
450            None => editor.delete(self.syntax()),
451        }
452    }
453}
454
455impl Removable for ast::Use {
456    fn remove(&self, editor: &SyntaxEditor) {
457        let make = editor.make();
458        let next_ws = self
459            .syntax()
460            .next_sibling_or_token()
461            .and_then(|it| it.into_token())
462            .and_then(ast::Whitespace::cast);
463        if let Some(next_ws) = next_ws {
464            let ws_text = next_ws.syntax().text();
465            if let Some(rest) = ws_text.strip_prefix('\n') {
466                let next_use_removed = next_ws
467                    .syntax()
468                    .next_sibling_or_token()
469                    .and_then(|it| it.into_node())
470                    .and_then(ast::Use::cast)
471                    .and_then(|use_| use_.use_tree())
472                    .is_some_and(|use_tree| editor.deleted(use_tree.syntax()));
473                if rest.is_empty() || next_use_removed {
474                    editor.delete(next_ws.syntax());
475                } else {
476                    editor.replace(next_ws.syntax(), make.whitespace(rest));
477                }
478            }
479        }
480        let prev_ws = self
481            .syntax()
482            .prev_sibling_or_token()
483            .and_then(|it| it.into_token())
484            .and_then(ast::Whitespace::cast);
485        if let Some(prev_ws) = prev_ws {
486            let ws_text = prev_ws.syntax().text();
487            let prev_newline = ws_text.rfind('\n').map(|x| x + 1).unwrap_or(0);
488            let rest = &ws_text[0..prev_newline];
489            if rest.is_empty() {
490                editor.delete(prev_ws.syntax());
491            } else {
492                editor.replace(prev_ws.syntax(), make.whitespace(rest));
493            }
494        }
495
496        editor.delete(self.syntax());
497    }
498}
499
500impl Removable for ast::UseTree {
501    fn remove(&self, editor: &SyntaxEditor) {
502        for dir in [Direction::Next, Direction::Prev] {
503            if let Some(next_use_tree) = neighbor(self, dir) {
504                let separators = self
505                    .syntax()
506                    .siblings_with_tokens(dir)
507                    .skip(1)
508                    .take_while(|it| it.as_node() != Some(next_use_tree.syntax()));
509                for sep in separators {
510                    editor.delete(sep);
511                }
512                break;
513            }
514        }
515        editor.delete(self.syntax());
516    }
517}

529    fn ast_from_text<N: AstNode>(text: &str) -> N {
530        let parse = SourceFile::parse(text, Edition::CURRENT);
531        let node = match parse.tree().syntax().descendants().find_map(N::cast) {
532            Some(it) => it,
533            None => {
534                let node = std::any::type_name::<N>();
535                panic!("Failed to make ast node `{node}` from text `{text}`")
536            }
537        };
538        let node = node.clone_subtree();
539        assert_eq!(node.syntax().text_range().start(), 0.into());
540        node
541    }
