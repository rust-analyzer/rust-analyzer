1//! Spans are memory heavy, and we have a lot of token trees. Storing them straight
2//! will waste a lot of memory. So instead we implement a clever compression mechanism:
3//!
4//! A `TopSubtree` has a list of [`CompressedSpanPart`], which are the parts of a span
5//! that tend to be shared between tokens - namely, without the range.
6//!
7//! The main list of token trees is stored in a variable-length encoding as bytes.
8//! The encoding is documented in the [`decode()`] function (which decodes one [`TokenTree`]).
9

42trait Encodable: Sized {
43    #[cfg(all(debug_assertions, not(miri)))]
44    fn write(self, buffer: &[Cell<u8>]);
45    #[cfg(all(debug_assertions, not(miri)))]
46    fn read(buffer: &[u8]) -> Self;
47}
48
49impl Encodable for u8 {
50    #[cfg(all(debug_assertions, not(miri)))]
51    fn write(self, buffer: &[Cell<u8>]) {
52        buffer[0].set(self);
53    }
54    #[cfg(all(debug_assertions, not(miri)))]
55    fn read(buffer: &[u8]) -> Self {
56        buffer[0]
57    }
58}

60impl Encodable for u16 {
61    #[cfg(all(debug_assertions, not(miri)))]
62    fn write(self, buffer: &[Cell<u8>]) {
63        let value = self.to_ne_bytes();
64        let buffer: &[Cell<u8>; size_of::<Self>()] = buffer.try_into().unwrap();
65        for (b, v) in std::iter::zip(buffer, value) {
66            b.set(v);
67        }
68    }
69    #[cfg(all(debug_assertions, not(miri)))]
70    fn read(buffer: &[u8]) -> Self {
71        Self::from_ne_bytes(buffer.try_into().unwrap())
72    }
73}

101struct UninitBuffer {
102    #[cfg(all(debug_assertions, not(miri)))]
103    buffer: Box<[u8]>,
104    #[cfg(not(all(debug_assertions, not(miri))))]
105    buffer: Box<[std::mem::MaybeUninit<u8>]>,
106}
107
108impl UninitBuffer {
109    #[inline]
110    fn new(capacity: usize) -> Self {
111        Self {
112            #[cfg(all(debug_assertions, not(miri)))]
113            buffer: vec![0; capacity].into_boxed_slice(),
114            #[cfg(not(all(debug_assertions, not(miri))))]
115            buffer: Box::new_uninit_slice(capacity),
116        }
117    }
118
119    #[inline]
120    fn writer(&mut self) -> BufferWriter<'_> {
121        BufferWriter {
122            #[cfg(all(debug_assertions, not(miri)))]
123            buffer: Cell::from_mut(&mut *self.buffer).as_slice_of_cells(),
124            #[cfg(not(all(debug_assertions, not(miri))))]
125            ptr: self.buffer.as_mut_ptr_range().end.cast::<u8>(),
126            #[cfg(not(all(debug_assertions, not(miri))))]
127            _marker: stdx::variance::PhantomCovariantLifetime::new(),
128        }
129    }
130
131    #[cfg(all(debug_assertions, not(miri)))]
132    unsafe fn finish(self, writer_finish: usize) -> Box<[u8]> {
133        self.buffer[writer_finish..].into()
134    }
135
136    #[cfg(not(all(debug_assertions, not(miri))))]
137    #[inline]
138    unsafe fn finish(mut self, writer_finish: *mut u8) -> Box<[u8]> {
139        let end = self.buffer.as_mut_ptr_range().end.cast::<u8>();
140        unsafe {
141            let bytes_len = end.offset_from_unsigned(writer_finish);
142            let mut buffer = Box::<[u8]>::new_uninit_slice(bytes_len);
143            buffer.as_mut_ptr().cast::<u8>().copy_from_nonoverlapping(writer_finish, bytes_len);
144            buffer.assume_init()
145        }
146    }
147}

149#[derive(Clone, Copy)]
150struct BufferWriter<'a> {
151    #[cfg(all(debug_assertions, not(miri)))]
152    buffer: &'a [Cell<u8>],
153    #[cfg(not(all(debug_assertions, not(miri))))]
154    ptr: *mut u8,
155    #[cfg(not(all(debug_assertions, not(miri))))]
156    _marker: stdx::variance::PhantomCovariantLifetime<'a>,
157}
158
159impl<'a> BufferWriter<'a> {
160    #[inline]
161    unsafe fn new(buffer: &'a mut [u8]) -> Self {
162        BufferWriter {
163            #[cfg(all(debug_assertions, not(miri)))]
164            buffer: Cell::from_mut(buffer).as_slice_of_cells(),
165            #[cfg(not(all(debug_assertions, not(miri))))]
166            ptr: buffer.as_mut_ptr_range().end,
167            #[cfg(not(all(debug_assertions, not(miri))))]
168            _marker: stdx::variance::PhantomCovariantLifetime::new(),
169        }
170    }
171
172    #[cfg_attr(not(all(debug_assertions, not(miri))), inline(always))]
173    unsafe fn write<T: Encodable>(&mut self, value: T) {
174        #[cfg(all(debug_assertions, not(miri)))]
175        {
176            let write_at = self.buffer.split_off(self.buffer.len() - size_of::<T>()..).unwrap();
177            value.write(write_at);
178        }
179        #[cfg(not(all(debug_assertions, not(miri))))]
180        unsafe {
181            self.ptr = self.ptr.sub(size_of::<T>());
182            self.ptr.cast::<T>().write_unaligned(value);
183        }
184    }
185
186    fn len_since(self, other: BufferWriter<'_>) -> usize {
187        #[cfg(all(debug_assertions, not(miri)))]
188        {
189            other.buffer.len() - self.buffer.len()
190        }
191        #[cfg(not(all(debug_assertions, not(miri))))]
192        {
193            other.ptr.addr() - self.ptr.addr()
194        }
195    }
196
197    #[cfg(all(debug_assertions, not(miri)))]
198    fn finish(self) -> usize {
199        self.buffer.len()
200    }
201
202    #[cfg(not(all(debug_assertions, not(miri))))]
203    #[inline]
204    fn finish(self) -> *mut u8 {
205        self.ptr
206    }
207}

219// Encoding is done in reverse, from the end to the beginning. This is in order
220// to be able to tell how many bytes the children of a `Subtree` occupy, and store
221// *that* efficiently, with the smallest possible type.
222
223#[must_use]
224unsafe fn encode_span<'a>(
225    ptr: BufferWriter<'a>,
226    span: &Span,
227    span_parts_map: &FxHashMap<CompressedSpanPart, usize>,
228    extra_two_bits: u32,
229    force_heavy_encoding: bool,
230) -> BufferWriter<'a> {
231    let span_parts_index = span_parts_map[&CompressedSpanPart::from_span(span)] as u32;
232    let offset = u32::from(span.range.start());
233    let len = u32::from(span.range.len());
234    unsafe {
235        encode_span_no_map(ptr, span_parts_index, offset, len, extra_two_bits, force_heavy_encoding)
236    }
237}
238
239#[must_use]
240unsafe fn encode_span_no_map(
241    mut ptr: BufferWriter<'_>,
242    mut span_parts_index: u32,
243    mut offset: u32,
244    mut len: u32,
245    extra_two_bits: u32,
246    force_heavy_encoding: bool,
247) -> BufferWriter<'_> {
248    debug_assert!(extra_two_bits & !0b11 == 0);
249    let mut first_u32 = extra_two_bits;
250    first_u32 |= (span_parts_index & n_bits_mask(4)) << 2;
251    span_parts_index >>= 4;
252    first_u32 |= (len & n_bits_mask(8)) << (2 + 4);
253    len >>= 8;
254    first_u32 |= (offset & n_bits_mask(17)) << (2 + 4 + 8 + 1);
255    offset >>= 17;
256
257    let extends_to_next = span_parts_index != 0 || len != 0 || offset != 0 || force_heavy_encoding;
258    first_u32 |= u32::from(extends_to_next) << (2 + 4 + 8);
259
260    if extends_to_next {
261        ptr = unsafe {
262            encode_extended_span(ptr, span_parts_index, len, offset, force_heavy_encoding)
263        };
264    }
265    unsafe { ptr.write::<u32>(first_u32) };
266
267    ptr
268}

270#[cold]
271#[must_use]
272unsafe fn encode_extended_span(
273    mut ptr: BufferWriter<'_>,
274    span_parts_index: u32,
275    len: u32,
276    offset: u32,
277    force_heavy_encoding: bool,
278) -> BufferWriter<'_> {
279    if span_parts_index <= n_bits_mask(11)
280        && len <= n_bits_mask(10)
281        && offset <= n_bits_mask(10)
282        && !force_heavy_encoding
283    {
284        let mut second_u32 = span_parts_index;
285        second_u32 |= len << 11;
286        second_u32 |= offset << (11 + 10);
287        second_u32 <<= 1;
288        unsafe { ptr.write::<u32>(second_u32) };
289    } else {
290        assert!(span_parts_index <= n_bits_mask(24), "too big `span_parts_index`");
291
292        let mut u64 = u64::from(span_parts_index);
293        u64 |= u64::from(len) << 24;
294        u64 |= u64::from(offset) << (24 + 24);
295        let third_u32 = u64 as u32;
296        let mut second_u32 = (u64 >> u32::BITS) as u32;
297        second_u32 <<= 1;
298        second_u32 |= 0b1;
299        unsafe {
300            ptr.write::<u32>(third_u32);
301            ptr.write::<u32>(second_u32);
302        }
303    }
304
305    ptr
306}

308#[must_use]
309unsafe fn encode_symbol<'a>(
310    mut ptr: BufferWriter<'a>,
311    symbol: &Symbol,
312    tag: u32,
313    symbols_map: &FxHashMap<Symbol, usize>,
314) -> BufferWriter<'a> {
315    let symbol_idx = symbols_map[symbol] as u32;
316    unsafe {
317        if symbol_idx <= n_bits_mask(6) {
318            ptr.write::<u8>(((symbol_idx << 2) | tag) as u8);
319        } else if symbol_idx <= n_bits_mask(13) {
320            ptr.write::<u8>(((symbol_idx >> 6) << 1) as u8);
321            ptr.write::<u8>(((symbol_idx << 2) | 0b10 | tag) as u8);
322        } else {
323            ptr.write::<u16>((symbol_idx >> 13) as u16);
324            ptr.write::<u8>((((symbol_idx >> 6) << 1) | 0b1) as u8);
325            ptr.write::<u8>(((symbol_idx << 2) | 0b10 | tag) as u8);
326        }
327    }
328    ptr
329}

331#[must_use]
332unsafe fn encode<'a>(
333    mut ptr: BufferWriter<'a>,
334    tt: TokenTree,
335    span_parts_map: &FxHashMap<CompressedSpanPart, usize>,
336    byte_size_after: &mut [u32],
337    symbols_map: &FxHashMap<Symbol, usize>,
338) -> BufferWriter<'a> {
339    let before_ptr = ptr;
340    unsafe {
341        match tt {
342            TokenTree::Leaf(Leaf::Punct(Punct { char, spacing, span })) => {
343                if char.is_ascii() {
344                    let spacing = spacing as u8;
345                    let span_extra = 0b1 | (u32::from(spacing) & 0b10);
346                    let char = ((char as u8) << 1) | (spacing & 0b1);
347                    ptr.write::<u8>(char);
348                    ptr = encode_span(ptr, &span, span_parts_map, span_extra, false);
349                } else {
350                    let mut control_byte = 0b110;
351                    control_byte |= (spacing as u8) << 3;
352                    ptr.write::<char>(char);
353                    ptr.write::<u8>(control_byte);
354                    ptr = encode_span(ptr, &span, span_parts_map, 0b00, false);
355                }
356            }
357            TokenTree::Leaf(Leaf::Ident(Ident { sym, span, kind })) => {
358                let is_not_normal = kind != IdentKind::Normal;
359                if is_not_normal {
360                    ptr.write::<u8>(kind as u8);
361                }
362                ptr = encode_symbol(ptr, &sym, u32::from(is_not_normal), symbols_map);
363                ptr = encode_span(ptr, &span, span_parts_map, 0b10, false);
364            }
365            TokenTree::Leaf(Leaf::Literal(Literal { text_and_suffix, span, kind, suffix_len })) => {
366                if matches!(kind, LitKind::Str | LitKind::StrRaw(0 | 1) | LitKind::Integer)
367                    && u32::from(suffix_len) <= n_bits_mask(4)
368                {
369                    // Literal, format 1.
370                    let mut control_byte = match kind {
371                        LitKind::Str => 0b0_011,
372                        LitKind::StrRaw(0) => 0b0_100,
373                        LitKind::StrRaw(1) => 0b1_011,
374                        LitKind::Integer => 0b1_100,
375                        _ => unreachable!(),
376                    };
377                    control_byte |= suffix_len << 4;
378                    ptr = encode_symbol(ptr, &text_and_suffix, 0, symbols_map);
379                    ptr.write::<u8>(control_byte);
380                } else {
381                    // Literal, format 2.
382                    let mut control_byte = 0b111;
383                    let (kind, raw_count) = match kind {
384                        LitKind::Byte => (0, None),
385                        LitKind::Char => (1, None),
386                        LitKind::Integer => (2, None),
387                        LitKind::Float => (3, None),
388                        LitKind::Str => (4, None),
389                        LitKind::StrRaw(count) => (5, Some(count)),
390                        LitKind::ByteStr => (6, None),
391                        LitKind::ByteStrRaw(count) => (7, Some(count)),
392                        LitKind::CStr => (8, None),
393                        LitKind::CStrRaw(count) => (9, Some(count)),
394                        LitKind::Err(()) => (10, None),
395                    };
396                    control_byte |= kind << 3;
397                    ptr = encode_symbol(ptr, &text_and_suffix, 0, symbols_map);
398                    ptr.write::<u8>(suffix_len);
399                    if let Some(raw_count) = raw_count {
400                        ptr.write::<u8>(raw_count);
401                    }
402                    ptr.write::<u8>(control_byte);
403                }
404                ptr = encode_span(ptr, &span, span_parts_map, 0b00, false);
405            }
406            TokenTree::Subtree(Subtree { delimiter, len }) => {
407                let open_span_parts_index =
408                    span_parts_map[&CompressedSpanPart::from_span(&delimiter.open)] as u32;
409                let close_span_parts_index =
410                    span_parts_map[&CompressedSpanPart::from_span(&delimiter.close)] as u32;
411                let close_span_offset_from_open = delimiter
412                    .close
413                    .range
414                    .start()
415                    .checked_sub(delimiter.open.range.start())
416                    .map_or(u32::MAX, u32::from);
417                let children_byte_len = byte_size_after[1] - byte_size_after[1 + len as usize];
418                if open_span_parts_index == close_span_parts_index
419                    && len <= n_bits_mask(u8::BITS)
420                    && children_byte_len <= n_bits_mask(u8::BITS)
421                    && delimiter.open.range.len() == TextSize::new(1)
422                    && delimiter.close.range.len() == TextSize::new(1)
423                    && close_span_offset_from_open <= n_bits_mask(11)
424                {
425                    // Subtree, format 1.
426                    let span = Span {
427                        range: TextRange::at(
428                            delimiter.open.range.start(),
429                            TextSize::new(close_span_offset_from_open),
430                        ),
431                        anchor: delimiter.open.anchor,
432                        ctx: delimiter.open.ctx,
433                    };
434                    let mut control_byte = 0b000;
435                    control_byte |= (delimiter.kind as u8) << 3;
436                    control_byte |= ((close_span_offset_from_open >> 8) << (3 + 2)) as u8;
437                    ptr.write::<u8>(children_byte_len as u8);
438                    ptr.write::<u8>(len as u8);
439                    ptr.write::<u8>(control_byte);
440                    ptr = encode_span(ptr, &span, span_parts_map, 0b00, false);
441                } else if open_span_parts_index == close_span_parts_index
442                    && len <= n_bits_mask(u8::BITS)
443                    && children_byte_len <= n_bits_mask(u8::BITS)
444                    && delimiter.open.range.len() == delimiter.close.range.len()
445                    && close_span_offset_from_open <= n_bits_mask(3)
446                {
447                    // Subtree, format 2.
448                    let mut control_byte = 0b001;
449                    control_byte |= (delimiter.kind as u8) << 3;
450                    control_byte |= (close_span_offset_from_open << (3 + 2)) as u8;
451                    ptr.write::<u8>(children_byte_len as u8);
452                    ptr.write::<u8>(len as u8);
453                    ptr.write::<u8>(control_byte);
454                    ptr = encode_span(ptr, &delimiter.open, span_parts_map, 0b00, false);
455                } else if len <= n_bits_mask(u8::BITS)
456                    && children_byte_len <= n_bits_mask(12)
457                    && delimiter.open.range.len() == delimiter.close.range.len()
458                    && close_span_offset_from_open <= n_bits_mask(8)
459                    && close_span_parts_index <= n_bits_mask(7)
460                {
461                    // Subtree, format 3.
462                    let mut control_byte = 0b010;
463                    control_byte |= (delimiter.kind as u8) << 3;
464                    control_byte |= (close_span_parts_index << (3 + 2)) as u8;
465                    let mut children_byte_len = children_byte_len << 4;
466                    children_byte_len |= close_span_parts_index >> 3;
467                    ptr.write::<u16>(children_byte_len as u16);
468                    ptr.write::<u8>(len as u8);
469                    ptr.write::<u8>(close_span_offset_from_open as u8);
470                    ptr.write::<u8>(control_byte);
471                    ptr = encode_span(ptr, &delimiter.open, span_parts_map, 0b00, false);
472                } else {
473                    // Subtree, format 4.
474                    let mut control_byte = 0b101;
475                    control_byte |= (delimiter.kind as u8) << 3;
476                    ptr.write::<u32>(children_byte_len);
477                    ptr.write::<u32>(len);
478                    ptr = encode_span(ptr, &delimiter.close, span_parts_map, 0b00, false);
479                    ptr.write::<u8>(control_byte);
480                    ptr = encode_span(ptr, &delimiter.open, span_parts_map, 0b00, false);
481                }
482            }
483        }
484
485        let element_byte_size: u32 = ptr.len_since(before_ptr).try_into().unwrap();
486        byte_size_after[0] = byte_size_after[1] + element_byte_size;
487    }
488
489    ptr
490}

492/// We always encode the top subtree with the heaviest encoding because we sometimes want to change it.
493unsafe fn encode_top_subtree<'a>(
494    mut ptr: BufferWriter<'a>,
495    top_subtree: Subtree,
496    open_span_parts_index: u32,
497    byte_size_after: &[u32],
498) -> BufferWriter<'a> {
499    unsafe {
500        let Subtree { delimiter, len } = top_subtree;
501        let children_byte_len = byte_size_after[1] - byte_size_after[1 + len as usize];
502
503        let mut control_byte = 0b101;
504        control_byte |= (delimiter.kind as u8) << 3;
505        ptr.write::<u32>(children_byte_len);
506        ptr.write::<u32>(len);
507        ptr = encode_span_no_map(
508            ptr,
509            open_span_parts_index + 1,
510            delimiter.close.range.start().into(),
511            delimiter.close.range.len().into(),
512            0b00,
513            true,
514        );
515        ptr.write::<u8>(control_byte);
516        ptr = encode_span_no_map(
517            ptr,
518            open_span_parts_index,
519            delimiter.open.range.start().into(),
520            delimiter.open.range.len().into(),
521            0b00,
522            true,
523        );
524    }
525    ptr
526}

528fn change_root_delimiter(buffer: &mut [u8], new_delim: DelimiterKind) {
529    // The span is 3*u32 and then the control byte, in which the delimiter comes.
530    let control_byte_index = 3 * size_of::<u32>();
531    let mut control_byte = buffer[control_byte_index];
532    control_byte &= 0b111; // Remove previous delimiter.
533    control_byte |= (new_delim as u8) << 3;
534    buffer[control_byte_index] = control_byte;
535}

537unsafe fn change_root_spans(
538    buffer: &mut [u8],
539    open_span_parts_index: u32,
540    close_span_parts_index: u32,
541    open_range: TextRange,
542    close_range: TextRange,
543) {
544    // Remember we write in reverse, so we add `3 * size_of::<u32>()`.
545    unsafe {
546        _ = encode_span_no_map(
547            BufferWriter::new(&mut buffer[..3 * size_of::<u32>()]),
548            open_span_parts_index,
549            u32::from(open_range.start()),
550            u32::from(open_range.len()),
551            0b00,
552            true,
553        );
554        _ = encode_span_no_map(
555            BufferWriter::new(
556                &mut buffer[..3 * size_of::<u32>() + size_of::<u8>() + 3 * size_of::<u32>()],
557            ),
558            close_span_parts_index,
559            u32::from(close_range.start()),
560            u32::from(close_range.len()),
561            0b00,
562            true,
563        );
564    }

567/// This is subtree in format 4: two spans, each at most 3*u32, a u8 control byte, a u32 length and a u32 bytes length.
568const BIGGEST_POSSIBLE_TT_ENCODING: usize =
569    2 * 3 * size_of::<u32>() + size_of::<u8>() + size_of::<u32>() + size_of::<u32>();

571fn encode_all(
572    tts: std::vec::IntoIter<TokenTree>,
573    mut compressed_span_frequencies: FxHashMap<CompressedSpanPart, usize>,
574    mut symbol_frequencies: FxHashMap<Symbol, usize>,
575) -> TopSubtree {
576    let tts_len = tts.len();
577    let mut token_trees = tts.enumerate();
578    let Some((_, TokenTree::Subtree(top_subtree))) = token_trees.next() else {
579        panic!("must always have a top subtree");
580    };
581
582    let (span_parts, span_parts_map) = {
583        let mut compressed_spans = compressed_span_frequencies
584            .keys()
585            .copied()
586            .chain([
587                CompressedSpanPart::from_span(&top_subtree.delimiter.open),
588                CompressedSpanPart::from_span(&top_subtree.delimiter.close),
589            ])
590            .collect::<Box<[_]>>();
591        {
592            // For this purpose, do not consider the top delimiters. They should stay last and not affect the other spans,
593            // since we might want to change them.
594            let len = compressed_spans.len();
595            let compressed_spans = &mut compressed_spans[..len - 2];
596            // No need sort if there is already enough space for everyone to be encoded efficiently.
597            if compressed_span_frequencies.len() > n_bits_mask(4) as usize {
598                // We want more used spans to have lower indices, so they can be encoded more efficiently.
599                compressed_spans.sort_unstable_by_key(|span| {
600                    std::cmp::Reverse(compressed_span_frequencies[span])
601                });
602            }
603            for (index, span) in compressed_spans.iter().enumerate() {
604                *compressed_span_frequencies.get_mut(span).unwrap() = index;
605            }
606        }
607        (compressed_spans, compressed_span_frequencies)
608    };
609
610    let (symbols, symbols_map) = {
611        let mut symbols = symbol_frequencies.keys().cloned().collect::<Box<[_]>>();
612        // No need sort if there is already enough space for everyone to be encoded efficiently.
613        if symbol_frequencies.len() > n_bits_mask(6) as usize {
614            // We want more used spans to have lower indices, so they can be encoded more efficiently.
615            symbols.sort_unstable_by_key(|symbol| std::cmp::Reverse(symbol_frequencies[symbol]));
616        }
617        for (index, span) in symbols.iter().enumerate() {
618            *symbol_frequencies.get_mut(span).unwrap() = index;
619        }
620        (symbols, symbol_frequencies)
621    };
622
623    // +1 because each `encode()` calls reads the previous value.
624    let mut byte_size_after = vec![0u32; tts_len + 1];
625    let bytes_capacity = tts_len * BIGGEST_POSSIBLE_TT_ENCODING;
626    unsafe {
627        let mut temp_buffer = UninitBuffer::new(bytes_capacity);
628        let mut ptr = temp_buffer.writer();
629        for (index, tt) in token_trees.rev() {
630            ptr = encode(ptr, tt, &span_parts_map, &mut byte_size_after[index..], &symbols_map);
631        }
632        ptr = encode_top_subtree(ptr, top_subtree, (span_parts.len() - 2) as u32, &byte_size_after);
633
634        let writer_finish = ptr.finish();
635        let buffer = temp_buffer.finish(writer_finish);
636
637        TopSubtree { buffer, span_parts, len: tts_len, symbols }
638    }
639}

641fn compute_span_frequencies_and_symbols(
642    tts: &[TokenTree],
643) -> (FxHashMap<CompressedSpanPart, usize>, FxHashMap<Symbol, usize>) {
644    let mut span_frequencies = FxHashMap::default();
645    let mut symbols = FxHashMap::default();
646    for tt in tts {
647        match tt {
648            TokenTree::Leaf(leaf) => {
649                if let Some(symbol) = leaf.symbol() {
650                    *symbols.entry(symbol.clone()).or_insert(0) += 1;
651                }
652
653                *span_frequencies.entry(CompressedSpanPart::from_span(leaf.span())).or_insert(0) +=
654                    1;
655            }
656            TokenTree::Subtree(subtree) => {
657                *span_frequencies
658                    .entry(CompressedSpanPart::from_span(&subtree.delimiter.open))
659                    .or_insert(0) += 1;
660                *span_frequencies
661                    .entry(CompressedSpanPart::from_span(&subtree.delimiter.close))
662                    .or_insert(0) += 1;
663            }
664        }
665    }
666    (span_frequencies, symbols)
667}

669#[derive(Clone, Copy)]
670struct BufferReader<'a> {
671    #[cfg(all(debug_assertions, not(miri)))]
672    buffer: &'a [u8],
673    #[cfg(not(all(debug_assertions, not(miri))))]
674    ptr: *const u8,
675    #[cfg(not(all(debug_assertions, not(miri))))]
676    _marker: stdx::variance::PhantomCovariantLifetime<'a>,
677}
678
679impl<'a> BufferReader<'a> {
680    #[inline]
681    fn start_end(slice: &'a [u8]) -> (Self, Self) {
682        #[cfg(all(debug_assertions, not(miri)))]
683        {
684            let start = Self { buffer: slice };
685            let end = Self { buffer: &slice[slice.len()..] };
686            (start, end)
687        }
688        #[cfg(not(all(debug_assertions, not(miri))))]
689        {
690            let ptrs = slice.as_ptr_range();
691            let start = Self {
692                ptr: ptrs.start,
693                #[cfg(not(all(debug_assertions, not(miri))))]
694                _marker: stdx::variance::PhantomCovariantLifetime::new(),
695            };
696            let end = Self {
697                ptr: ptrs.end,
698                #[cfg(not(all(debug_assertions, not(miri))))]
699                _marker: stdx::variance::PhantomCovariantLifetime::new(),
700            };
701            (start, end)
702        }
703    }
704
705    #[cfg_attr(not(all(debug_assertions, not(miri))), inline(always))]
706    unsafe fn read<T: Encodable>(&mut self) -> T {
707        #[cfg(all(debug_assertions, not(miri)))]
708        {
709            let read_at = self.buffer.split_off(..size_of::<T>()).unwrap();
710            T::read(read_at)
711        }
712        #[cfg(not(all(debug_assertions, not(miri))))]
713        unsafe {
714            let result = self.ptr.cast::<T>().read_unaligned();
715            self.ptr = self.ptr.add(size_of::<T>());
716            result
717        }
718    }
719
720    #[cfg_attr(not(all(debug_assertions, not(miri))), inline(always))]
721    unsafe fn skip(&mut self, amount: usize) {
722        #[cfg(all(debug_assertions, not(miri)))]
723        {
724            self.buffer = &self.buffer[amount..];
725        }
726        #[cfg(not(all(debug_assertions, not(miri))))]
727        unsafe {
728            self.ptr = self.ptr.add(amount);
729        }
730    }
731}
732
733impl PartialEq for BufferReader<'_> {
734    #[inline]
735    fn eq(&self, other: &Self) -> bool {
736        #[cfg(all(debug_assertions, not(miri)))]
737        let self_ptr = self.buffer.as_ptr();
738        #[cfg(not(all(debug_assertions, not(miri))))]
739        let self_ptr = self.ptr;
740        #[cfg(all(debug_assertions, not(miri)))]
741        let other_ptr = other.buffer.as_ptr();
742        #[cfg(not(all(debug_assertions, not(miri))))]
743        let other_ptr = other.ptr;
744
745        self_ptr == other_ptr
746    }
747}

749struct InlineSpanParts {
750    span_parts_index: usize,
751    text_range: TextRange,
752}
753
754#[inline]
755unsafe fn decode_span(
756    ptr: BufferReader<'_>,
757    mut first_u32: u32,
758) -> (BufferReader<'_>, InlineSpanParts) {
759    first_u32 >>= 2;
760    let span_parts_index = first_u32 & n_bits_mask(4);
761    let len = (first_u32 >> 4) & n_bits_mask(8);
762    let extends_to_next = (first_u32 & (1 << (4 + 8))) != 0;
763    let offset = first_u32 >> (4 + 8 + 1);
764    if !extends_to_next {
765        let result = InlineSpanParts {
766            span_parts_index: span_parts_index as usize,
767            text_range: TextRange::at(TextSize::new(offset), TextSize::new(len)),
768        };
769        (ptr, result)
770    } else {
771        unsafe { decode_extended_span(ptr, span_parts_index, len, offset) }
772    }
773}
774
775#[cold]
776unsafe fn decode_extended_span(
777    mut ptr: BufferReader<'_>,
778    mut span_parts_index: u32,
779    mut len: u32,
780    mut offset: u32,
781) -> (BufferReader<'_>, InlineSpanParts) {
782    unsafe {
783        let mut second_u32 = ptr.read::<u32>();
784        let extends_to_next = (second_u32 & 0b1) != 0;
785        second_u32 >>= 1;
786        if extends_to_next {
787            let third_u32 = ptr.read::<u32>();
788            let u64 = u64::from(third_u32) | (u64::from(second_u32) << u32::BITS);
789            let rest_span_parts_index = (u64 & n_bits_mask_u64(24)) as u32;
790            span_parts_index |= rest_span_parts_index << 4;
791            let rest_len = ((u64 >> 24) & n_bits_mask_u64(24)) as u32;
792            len |= rest_len << 8;
793            let rest_offset = (u64 >> (24 + 24)) as u32;
794            offset |= rest_offset << 17;
795        } else {
796            let rest_span_parts_index = second_u32 & n_bits_mask(11);
797            span_parts_index |= rest_span_parts_index << 4;
798            let rest_len = (second_u32 >> 11) & n_bits_mask(10);
799            len |= rest_len << 8;
800            let rest_offset = second_u32 >> (11 + 10);
801            offset |= rest_offset << 17;
802        };
803        let result = InlineSpanParts {
804            span_parts_index: span_parts_index as usize,
805            text_range: TextRange::at(TextSize::new(offset), TextSize::new(len)),
806        };
807        (ptr, result)
808    }
809}

811unsafe fn decode_symbol<'a>(
812    mut ptr: BufferReader<'a>,
813    first_byte: u8,
814    symbols: &[Symbol],
815) -> (BufferReader<'a>, Symbol) {
816    let mut symbol_idx = u32::from(first_byte) >> 1;
817    let extends_to_next = symbol_idx & 0b1 == 0b1;
818    symbol_idx >>= 1;
819    if extends_to_next {
820        let mut second_byte = unsafe { ptr.read::<u8>() };
821        let rest_bytes =
822            if second_byte & 0b1 == 0b1 { u32::from(unsafe { ptr.read::<u16>() }) } else { 0 };
823        second_byte >>= 1;
824        symbol_idx |= u32::from(second_byte) << 6;
825        symbol_idx |= rest_bytes << 13;
826    }
827    (ptr, symbols[symbol_idx as usize].clone())
828}

830/// The returned `u32` is the length of the children *in bytes*, if we read a subtree. Otherwise it's zero.
831unsafe fn decode<'a>(
832    mut ptr: BufferReader<'a>,
833    compressed: &[CompressedSpanPart],
834    symbols: &[Symbol],
835) -> (BufferReader<'a>, TokenTree, u32) {
836    unsafe {
837        let span_and_extra = ptr.read::<u32>();
838        let span;
839        (ptr, span) = decode_span(ptr, span_and_extra);
840        let span = compressed[span.span_parts_index].recombine(span.text_range);
841
842        if span_and_extra & 0b1 == 0b1 {
843            // An ASCII punct.
844            let char_and_half_spacing = ptr.read::<u8>();
845
846            let spacing = (span_and_extra & 0b10) | (u32::from(char_and_half_spacing) & 0b1);
847            let spacing = transmute::<u8, Spacing>(spacing as u8);
848
849            let char = char::from(char_and_half_spacing >> 1);
850
851            return (ptr, TokenTree::Leaf(Leaf::Punct(Punct { char, spacing, span })), 0);
852        } else if span_and_extra & 0b10 == 0b10 {
853            // An ident. If not `IdentKind::Normal`, an additional byte encodes the `IdentKind`
854            // (since raw or forced-keyword idents are rare).
855            let symbol_first_byte = ptr.read::<u8>();
856            let is_not_normal = (symbol_first_byte & 0b1) != 0;
857            let symbol;
858            (ptr, symbol) = decode_symbol(ptr, symbol_first_byte, symbols);
859            let kind = if is_not_normal {
860                transmute::<u8, IdentKind>(ptr.read::<u8>())
861            } else {
862                IdentKind::Normal
863            };
864
865            return (ptr, TokenTree::Leaf(Leaf::Ident(Ident { sym: symbol, span, kind })), 0);
866        }
867
868        let mut children_byte_len = 0;
869
870        let control_byte = u32::from(ptr.read::<u8>());
871        let control_byte_extra_data = control_byte >> 3;
872        let result = match control_byte & 0b111 {
873            0b000 => {
874                // Subtree, format 1:
875                //  - Same span_parts_index for open and close span.
876                //  - Subtree length is a u8.
877                //  - Subtree length in bytes is a u8.
878                //  - The length of both the open and close span is 1 - so we use the already-parsed length for the open span
879                //    for other things (we can only assume it has 8 bits available, the minimum format for a length).
880                //  - The offset between the open span's start and the close span's start is stored in 11 bits.
881                let kind = transmute::<u8, DelimiterKind>((control_byte_extra_data & 0b11) as u8);
882                let mut open_span = span;
883                let mut close_span_offset_from_open = u32::from(span.range.len());
884                open_span.range = TextRange::at(open_span.range.start(), TextSize::new(1));
885                close_span_offset_from_open |= (control_byte_extra_data >> 2) << 8;
886                let close_span = Span {
887                    range: open_span.range + TextSize::new(close_span_offset_from_open),
888                    anchor: open_span.anchor,
889                    ctx: open_span.ctx,
890                };
891                let len = u32::from(ptr.read::<u8>());
892                children_byte_len = u32::from(ptr.read::<u8>());
893
894                TokenTree::Subtree(Subtree {
895                    delimiter: Delimiter { open: open_span, close: close_span, kind },
896                    len,
897                })
898            }
899            0b001 => {
900                // Subtree, format 2:
901                //  - Same span_parts_index for open and close span.
902                //  - Subtree length is a u8.
903                //  - Subtree length in bytes is a u8.
904                //  - The open and close span have the same length. This covers many cases because most cases either give
905                //    both length 1 (the brackets themselves) or the same span (usually, the whole range they encompass).
906                //  - The offset between the open span's start and the close span's start is stored in 3 bits.
907                let kind = transmute::<u8, DelimiterKind>((control_byte_extra_data & 0b11) as u8);
908                let open_span = span;
909                let close_span_offset_from_open = control_byte_extra_data >> 2;
910                let close_span = Span {
911                    range: open_span.range + TextSize::new(close_span_offset_from_open),
912                    anchor: open_span.anchor,
913                    ctx: open_span.ctx,
914                };
915                let len = u32::from(ptr.read::<u8>());
916                children_byte_len = u32::from(ptr.read::<u8>());
917
918                TokenTree::Subtree(Subtree {
919                    delimiter: Delimiter { open: open_span, close: close_span, kind },
920                    len,
921                })
922            }
923            0b010 => {
924                // Subtree, format 3:
925                //  - Subtree length is a u8.
926                //  - Subtree length in bytes is 12 bits.
927                //  - The open and close span have the same length.
928                //  - The offset between the open span's start and the close span's start is stored in 8 bits.
929                //  - The close span's span_parts_index is stored in 7 bits.
930                // This is less efficient than formats 1 and 2 (requires two more bytes), but more efficient than the general format.
931                let kind = transmute::<u8, DelimiterKind>((control_byte_extra_data & 0b11) as u8);
932                let open_span = span;
933                let close_span_offset_from_open = u32::from(ptr.read::<u8>());
934                let len = u32::from(ptr.read::<u8>());
935                children_byte_len = u32::from(ptr.read::<u16>());
936                let mut close_span_parts_index = control_byte_extra_data >> 2;
937                close_span_parts_index |= (children_byte_len & 0b1111) << 3;
938                children_byte_len >>= 4;
939                let close_span = compressed[close_span_parts_index as usize]
940                    .recombine(open_span.range + TextSize::new(close_span_offset_from_open));
941
942                TokenTree::Subtree(Subtree {
943                    delimiter: Delimiter { open: open_span, close: close_span, kind },
944                    len,
945                })
946            }
947            0b101 => {
948                cold_path();
949
950                // Subtree, format 4 - most general format: subtree length is u32, subtree length in bytes is u32, full decoded
951                // span for close span, delimiter kind is 5 bits (not needed now, will be needed when we have different kinds of
952                // invisible delimiters).
953                let kind = transmute::<u8, DelimiterKind>(control_byte_extra_data as u8);
954                let open_span = span;
955                let close_span_start = ptr.read::<u32>();
956                let close_span;
957                (ptr, close_span) = decode_span(ptr, close_span_start);
958                let close_span =
959                    compressed[close_span.span_parts_index].recombine(close_span.text_range);
960                let len = ptr.read::<u32>();
961                children_byte_len = ptr.read::<u32>();
962
963                TokenTree::Subtree(Subtree {
964                    delimiter: Delimiter { open: open_span, close: close_span, kind },
965                    len,
966                })
967            }
968            0b110 => {
969                cold_path();
970
971                // Non-ASCII punct. Extremely rare but technically possible.
972                let spacing = transmute::<u8, Spacing>(control_byte_extra_data as u8);
973                let char = ptr.read::<char>();
974
975                TokenTree::Leaf(Leaf::Punct(Punct { char, spacing, span }))
976            }
977            0b011 | 0b100 => {
978                // Literal, format 1: the 6 bits remaining from `control_byte` decide the kind and the suffix len from a constant set
979                // (of the most common).
980                let control_byte_extra_data = control_byte >> 2;
981                let text_and_suffix_first_byte = ptr.read::<u8>();
982                let text_and_suffix;
983                (ptr, text_and_suffix) = decode_symbol(ptr, text_and_suffix_first_byte, symbols);
984                let kind = match control_byte_extra_data & 0b11 {
985                    0b00 => LitKind::Str,
986                    0b01 => LitKind::StrRaw(0),
987                    0b10 => LitKind::StrRaw(1),
988                    0b11 => LitKind::Integer,
989                    _ => unreachable!(),
990                };
991                let suffix_len = (control_byte_extra_data >> 2) as u8;
992
993                TokenTree::Leaf(Leaf::Literal(Literal { text_and_suffix, span, kind, suffix_len }))
994            }
995            0b111 => {
996                cold_path();
997
998                // Literal, format 2: most general format.
999                let kind = match control_byte_extra_data {
1000                    0 => LitKind::Byte,
1001                    1 => LitKind::Char,
1002                    2 => LitKind::Integer,
1003                    3 => LitKind::Float,
1004                    4 => LitKind::Str,
1005                    5 => LitKind::StrRaw(ptr.read::<u8>()),
1006                    6 => LitKind::ByteStr,
1007                    7 => LitKind::ByteStrRaw(ptr.read::<u8>()),
1008                    8 => LitKind::CStr,
1009                    9 => LitKind::CStrRaw(ptr.read::<u8>()),
1010                    10 => LitKind::Err(()),
1011                    _ => unreachable!(),
1012                };
1013                let suffix_len = ptr.read::<u8>();
1014                let text_and_suffix_first_byte = ptr.read::<u8>();
1015                let text_and_suffix;
1016                (ptr, text_and_suffix) = decode_symbol(ptr, text_and_suffix_first_byte, symbols);
1017
1018                TokenTree::Leaf(Leaf::Literal(Literal { text_and_suffix, span, kind, suffix_len }))
1019            }
1020            _ => unreachable!(),
1021        };
1022        (ptr, result, children_byte_len)
1023    }
1024}

1026#[derive(Clone, Copy)]
1027pub(crate) struct TokenTreesSlice<'a> {
1028    current: BufferReader<'a>,
1029    end: BufferReader<'a>,
1030    span_parts: &'a [CompressedSpanPart],
1031    symbols: &'a [Symbol],
1032}
1033
1034unsafe impl Send for TokenTreesSlice<'_> {}
1035unsafe impl Sync for TokenTreesSlice<'_> {}
1036
1037impl<'a> TokenTreesSlice<'a> {
1038    #[inline]
1039    fn new(top_subtree: &'a TopSubtree) -> Self {
1040        let (current, end) = BufferReader::start_end(&top_subtree.buffer);
1041        Self { current, end, span_parts: &top_subtree.span_parts, symbols: &top_subtree.symbols }
1042    }
1043
1044    #[inline]
1045    pub(crate) fn empty() -> Self {
1046        let (current, end) = BufferReader::start_end(&[]);
1047        Self { current, end, span_parts: &[], symbols: &[] }
1048    }
1049
1050    pub(crate) fn advance(&mut self) -> Option<TokenTree> {
1051        if self.current == self.end {
1052            return None;
1053        }
1054
1055        let (new_current, token_tree, _children_byte_len) =
1056            unsafe { decode(self.current, self.span_parts, self.symbols) };
1057        self.current = new_current;
1058        Some(token_tree)
1059    }
1060
1061    /// This is like `advance()`, but when encountering a subtree, it changes `self` to skip it and returns a
1062    /// slice into it (what `advance()` would have done to `self`).
1063    pub(crate) fn advance_skip_subtree(&mut self) -> Option<(TokenTree, TokenTreesSlice<'a>)> {
1064        if self.current == self.end {
1065            return None;
1066        }
1067
1068        let (new_current, token_tree, children_byte_len) =
1069            unsafe { decode(self.current, self.span_parts, self.symbols) };
1070        self.current = new_current;
1071        let subtree_slice = *self;
1072        unsafe { self.current.skip(children_byte_len as usize) };
1073        Some((token_tree, subtree_slice))
1074    }
1075
1076    pub(crate) fn iter(mut self) -> impl Iterator<Item = TokenTree> {
1077        std::iter::from_fn(move || self.advance())
1078    }
1079}

1081#[derive(Clone, PartialEq, Eq, Hash)]
1082pub struct TopSubtree {
1083    buffer: Box<[u8]>,
1084    /// The last two are the top subtree's open and close span, in this order.
1085    span_parts: Box<[CompressedSpanPart]>,
1086    symbols: Box<[Symbol]>,
1087    len: usize,
1088}
1089
1090impl TopSubtree {
1091    pub fn empty(span: DelimSpan) -> Self {
1092        encode_all(
1093            vec![TokenTree::Subtree(Subtree {
1094                delimiter: Delimiter::invisible_delim_spanned(span),
1095                len: 0,
1096            })]
1097            .into_iter(),
1098            FxHashMap::default(),
1099            FxHashMap::default(),
1100        )
1101    }
1102
1103    pub fn invisible_from_leaves<const N: usize>(delim_span: Span, leaves: [Leaf; N]) -> Self {
1104        Self::from_serialized(
1105            std::iter::chain(
1106                [TokenTree::Subtree(Subtree {
1107                    delimiter: Delimiter::invisible_spanned(delim_span),
1108                    len: leaves.len() as u32,
1109                })],
1110                leaves.into_iter().map(TokenTree::Leaf),
1111            )
1112            .collect(),
1113        )
1114    }
1115
1116    pub fn from_token_trees(delimiter: Delimiter, token_trees: TokenTreesView<'_>) -> Self {
1117        let mut builder = TopSubtreeBuilder::new(delimiter);
1118        builder.extend_with_tt(token_trees);
1119        builder.build()
1120    }
1121
1122    pub fn from_serialized(tts: Vec<TokenTree>) -> Self {
1123        let (span_frequencies, symbols) = compute_span_frequencies_and_symbols(&tts[1..]); // Do not include the top subtree.
1124        encode_all(tts.into_iter(), span_frequencies, symbols)

1189#[derive(Debug, Clone)]
1190pub struct TopSubtreeBuilder {
1191    unclosed_subtree_indices: Vec<usize>,
1192    token_trees: Vec<TokenTree>,
1193    span_parts_frequencies: FxHashMap<CompressedSpanPart, usize>,
1194    last_closed_subtree: Option<usize>,
1195    symbol_frequencies: FxHashMap<Symbol, usize>,
1196}
1197
1198impl TopSubtreeBuilder {
1199    fn insert_span(&mut self, span: &Span) {
1200        *self.span_parts_frequencies.entry(CompressedSpanPart::from_span(span)).or_insert(0) += 1;
1201    }
1202
1203    fn remove_span(span_parts_frequencies: &mut FxHashMap<CompressedSpanPart, usize>, span: &Span) {
1204        let hash_map::Entry::Occupied(mut entry) =
1205            span_parts_frequencies.entry(CompressedSpanPart::from_span(span))
1206        else {
1207            panic!("span not present");
1208        };
1209        *entry.get_mut() -= 1;
1210        if *entry.get() == 0 {
1211            entry.remove();
1212        }
1213    }
1214
1215    fn insert_symbol(&mut self, symbol: Symbol) {
1216        *self.symbol_frequencies.entry(symbol).or_insert(0) += 1;
1217    }
1218
1219    fn remove_symbol(symbol_frequencies: &mut FxHashMap<Symbol, usize>, symbol: Symbol) {
1220        let hash_map::Entry::Occupied(mut entry) = symbol_frequencies.entry(symbol) else {
1221            panic!("span not present");
1222        };
1223        *entry.get_mut() -= 1;
1224        if *entry.get() == 0 {
1225            entry.remove();
1226        }
1227    }
1228
1229    pub fn new(top_delimiter: Delimiter) -> Self {
1230        let mut result = Self {
1231            unclosed_subtree_indices: Vec::new(),
1232            token_trees: Vec::new(),
1233            span_parts_frequencies: FxHashMap::default(),
1234            last_closed_subtree: None,
1235            symbol_frequencies: FxHashMap::default(),
1236        };
1237        // Do not insert the top delimiters, they have their own place because we sometimes need to change them.
1238        result.token_trees.push(TokenTree::Subtree(Subtree { delimiter: top_delimiter, len: 0 }));
1239        result
1240    }
1241
1242    /// Not to be exposed, this assumes the subtree's children will be filled in immediately.
1243    fn push_subtree(&mut self, subtree: Subtree) {
1244        self.insert_span(&subtree.delimiter.open);
1245        self.insert_span(&subtree.delimiter.close);
1246        self.token_trees.push(subtree.into());
1247    }
1248
1249    pub fn open(&mut self, delimiter_kind: DelimiterKind, open_span: Span) {
1250        self.insert_span(&open_span);
1251        let subtree_idx = self.token_trees.len();
1252        self.token_trees.push(TokenTree::Subtree(Subtree {
1253            delimiter: Delimiter { open: open_span, close: open_span, kind: delimiter_kind },
1254            len: 0, // Will be overwritten on close.
1255        }));
1256        self.unclosed_subtree_indices.push(subtree_idx);
1257    }
1258
1259    pub fn close(&mut self, close_span: Span) {
1260        self.insert_span(&close_span);
1261
1262        let last_unclosed_index = self
1263            .unclosed_subtree_indices
1264            .pop()
1265            .expect("attempt to close a `tt::Subtree` when none is open");
1266        let token_trees_len = self.token_trees.len();
1267        let TokenTree::Subtree(Subtree { delimiter: Delimiter { open: _, close, kind: _ }, len }) =
1268            &mut self.token_trees[last_unclosed_index]
1269        else {
1270            unreachable!("unclosed token tree is always a subtree");
1271        };
1272        *len = (token_trees_len - last_unclosed_index - 1) as u32;
1273        *close = close_span;
1274        self.last_closed_subtree = Some(last_unclosed_index);
1275    }
1276
1277    /// You cannot call this consecutively, it will only work once after close.
1278    pub fn remove_last_subtree_if_invisible(&mut self) {
1279        let Some(last_subtree_idx) = self.last_closed_subtree else { return };
1280        if let TokenTree::Subtree(Subtree {
1281            delimiter: Delimiter { kind: DelimiterKind::Invisible, .. },
1282            ..
1283        }) = self.token_trees[last_subtree_idx]
1284        {
1285            self.token_trees.remove(last_subtree_idx);
1286        }
1287        self.last_closed_subtree = None;
1288    }
1289
1290    pub fn push(&mut self, leaf: Leaf) {
1291        self.insert_span(leaf.span());
1292        if let Some(symbol) = leaf.symbol() {
1293            self.insert_symbol(symbol.clone());
1294        }
1295        self.token_trees.push(leaf.into());
1296    }
1297
1298    fn push_token_tree(&mut self, tt: TokenTree) {
1299        match tt {
1300            TokenTree::Leaf(leaf) => self.push(leaf),
1301            TokenTree::Subtree(subtree) => self.push_subtree(subtree),
1302        }
1303    }
1304
1305    pub fn extend(&mut self, leaves: impl IntoIterator<Item = Leaf>) {
1306        leaves.into_iter().for_each(|leaf| self.push(leaf));
1307    }
1308
1309    pub fn extend_with_tt(&mut self, tt: TokenTreesView<'_>) {
1310        tt.iter_flat_tokens().for_each(|tt| self.push_token_tree(tt));
1311    }
1312
1313    /// Like [`Self::extend_with_tt()`], but makes sure the new tokens will never be
1314    /// joint with whatever comes after them.
1315    pub fn extend_with_tt_alone(&mut self, tt: TokenTreesView<'_>) {
1316        self.extend_with_tt(tt);
1317        if !tt.is_empty()
1318            && let Some(TokenTree::Leaf(Leaf::Punct(Punct { spacing, .. }))) =
1319                self.token_trees.last_mut()
1320        {
1321            *spacing = Spacing::Alone;
1322        }
1323    }
1324
1325    pub fn expected_delimiters(&self) -> impl Iterator<Item = DelimiterKind> {
1326        self.unclosed_subtree_indices.iter().rev().map(|&subtree_idx| {
1327            let TokenTree::Subtree(Subtree { delimiter, .. }) = self.token_trees[subtree_idx]
1328            else {
1329                unreachable!("unclosed token tree is always a subtree")
1330            };
1331            delimiter.kind
1332        })
1333    }

1335    /// Builds, and remove the top subtree if it has only one subtree child.
1336    pub fn build_skip_top_subtree(mut self) -> TopSubtree {
1337        assert!(
1338            self.unclosed_subtree_indices.is_empty(),
1339            "attempt to build an unbalanced `TopSubtreeBuilder`"
1340        );
1341        let tt_len = self.token_trees.len();
1342        if let Some(&TokenTree::Subtree(Subtree { len, delimiter })) = self.token_trees.get(1)
1343            && (len as usize) == (tt_len - 2)
1344        {
1345            // The top subtree's delimiters should not be included.
1346            Self::remove_span(&mut self.span_parts_frequencies, &delimiter.open);
1347            Self::remove_span(&mut self.span_parts_frequencies, &delimiter.close);
1348
1349            let mut token_trees = self.token_trees.into_iter();
1350            token_trees.next(); // Remove the first subtree.
1351            encode_all(token_trees, self.span_parts_frequencies, self.symbol_frequencies)
1352        } else {
1353            self.build()
1354        }
1355    }
1356
1357    pub fn build(mut self) -> TopSubtree {
1358        assert!(
1359            self.unclosed_subtree_indices.is_empty(),
1360            "attempt to build an unbalanced `TopSubtreeBuilder`"
1361        );
1362        let tts_len = self.token_trees.len();
1363        let TokenTree::Subtree(top_subtree) = &mut self.token_trees[0] else {
1364            panic!("first token tree must be a subtree");
1365        };
1366        top_subtree.len = (tts_len - 1).try_into().unwrap();
1367        encode_all(
1368            self.token_trees.into_iter(),
1369            self.span_parts_frequencies,
1370            self.symbol_frequencies,
1371        )
1372    }

1374    pub fn restore_point(&mut self) -> SubtreeBuilderRestorePoint {
1375        // We reset the `last_closed_subtree`, since restoring from a restore point doesn't play well with removing the last subtree.
1376        self.last_closed_subtree = None;
1377        SubtreeBuilderRestorePoint {
1378            unclosed_subtree_indices_len: self.unclosed_subtree_indices.len(),
1379            token_trees_len: self.token_trees.len(),
1380        }
1381    }
1382
1383    pub fn restore(&mut self, restore_point: SubtreeBuilderRestorePoint) {
1384        if restore_point.token_trees_len >= self.token_trees.len() {
1385            // This means we restored twice, potentially with an earlier restore point first.
1386            return;
1387        }
1388
1389        for tt in &self.token_trees[restore_point.token_trees_len..] {
1390            match tt {
1391                TokenTree::Leaf(leaf) => {
1392                    Self::remove_span(&mut self.span_parts_frequencies, leaf.span());
1393
1394                    if let Some(symbol) = leaf.symbol() {
1395                        Self::remove_symbol(&mut self.symbol_frequencies, symbol.clone());
1396                    }
1397                }
1398                TokenTree::Subtree(subtree) => {
1399                    Self::remove_span(&mut self.span_parts_frequencies, &subtree.delimiter.open);
1400                    Self::remove_span(&mut self.span_parts_frequencies, &subtree.delimiter.close);
1401                }
1402            }
1403        }
1404
1405        self.unclosed_subtree_indices.truncate(restore_point.unclosed_subtree_indices_len);
1406        self.token_trees.truncate(restore_point.token_trees_len);
1407        self.last_closed_subtree = None;
1408    }
1409}
