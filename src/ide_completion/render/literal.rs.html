44fn render(
45    ctx @ RenderContext { completion, .. }: RenderContext<'_, '_>,
46    path_ctx: &PathCompletionCtx<'_>,
47    thing: Variant,
48    path: hir::ModPath,
49) -> Option<Builder> {
50    let db = completion.db;
51    let mut kind = thing.kind(db);
52    let should_add_parens = !matches!(
53        path_ctx,
54        PathCompletionCtx { has_call_parens: true, .. }
55            | PathCompletionCtx { kind: PathKind::Use | PathKind::Type { .. }, .. }
56    );
57
58    let fields = thing.fields(completion)?;
59    let short_len = match thing {
60        Variant::Struct(_) => 1,
61        Variant::EnumVariant(_) => 2,
62    };
63    let short_qualified_name = hir::ModPath::from_segments(
64        hir::PathKind::Plain,
65        path.segments().iter().skip(path.segments().len().saturating_sub(short_len)).cloned(),
66    );
67    let qualified = path.len() > 1;
68    let (qualified_name, escaped_qualified_name) = (
69        path.display_verbatim(ctx.db()).to_string(),
70        path.display(ctx.db(), completion.edition).to_string(),
71    );
72    let snippet_cap = ctx.snippet_cap();
73
74    let mut rendered = match kind {
75        StructKind::Tuple if should_add_parens => {
76            render_tuple_lit(completion, snippet_cap, &fields, &escaped_qualified_name)
77        }
78        StructKind::Record if should_add_parens => {
79            render_record_lit(completion, snippet_cap, &fields, &escaped_qualified_name)
80        }
81        _ => RenderedLiteral {
82            literal: escaped_qualified_name.clone(),
83            detail: escaped_qualified_name,
84        },
85    };
86
87    if snippet_cap.is_some() {
88        rendered.literal.push_str("$0");
89    }
90
91    // only show name in label if not adding parens
92    if !should_add_parens {
93        kind = StructKind::Unit;
94    }
95    let label = format_literal_label(&qualified_name, kind, snippet_cap);
96    let lookup = if qualified {
97        format_literal_lookup(
98            &short_qualified_name.display(ctx.db(), completion.edition).to_string(),
99            kind,
100        )
101    } else {
102        format_literal_lookup(&qualified_name, kind)
103    };
104
105    let mut item = CompletionItem::new(
106        CompletionItemKind::SymbolKind(thing.symbol_kind()),
107        ctx.source_range(),
108        label,
109        completion.edition,
110    );
111
112    item.lookup_by(lookup);
113    item.detail(rendered.detail);
114
115    match snippet_cap {
116        Some(snippet_cap) => item.insert_snippet(snippet_cap, rendered.literal).trigger_call_info(),
117        None => item.insert_text(rendered.literal),
118    };
119
120    item.set_documentation(thing.docs(db)).set_deprecated(thing.is_deprecated(&ctx));
121
122    let ty = ctx.completion.rebase_ty(&thing.ty(db));
123    item.set_relevance(CompletionRelevance {
124        type_match: compute_type_match(ctx.completion, &ty),
125        // function is a misnomer here, this is more about constructor information
126        function: Some(CompletionRelevanceFn {
127            has_params: !fields.is_empty(),
128            has_self_param: false,
129            return_type: CompletionRelevanceReturnType::DirectConstructor,
130        }),
131        ..ctx.completion_relevance()
132    });
133
134    super::path_ref_match(completion, path_ctx, &ty, &mut item);
135
136    if let Some(import_to_add) = ctx.import_to_add {
137        item.add_import(import_to_add);
138    }
139    Some(item)
140}

142#[derive(Clone, Copy)]
143enum Variant {
144    Struct(hir::Struct),
145    EnumVariant(hir::EnumVariant),
146}
147
148impl Variant {
149    fn fields(self, ctx: &CompletionContext<'_, '_>) -> Option<Vec<hir::Field>> {
150        let fields = match self {
151            Variant::Struct(it) => it.fields(ctx.db),
152            Variant::EnumVariant(it) => it.fields(ctx.db),
153        };
154        let (visible_fields, fields_omitted) = match self {
155            Variant::Struct(it) => visible_fields(ctx, &fields, it)?,
156            Variant::EnumVariant(it) => visible_fields(ctx, &fields, it)?,
157        };
158        if !fields_omitted { Some(visible_fields) } else { None }
159    }
160
161    fn kind(self, db: &dyn HirDatabase) -> StructKind {
162        match self {
163            Variant::Struct(it) => it.kind(db),
164            Variant::EnumVariant(it) => it.kind(db),
165        }
166    }
167
168    fn symbol_kind(self) -> SymbolKind {
169        match self {
170            Variant::Struct(_) => SymbolKind::Struct,
171            Variant::EnumVariant(_) => SymbolKind::Variant,
172        }
173    }
174
175    fn docs(self, db: &dyn HirDatabase) -> Option<Documentation<'_>> {
176        match self {
177            Variant::Struct(it) => it.docs(db),
178            Variant::EnumVariant(it) => it.docs(db),
179        }
180    }
181
182    fn is_deprecated(self, ctx: &RenderContext<'_, '_>) -> bool {
183        match self {
184            Variant::Struct(it) => {
185                ctx.is_deprecated(it, None /* structs can't be assoc items */)
186            }
187            Variant::EnumVariant(it) => ctx.is_variant_deprecated(it),
188        }
189    }
190
191    fn ty(self, db: &dyn HirDatabase) -> hir::Type<'_> {
192        match self {
193            Variant::Struct(it) => it.ty(db),
194            Variant::EnumVariant(it) => it.parent_enum(db).ty(db),
195        }
196    }
