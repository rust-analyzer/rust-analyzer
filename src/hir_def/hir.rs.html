1//! This module describes hir-level representation of expressions.
2//!
3//! This representation is:
4//!
5//! 1. Identity-based. Each expression has an `id`, so we can distinguish
6//!    between different `1` in `1 + 1`.
7//! 2. Independent of syntax. Though syntactic provenance information can be
8//!    attached separately via id-based side map.
9//! 3. Unresolved. Paths are stored as sequences of names, and not as defs the
10//!    names refer to.
11//! 4. Desugared. There's no `if let`.
12//!
13//! See also a neighboring [`body`] module.
14//!
15//! [`body`]: crate::expr_store::body
16

79#[derive(Copy, Clone, Hash, PartialEq, Eq, salsa::SalsaValue)]
80pub struct ExprOrPatIdPacked(u32);
81
82const _: () = assert!(mem::size_of::<ExprOrPatIdPacked>() == mem::size_of::<u32>());
83
84impl ExprOrPatIdPacked {
85    const PAT_BIT: u32 = 1 << (u32::BITS - 1);
86    const INDEX_MASK: u32 = !Self::PAT_BIT;
87
88    pub fn unpack(self) -> ExprOrPatId {
89        match self.is_expr() {
90            true => ExprOrPatId::ExprId(ExprId::from_raw(RawIdx::from_u32(self.0))),
91            false => {
92                ExprOrPatId::PatId(PatId::from_raw(RawIdx::from_u32(self.0 & Self::INDEX_MASK)))
93            }
94        }
95    }
96
97    #[inline]
98    pub fn as_expr(self) -> Option<ExprId> {
99        self.is_expr().then(|| ExprId::from_raw(RawIdx::from_u32(self.0)))
100    }
101
102    #[inline]
103    pub fn is_expr(&self) -> bool {
104        self.0 & Self::PAT_BIT == 0
105    }
106
107    #[inline]
108    pub fn as_pat(self) -> Option<PatId> {
109        self.is_pat().then(|| PatId::from_raw(RawIdx::from_u32(self.0 & Self::INDEX_MASK)))
110    }
111
112    #[inline]
113    pub fn is_pat(&self) -> bool {
114        self.0 & Self::PAT_BIT != 0
115    }
116}
117
118impl fmt::Debug for ExprOrPatIdPacked {
119    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
120        match self.unpack() {
121            ExprOrPatId::ExprId(id) => f.debug_tuple("ExprId").field(&id).finish(),
122            ExprOrPatId::PatId(id) => f.debug_tuple("PatId").field(&id).finish(),
123        }
124    }
125}

127impl From<ExprId> for ExprOrPatIdPacked {
128    fn from(value: ExprId) -> Self {
129        let value = value.into_raw().into_u32();
130        // virtually impossible to have IDs that high
131        debug_assert_eq!(value & Self::PAT_BIT, 0);
132        Self(value)
133    }
134}

136impl From<PatId> for ExprOrPatId {
137    fn from(value: PatId) -> Self {
138        ExprOrPatId::PatId(value)
139    }
140}
141impl From<ExprId> for ExprOrPatId {
142    fn from(value: ExprId) -> Self {
143        ExprOrPatId::ExprId(value)
144    }
145}

147impl From<PatId> for ExprOrPatIdPacked {
148    fn from(value: PatId) -> Self {
149        let value = value.into_raw().into_u32();
150        // virtually impossible to have IDs that high
151        debug_assert_eq!(value & Self::PAT_BIT, 0);
152        Self(value | Self::PAT_BIT)
153    }
154}

225impl From<ast::LiteralKind> for Literal {
226    fn from(ast_lit_kind: ast::LiteralKind) -> Self {
227        use ast::LiteralKind;
228        match ast_lit_kind {
229            LiteralKind::IntNumber(lit) => {
230                if let builtin @ Some(_) = lit.suffix().and_then(BuiltinFloat::from_suffix) {
231                    Literal::Float(
232                        FloatTypeWrapper::new(Symbol::intern(&lit.value_string())),
233                        builtin,
234                    )
235                } else if let builtin @ Some(_) = lit.suffix().and_then(BuiltinUint::from_suffix) {
236                    Literal::Uint(lit.value().unwrap_or(0), builtin)
237                } else {
238                    let builtin = lit.suffix().and_then(BuiltinInt::from_suffix);
239                    Literal::Int(lit.value().unwrap_or(0) as i128, builtin)
240                }
241            }
242            LiteralKind::FloatNumber(lit) => {
243                let ty = lit.suffix().and_then(BuiltinFloat::from_suffix);
244                Literal::Float(FloatTypeWrapper::new(Symbol::intern(&lit.value_string())), ty)
245            }
246            LiteralKind::ByteString(bs) => {
247                let text = bs.value().map_or_else(|_| Default::default(), Box::from);
248                Literal::ByteString(text)
249            }
250            LiteralKind::String(s) => {
251                let text = s.value().map_or_else(|_| Symbol::empty(), |it| Symbol::intern(&it));
252                Literal::String(text)
253            }
254            LiteralKind::CString(s) => {
255                let text = s.value().map_or_else(|_| Default::default(), Box::from);
256                Literal::CString(text)
257            }
258            LiteralKind::Byte(b) => {
259                Literal::Uint(b.value().unwrap_or_default() as u128, Some(BuiltinUint::U8))
260            }
261            LiteralKind::Char(c) => Literal::Char(c.value().unwrap_or_default()),
262            LiteralKind::Bool(val) => Literal::Bool(val),
263        }
264    }
265}

402impl Expr {
403    pub fn precedence(&self) -> ast::prec::ExprPrecedence {
404        use ast::prec::ExprPrecedence;
405
406        match self {
407            Expr::Array(_)
408            | Expr::InlineAsm(_)
409            | Expr::Block { .. }
410            | Expr::Const(_)
411            | Expr::If { .. }
412            | Expr::Literal(_)
413            | Expr::Loop { .. }
414            | Expr::Match { .. }
415            | Expr::Missing
416            | Expr::Path(_)
417            | Expr::RecordLit { .. }
418            | Expr::Tuple { .. }
419            | Expr::OffsetOf(_)
420            | Expr::Underscore
421            | Expr::IncludeBytes => ExprPrecedence::Unambiguous,
422
423            Expr::Await { .. }
424            | Expr::Call { .. }
425            | Expr::Field { .. }
426            | Expr::Index { .. }
427            | Expr::MethodCall { .. } => ExprPrecedence::Postfix,
428
429            Expr::Let { .. } | Expr::UnaryOp { .. } | Expr::Ref { .. } => ExprPrecedence::Prefix,
430
431            Expr::Cast { .. } => ExprPrecedence::Cast,
432
433            Expr::BinaryOp { op, .. } => match op {
434                None => ExprPrecedence::Unambiguous,
435                Some(BinaryOp::LogicOp(LogicOp::Or)) => ExprPrecedence::LOr,
436                Some(BinaryOp::LogicOp(LogicOp::And)) => ExprPrecedence::LAnd,
437                Some(BinaryOp::CmpOp(_)) => ExprPrecedence::Compare,
438                Some(BinaryOp::Assignment { .. }) => ExprPrecedence::Assign,
439                Some(BinaryOp::ArithOp(arith_op)) => match arith_op {
440                    ArithOp::Add | ArithOp::Sub => ExprPrecedence::Sum,
441                    ArithOp::Mul | ArithOp::Div | ArithOp::Rem => ExprPrecedence::Product,
442                    ArithOp::Shl | ArithOp::Shr => ExprPrecedence::Shift,
443                    ArithOp::BitXor => ExprPrecedence::BitXor,
444                    ArithOp::BitOr => ExprPrecedence::BitOr,
445                    ArithOp::BitAnd => ExprPrecedence::BitAnd,
446                },
447            },
448
449            Expr::Assignment { .. } => ExprPrecedence::Assign,
450
451            Expr::Become { .. }
452            | Expr::Break { .. }
453            | Expr::Closure { .. }
454            | Expr::Return { .. }
455            | Expr::Yeet { .. }
456            | Expr::Yield { .. } => ExprPrecedence::Jump,
457
458            Expr::Continue { .. } => ExprPrecedence::Unambiguous,
459        }
460    }
461}

497#[derive(Clone, Copy, PartialEq, Eq, Hash)]
498pub struct AsmOptions(u16);
499bitflags::bitflags! {
500    impl AsmOptions: u16 {
501        const PURE            = 1 << 0;
502        const NOMEM           = 1 << 1;
503        const READONLY        = 1 << 2;
504        const PRESERVES_FLAGS = 1 << 3;
505        const NORETURN        = 1 << 4;
506        const NOSTACK         = 1 << 5;
507        const ATT_SYNTAX      = 1 << 6;
508        const RAW             = 1 << 7;
509        const MAY_UNWIND      = 1 << 8;
510    }
511}
512
513impl AsmOptions {
514    pub const COUNT: usize = Self::all().bits().count_ones() as usize;
515
516    pub const GLOBAL_OPTIONS: Self = Self::ATT_SYNTAX.union(Self::RAW);
517    pub const NAKED_OPTIONS: Self = Self::ATT_SYNTAX.union(Self::RAW).union(Self::NORETURN);
518
519    pub fn human_readable_names(&self) -> Vec<&'static str> {
520        let mut options = vec![];
521
522        if self.contains(AsmOptions::PURE) {
523            options.push("pure");
524        }
525        if self.contains(AsmOptions::NOMEM) {
526            options.push("nomem");
527        }
528        if self.contains(AsmOptions::READONLY) {
529            options.push("readonly");
530        }
531        if self.contains(AsmOptions::PRESERVES_FLAGS) {
532            options.push("preserves_flags");
533        }
534        if self.contains(AsmOptions::NORETURN) {
535            options.push("noreturn");
536        }
537        if self.contains(AsmOptions::NOSTACK) {
538            options.push("nostack");
539        }
540        if self.contains(AsmOptions::ATT_SYNTAX) {
541            options.push("att_syntax");
542        }
543        if self.contains(AsmOptions::RAW) {
544            options.push("raw");
545        }
546        if self.contains(AsmOptions::MAY_UNWIND) {
547            options.push("may_unwind");
548        }
549
550        options
551    }
552}

695/// Explicit binding annotations given in the HIR for a binding. Note
696/// that this is not the final binding *mode* that we infer after type
697/// inference.
698#[derive(Clone, PartialEq, Eq, Debug, Copy)]
699pub enum BindingAnnotation {
700    /// No binding annotation given: this means that the final binding mode
701    /// will depend on whether we have skipped through a `&` reference
702    /// when matching. For example, the `x` in `Some(x)` will have binding
703    /// mode `None`; if you do `let Some(x) = &Some(22)`, it will
704    /// ultimately be inferred to be by-reference.
705    Unannotated,
706
707    /// Annotated with `mut x` -- could be either ref or not, similar to `None`.
708    Mutable,
709
710    /// Annotated as `ref`, like `ref x`
711    Ref,
712
713    /// Annotated as `ref mut x`.
714    RefMut,
715}
716
717impl BindingAnnotation {
718    pub fn new(is_mutable: bool, is_ref: bool) -> Self {
719        match (is_mutable, is_ref) {
720            (true, true) => BindingAnnotation::RefMut,
721            (false, true) => BindingAnnotation::Ref,
722            (true, false) => BindingAnnotation::Mutable,
723            (false, false) => BindingAnnotation::Unannotated,
724        }
725    }
726}

734    /// <https://doc.rust-lang.org/stable/error_codes/E0408.html>
735    NotBoundAcrossAll,
736}

748#[derive(Debug, Clone, Eq, PartialEq)]
749pub struct RecordFieldPat {
750    pub name: Name,
751    pub pat: PatId,
752}
753
754/// Close relative to rustc's hir::PatKind
755#[derive(Debug, Clone, Eq, PartialEq)]
756pub enum Pat {
757    Missing,
758    /// A rest pattern. Not valid outside special context.
759    Rest,
760    Wild,
761    Tuple {
762        args: Box<[PatId]>,
763        ellipsis: Option<u32>,
764    },
765    Or(Box<[PatId]>),
766    Record {
767        path: Path,
768        args: Box<[RecordFieldPat]>,
769        ellipsis: bool,
770    },
771    Range {
772        start: Option<ExprId>,
773        end: Option<ExprId>,
774        range_type: RangeOp,
775    },
776    Slice {
777        prefix: Box<[PatId]>,
778        slice: Option<PatId>,
779        suffix: Box<[PatId]>,
780    },
781    Path(Path),
782    Lit(ExprId),
783    Bind {
784        id: BindingId,
785        subpat: Option<PatId>,
786    },
787    TupleStruct {
788        path: Path,
789        args: Box<[PatId]>,
790        ellipsis: Option<u32>,
791    },
792    Ref {
793        pat: PatId,
794        mutability: Mutability,
795    },
796    Box {
797        inner: PatId,
798    },
799    Deref {
800        inner: PatId,
801    },
802    NotNull,
803    /// An expression inside a pattern. That can only occur inside assignments.
804    ///
805    /// E.g. in `(a, *b) = (1, &mut 2)`, `*b` is an expression.
806    Expr(ExprId),
807}
808
809impl Pat {
810    pub fn walk_child_pats(&self, mut f: impl FnMut(PatId)) {
811        match self {
812            Pat::Range { .. }
813            | Pat::Lit(..)
814            | Pat::Path(..)
815            | Pat::Wild
816            | Pat::Missing
817            | Pat::Rest
818            | Pat::Expr(_)
819            | Pat::NotNull => {}
820            Pat::Bind { subpat, .. } => {
821                subpat.iter().copied().for_each(f);
822            }
823            Pat::Or(args) | Pat::Tuple { args, .. } | Pat::TupleStruct { args, .. } => {
824                args.iter().copied().for_each(f);
825            }
826            Pat::Ref { pat, .. } => f(*pat),
827            Pat::Slice { prefix, slice, suffix } => {
828                let total_iter = prefix.iter().chain(slice.iter()).chain(suffix.iter());
829                total_iter.copied().for_each(f);
830            }
831            Pat::Record { args, .. } => {
832                args.iter().map(|f| f.pat).for_each(f);
833            }
834            Pat::Box { inner } | Pat::Deref { inner } => f(*inner),
835        }
836    }
837}
