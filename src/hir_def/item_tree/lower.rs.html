60    pub(super) fn lower_module_items(mut self, item_owner: &dyn HasModuleItem) -> ItemTree {
61        self.top_level = item_owner.items().filter_map(|item| self.lower_mod_item(&item)).collect();
62        self.tree.vis.arena = self.visibilities.into_iter().collect();
63        self.tree.top_level = self.top_level.into_boxed_slice();
64        self.tree
65    }

67    pub(super) fn lower_macro_stmts(mut self, stmts: ast::MacroStmts) -> ItemTree {
68        self.top_level = stmts
69            .statements()
70            .filter_map(|stmt| {
71                match stmt {
72                    ast::Stmt::Item(item) => Some(item),
73                    // Macro calls can be both items and expressions. The syntax library always treats
74                    // them as expressions here, so we undo that.
75                    ast::Stmt::ExprStmt(es) => match es.expr()? {
76                        ast::Expr::MacroExpr(expr) => {
77                            cov_mark::hit!(macro_call_in_macro_stmts_is_added_to_item_tree);
78                            Some(expr.macro_call()?.into())
79                        }
80                        _ => None,
81                    },
82                    _ => None,
83                }
84            })
85            .filter_map(|item| self.lower_mod_item(&item))
86            .collect();
87
88        if let Some(ast::Expr::MacroExpr(tail_macro)) = stmts.expr()
89            && let Some(call) = tail_macro.macro_call()
90        {
91            cov_mark::hit!(macro_stmt_with_trailing_macro_expr);
92            if let Some(mod_item) = self.lower_mod_item(&call.into()) {
93                self.top_level.push(mod_item);
94            }
95        }
96
97        self.tree.vis.arena = self.visibilities.into_iter().collect();
98        self.tree.top_level = self.top_level.into_boxed_slice();
99        self.tree
100    }

102    pub(super) fn lower_block(mut self, block: &ast::BlockExpr) -> ItemTree {
103        self.add_attrs(ModItemId::TOP_OWNER, self.lower_attrs(block));
104        self.top_level = block
105            .statements()
106            .filter_map(|stmt| match stmt {
107                ast::Stmt::Item(item) => self.lower_mod_item(&item),
108                // Macro calls can be both items and expressions. The syntax library always treats
109                // them as expressions here, so we undo that.
110                ast::Stmt::ExprStmt(es) => match es.expr()? {
111                    ast::Expr::MacroExpr(expr) => self.lower_mod_item(&expr.macro_call()?.into()),
112                    _ => None,
113                },
114                _ => None,
115            })
116            .collect();
117        if let Some(ast::Expr::MacroExpr(expr)) = block.tail_expr()
118            && let Some(call) = expr.macro_call()
119            && let Some(mod_item) = self.lower_mod_item(&call.into())
120        {
121            self.top_level.push(mod_item);
122        }
123        self.tree.vis.arena = self.visibilities.into_iter().collect();
124        self.tree.top_level = self.top_level.into_boxed_slice();
125        self.tree
126    }

128    fn lower_mod_item(&mut self, item: &ast::Item) -> Option<ModItemId> {
129        let mod_item = match item {
130            ast::Item::Struct(ast) => self.lower_struct(ast)?,
131            ast::Item::Union(ast) => self.lower_union(ast)?,
132            ast::Item::Enum(ast) => self.lower_enum(ast)?,
133            ast::Item::Fn(ast) => self.lower_function(ast)?,
134            ast::Item::TypeAlias(ast) => self.lower_type_alias(ast)?,
135            ast::Item::Static(ast) => self.lower_static(ast)?,
136            ast::Item::Const(ast) => self.lower_const(ast),
137            ast::Item::Module(ast) => self.lower_module(ast)?,
138            ast::Item::Trait(ast) => self.lower_trait(ast)?,
139            ast::Item::Impl(ast) => self.lower_impl(ast),
140            ast::Item::Use(ast) => self.lower_use(ast)?,
141            ast::Item::ExternCrate(ast) => self.lower_extern_crate(ast)?,
142            ast::Item::MacroCall(ast) => self.lower_macro_call(ast)?,
143            ast::Item::MacroRules(ast) => self.lower_macro_rules(ast)?,
144            ast::Item::MacroDef(ast) => self.lower_macro_def(ast)?,
145            ast::Item::ExternBlock(ast) => self.lower_extern_block(ast),
146            // FIXME: Handle `global_asm!()`.
147            ast::Item::AsmExpr(_) => return None,
148        };
149        let attrs = self.lower_attrs(item);
150        self.add_attrs(mod_item, attrs);
151
152        Some(mod_item)
153    }

173    fn lower_struct(&mut self, strukt: &ast::Struct) -> Option<ModItemId> {
174        let visibility = self.lower_visibility_from_ast(strukt);
175        let visibility_id = self.alloc_visibility(visibility.clone());
176        let name = strukt.name()?.as_name();
177        let ast_id = self.source_ast_id_map.ast_id(strukt);
178        let value_ns_ctor = match strukt.kind() {
179            StructKind::Record(_) => StructValueNsCtor::NoValueNsCtor,
180            StructKind::Unit => StructValueNsCtor::ValueNsCtorWithVis(visibility_id),
181            StructKind::Tuple(fields) => self.tuple_struct_ctor_vis(visibility, fields),
182        };
183        let res = Struct { name, visibility: visibility_id, value_ns_ctor };
184        Some(self.alloc_big(ast_id.upcast(), BigModItem::Struct(res)))
185    }

187    fn tuple_struct_ctor_vis(
188        &mut self,
189        struct_vis: RawVisibility,
190        fields: ast::TupleFieldList,
191    ) -> StructValueNsCtor {
192        if let RawVisibility::PubSelf(_) = struct_vis {
193            // `pub(self)` <= anything.
194            return StructValueNsCtor::ValueNsCtorWithVis(self.alloc_visibility(struct_vis));
195        }
196
197        let mut value_ns_ctor_vis = Either::Left(struct_vis);
198        for field in fields.fields() {
199            let field_vis = self.lower_visibility_from_ast(&field);
200            match (&field_vis, &mut value_ns_ctor_vis) {
201                (RawVisibility::PubSelf(_), _) => {
202                    // If there is a field private, the minimal visibility is private.
203                    value_ns_ctor_vis = Either::Left(field_vis);
204                    break;
205                }
206                (_, Either::Left(RawVisibility::Public)) => {
207                    // anything <= `pub`.
208                    value_ns_ctor_vis = Either::Left(field_vis);
209                }
210                (RawVisibility::Module(..), Either::Left(RawVisibility::PubCrate)) => {
211                    // `pub(in module)` <= `pub(crate)`.
212                    value_ns_ctor_vis = Either::Left(field_vis);
213                }
214                (
215                    RawVisibility::Public | RawVisibility::PubCrate,
216                    Either::Right(_)
217                    | Either::Left(RawVisibility::Module(..) | RawVisibility::PubCrate),
218                ) => {
219                    // `pub(in module) | pub(crate)` <= `pub(crate) | pub`.
220                }
221                (_, Either::Left(RawVisibility::PubSelf(_))) => {
222                    unreachable!("we early-exit on `PubSelf`");
223                }
224                (RawVisibility::Module(..), Either::Left(RawVisibility::Module(..))) => {
225                    let old_vis = match std::mem::replace(
226                        &mut value_ns_ctor_vis,
227                        Either::Left(RawVisibility::Public),
228                    ) {
229                        Either::Left(it) => it,
230                        _ => unreachable!(),
231                    };
232                    let mut multiple = FxHashSet::default();
233                    multiple.insert(old_vis);
234                    multiple.insert(field_vis);
235                    value_ns_ctor_vis = Either::Right(multiple);
236                }
237                (RawVisibility::Module(..), Either::Right(multiple)) => {
238                    multiple.insert(field_vis);
239                }
240            }
241        }
242
243        match value_ns_ctor_vis {
244            Either::Left(vis) => StructValueNsCtor::ValueNsCtorWithVis(self.alloc_visibility(vis)),
245            Either::Right(multiple) => {
246                if multiple.len() == 1 {
247                    StructValueNsCtor::ValueNsCtorWithVis(
248                        self.alloc_visibility(multiple.into_iter().next().unwrap()),
249                    )
250                } else {
251                    StructValueNsCtor::ValueNsCtorWithMinVis(
252                        multiple.into_iter().map(|vis| self.alloc_visibility(vis)).collect(),
253                    )
254                }
255            }
256        }
257    }

310    fn lower_module(&mut self, module: &ast::Module) -> Option<ModItemId> {
311        let name = module.name()?.as_name();
312        let visibility = self.lower_visibility(module);
313        let kind = if module.semicolon_token().is_some() {
314            ModKind::Outline
315        } else {
316            ModKind::Inline {
317                items: module
318                    .item_list()
319                    .map(|list| {
320                        list.items().filter_map(|item| self.lower_mod_item(&item)).collect()
321                    })
322                    .unwrap_or_else(|| {
323                        cov_mark::hit!(name_res_works_for_broken_modules);
324                        Box::new([]) as Box<[_]>
325                    }),
326            }
327        };
328        let ast_id = self.source_ast_id_map.ast_id(module);
329        let res = Mod { name, visibility, kind };
330        Some(self.alloc_big(ast_id.upcast(), BigModItem::Mod(res)))
331    }

404    fn lower_extern_block(&mut self, block: &ast::ExternBlock) -> ModItemId {
405        let ast_id = self.source_ast_id_map.ast_id(block);
406        let children: Box<[_]> = block.extern_item_list().map_or(Box::new([]), |list| {
407            list.extern_items()
408                .filter_map(|item| {
409                    // Note: All items in an `extern` block need to be lowered as if they're outside of one
410                    // (in other words, the knowledge that they're in an extern block must not be used).
411                    // This is because an extern block can contain macros whose ItemTree's top-level items
412                    // should be considered to be in an extern block too.
413                    let mod_item = match &item {
414                        ast::ExternItem::Fn(ast) => self.lower_function(ast)?,
415                        ast::ExternItem::Static(ast) => self.lower_static(ast)?,
416                        ast::ExternItem::TypeAlias(ty) => self.lower_type_alias(ty)?,
417                        ast::ExternItem::MacroCall(call) => self.lower_macro_call(call)?,
418                    };
419                    let attrs = self.lower_attrs(&item);
420                    self.add_attrs(mod_item, attrs);
421                    Some(mod_item)
422                })
423                .collect()
424        });
425
426        let res = ExternBlock { children };
427        self.alloc_small(ast_id.upcast(), SmallModItem::ExternBlock(res))
428    }

441    fn alloc_visibility(&mut self, vis: RawVisibility) -> RawVisibilityId {
442        match &vis {
443            RawVisibility::Public => RawVisibilityId::PUB,
444            RawVisibility::PubSelf(VisibilityExplicitness::Explicit) => {
445                RawVisibilityId::PRIV_EXPLICIT
446            }
447            RawVisibility::PubSelf(VisibilityExplicitness::Implicit) => {
448                RawVisibilityId::PRIV_IMPLICIT
449            }
450            RawVisibility::PubCrate => RawVisibilityId::PUB_CRATE,
451            _ => RawVisibilityId(self.visibilities.insert_full(vis).0 as u32),
452        }
453    }

456struct UseTreeLowering<'a> {
457    db: &'a dyn SourceDatabase,
458    mapping: Arena<ast::UseTree>,
459}
460
461impl UseTreeLowering<'_> {
462    fn lower_use_tree(
463        &mut self,
464        tree: ast::UseTree,
465        span_for_range: &mut dyn FnMut(::tt::TextRange) -> SyntaxContext,
466    ) -> Option<UseTree> {
467        if let Some(use_tree_list) = tree.use_tree_list() {
468            let prefix = match tree.path() {
469                // E.g. use something::{{{inner}}};
470                None => None,
471                // E.g. `use something::{inner}` (prefix is `None`, path is `something`)
472                // or `use something::{path::{inner::{innerer}}}` (prefix is `something::path`, path is `inner`)
473                Some(path) => {
474                    match ModPath::from_src(self.db, path, span_for_range) {
475                        Some(it) => Some(it),
476                        None => {
477                            // FIXME: report errors somewhere
478                            return None;
479                        }
480                    }
481                }
482            };
483
484            self.mapping.alloc(tree.clone());
485            let list = use_tree_list
486                .use_trees()
487                .filter_map(|tree| self.lower_use_tree(tree, span_for_range))
488                .collect();
489
490            Some(UseTree {
491                kind: UseTreeKind::Prefixed { prefix: prefix.map(Interned::new), list },
492            })
493        } else {
494            let is_glob = tree.star_token().is_some();
495            let path = match tree.path() {
496                Some(path) => Some(ModPath::from_src(self.db, path, span_for_range)?),
497                None => None,
498            };
499            let alias = tree.rename().map(|a| {
500                a.name().map(|it| it.as_name()).map_or(ImportAlias::Underscore, ImportAlias::Alias)
501            });
502            if alias.is_some() && is_glob {
503                return None;
504            }
505
506            match (path, alias, is_glob) {
507                (path, None, true) => {
508                    if path.is_none() {
509                        cov_mark::hit!(glob_enum_group);
510                    }
511                    self.mapping.alloc(tree.clone());
512                    Some(UseTree { kind: UseTreeKind::Glob { path: path.map(Interned::new) } })
513                }
514                // Globs can't be renamed
515                (_, Some(_), true) | (None, None, false) => None,
516                // `bla::{ as Name}` is invalid
517                (None, Some(_), false) => None,
518                (Some(path), alias, false) => {
519                    self.mapping.alloc(tree.clone());
520                    Some(UseTree { kind: UseTreeKind::Single { path: Interned::new(path), alias } })
521                }
522            }
523        }
524    }
525}

527pub(crate) fn lower_use_tree(
528    db: &dyn SourceDatabase,
529    tree: ast::UseTree,
530    span_for_range: &mut dyn FnMut(::tt::TextRange) -> SyntaxContext,
531) -> Option<(UseTree, Arena<ast::UseTree>)> {
532    let mut lowering = UseTreeLowering { db, mapping: Arena::new() };
533    let tree = lowering.lower_use_tree(tree, span_for_range)?;
534    Some((tree, lowering.mapping))
535}

541pub(crate) fn visibility_from_ast(
542    db: &dyn SourceDatabase,
543    node: Option<ast::Visibility>,
544    span_for_range: &mut dyn FnMut(::tt::TextRange) -> SyntaxContext,
545) -> RawVisibility {
546    let Some(node) = node else { return private_vis() };
547    let path = match node.kind() {
548        ast::VisibilityKind::In(path) => {
549            let path = ModPath::from_src(db, path, span_for_range);
550            match path {
551                None => return private_vis(),
552                Some(path) => path,
553            }
554        }
555        ast::VisibilityKind::PubCrate => return RawVisibility::PubCrate,
