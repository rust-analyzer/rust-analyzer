124#[derive(Debug, Default, PartialEq, Eq)]
125pub struct ItemScope {
126    /// Defs visible in this scope. This includes `declarations`, but also
127    /// imports. The imports belong to this module and can be resolved by using them on
128    /// the `use_imports_*` fields.
129    types: ThinFxHashMap<Name, TypesItem>,
130    values: ThinFxHashMap<Name, ValuesItem>,
131    macros: ThinFxHashMap<Name, MacrosItem>,
132    unresolved: ThinFxHashSet<Name>,
133
134    /// The defs declared in this scope. Each def has a single scope where it is
135    /// declared.
136    declarations: ThinVec<ModuleDefId>,
137
138    impls: ThinVec<(ImplId, /* trait impl */ bool)>,
139    builtin_derive_impls: ThinVec<BuiltinDeriveImplId>,
140    extern_blocks: ThinVec<ExternBlockId>,
141    unnamed_consts: ThinVec<ConstId>,
142    /// Traits imported via `use Trait as _;`.
143    unnamed_trait_imports: ThinVec<(TraitId, Item<()>)>,
144
145    // the resolutions of the imports of this scope
146    use_imports_types: ThinFxHashMap<ImportOrExternCrate, ImportOrDef>,
147    use_imports_values: ThinFxHashMap<ImportOrGlob, ImportOrDef>,
148    use_imports_macros: ThinFxHashMap<ImportOrExternCrate, ImportOrDef>,
149
150    use_decls: ThinVec<UseId>,
151    extern_crate_decls: ThinVec<ExternCrateId>,
152    /// Macros visible in current module in legacy textual scope
153    ///
154    /// For macros invoked by an unqualified identifier like `bar!()`, `legacy_macros` will be searched in first.
155    /// If it yields no result, then it turns to module scoped `macros`.
156    /// It macros with name qualified with a path like `crate::foo::bar!()`, `legacy_macros` will be skipped,
157    /// and only normal scoped `macros` will be searched in.
158    ///
159    /// Note that this automatically inherit macros defined textually before the definition of module itself.
160    ///
161    /// Module scoped macros will be inserted into `items` instead of here.
162    // FIXME: Macro shadowing in one module is not properly handled. Non-item place macros will
163    // be all resolved to the last one defined if shadowing happens.
164    legacy_macros: ThinFxHashMap<Name, SmallVec<[MacroId; 1]>>,
165    /// The attribute macro invocations in this scope.
166    attr_macros: ThinFxHashMap<AstId<ast::Item>, MacroCallId>,
167    /// The macro invocations in this scope.
168    macro_invocations: ThinFxHashMap<AstId<ast::MacroCall>, MacroCallId>,
169    /// The derive macro invocations in this scope, keyed by the owner item over the actual derive attributes
170    /// paired with the derive macro invocations for the specific attribute.
171    derive_macros: ThinFxHashMap<AstId<ast::Adt>, SmallVec<[DeriveMacroInvocation; 1]>>,
172}

239    pub fn fully_resolve_import(&self, db: &dyn SourceDatabase, mut import: ImportId) -> PerNs {
240        let mut res = PerNs::none();
241
242        let mut scope = self;
243        while let Some(&m) = scope.use_imports_macros.get(&ImportOrExternCrate::Import(import)) {
244            match m {
245                ImportOrDef::Import(i) => {
246                    let module_id = i.use_.lookup(db).container;
247                    scope = &module_id.def_map(db)[module_id].scope;
248                    import = i;
249                }
250                ImportOrDef::Def(ModuleDefId::MacroId(def)) => {
251                    res.macros = Some(Item { def, vis: Visibility::Public, import: None });
252                    break;
253                }
254                _ => break,
255            }
256        }
257        let mut scope = self;
258        while let Some(&m) = scope.use_imports_types.get(&ImportOrExternCrate::Import(import)) {
259            match m {
260                ImportOrDef::Import(i) => {
261                    let module_id = i.use_.lookup(db).container;
262                    scope = &module_id.def_map(db)[module_id].scope;
263                    import = i;
264                }
265                ImportOrDef::Def(def) => {
266                    res.types = Some(Item { def, vis: Visibility::Public, import: None });
267                    break;
268                }
269                _ => break,
270            }
271        }
272        let mut scope = self;
273        while let Some(&m) = scope.use_imports_values.get(&ImportOrGlob::Import(import)) {
274            match m {
275                ImportOrDef::Import(i) => {
276                    let module_id = i.use_.lookup(db).container;
277                    scope = &module_id.def_map(db)[module_id].scope;
278                    import = i;
279                }
280                ImportOrDef::Def(def) => {
281                    res.values = Some(Item { def, vis: Visibility::Public, import: None });
282                    break;
283                }
284                _ => break,
285            }
286        }
287        res
288    }

373            }),
374            ItemInNs::Values(def) => self.values.iter().find_map(|(name, other_def)| {
375                (other_def.def == def).then_some((name, other_def.vis, other_def.import.is_none()))
376            }),
377        }

382        &self,
383        item: ItemInNs,
384        mut cb: impl FnMut(&Name, Visibility, /*declared*/ bool) -> Option<T>,

386        match item {

395                    ))
396                })
397                .find_map(|(a, b, c)| cb(a, b, c)),
398            ItemInNs::Types(def) => self
399                .types
400                .iter()
401                .filter_map(|(name, other_def)| {
402                    (other_def.def == def).then_some((
403                        name,
404                        other_def.vis,
405                        other_def.import.is_none(),
406                    ))
407                })
408                .find_map(|(a, b, c)| cb(a, b, c)),
409            ItemInNs::Values(def) => self
410                .values
411                .iter()
412                .filter_map(|(name, other_def)| {
413                    (other_def.def == def).then_some((
414                        name,
415                        other_def.vis,
416                        other_def.import.is_none(),

418                })

585    pub(crate) fn push_res_with_import(
586        &mut self,

594        // FIXME: Document and simplify this
595
596        if let Some(mut fld) = def.types {
597            let existing = self.types.entry(lookup.1.clone());
598            match existing {
599                Entry::Vacant(entry) => {
600                    match import {
601                        Some(ImportOrExternCrate::Glob(_)) => {
602                            glob_imports.types.insert(lookup.clone());
603                        }
604                        _ => _ = glob_imports.types.remove(&lookup),
605                    }
606                    let prev = std::mem::replace(&mut fld.import, import);
607                    if let Some(import) = import {
608                        self.use_imports_types
609                            .insert(import, prev.map_or(ImportOrDef::Def(fld.def), Into::into));
610                    }
611                    entry.insert(fld);
612                    changed = true;
613                }
614                Entry::Occupied(mut entry) => {
615                    match import {
616                        Some(ImportOrExternCrate::Glob(..)) => {
617                            // Multiple globs may import the same item and they may
618                            // override visibility from previously resolved globs. This is
619                            // currently handled by `DefCollector`, because we need to
620                            // compute the max visibility for items and we need `DefMap`
621                            // for that.
622                        }
623                        _ => {
624                            // A non-glob import either shadows a glob import of the same
625                            // name, or re-resolves a stale binding it recorded earlier.
626                            if glob_imports.types.remove(&lookup)
627                                || entry.get().is_reresolved_by(&fld.def, import)
628                            {
629                                let prev = std::mem::replace(&mut fld.import, import);
630                                if let Some(import) = import {
631                                    self.use_imports_types.insert(
632                                        import,
633                                        prev.map_or(ImportOrDef::Def(fld.def), Into::into),
634                                    );
635                                }
636                                cov_mark::hit!(import_shadowed);
637                                entry.insert(fld);
638                                changed = true;
639                            }
640                        }
641                    }
642                }
643            }
644        }
645
646        if let Some(mut fld) = def.values {
647            let existing = self.values.entry(lookup.1.clone());
648            match existing {
649                Entry::Vacant(entry) => {
650                    match import {
651                        Some(ImportOrExternCrate::Glob(_)) => {
652                            glob_imports.values.insert(lookup.clone());
653                        }
654                        _ => _ = glob_imports.values.remove(&lookup),
655                    }
656                    let import = import.and_then(ImportOrExternCrate::import_or_glob);
657                    let prev = std::mem::replace(&mut fld.import, import);
658                    if let Some(import) = import {
659                        self.use_imports_values
660                            .insert(import, prev.map_or(ImportOrDef::Def(fld.def), Into::into));
661                    }
662                    entry.insert(fld);
663                    changed = true;
664                }
665                Entry::Occupied(mut entry)
666                    if !matches!(import, Some(ImportOrExternCrate::Glob(..))) =>
667                {
668                    let import = import.and_then(ImportOrExternCrate::import_or_glob);
669                    if glob_imports.values.remove(&lookup)
670                        || entry.get().is_reresolved_by(&fld.def, import)
671                    {
672                        cov_mark::hit!(import_shadowed);
673
674                        let prev = std::mem::replace(&mut fld.import, import);
675                        if let Some(import) = import {
676                            self.use_imports_values
677                                .insert(import, prev.map_or(ImportOrDef::Def(fld.def), Into::into));
678                        }
679                        entry.insert(fld);
680                        changed = true;
681                    }
682                }
683                _ => {}
684            }
685        }
686
687        if let Some(mut fld) = def.macros {
688            let existing = self.macros.entry(lookup.1.clone());
689            match existing {
690                Entry::Vacant(entry) => {
691                    match import {
692                        Some(ImportOrExternCrate::Glob(_)) => {
693                            glob_imports.macros.insert(lookup.clone());
694                        }
695                        _ => _ = glob_imports.macros.remove(&lookup),
696                    }
697                    let prev = std::mem::replace(&mut fld.import, import);
698                    if let Some(import) = import {
699                        self.use_imports_macros.insert(
700                            import,
701                            prev.map_or_else(|| ImportOrDef::Def(fld.def.into()), Into::into),
702                        );
703                    }
704                    entry.insert(fld);
705                    changed = true;
706                }
707                Entry::Occupied(mut entry)
708                    if !matches!(import, Some(ImportOrExternCrate::Glob(..)))
709                        && (glob_imports.macros.remove(&lookup)
710                            || entry.get().is_reresolved_by(&fld.def, import)) =>
711                {
712                    cov_mark::hit!(import_shadowed);
713                    let prev = std::mem::replace(&mut fld.import, import);
714                    if let Some(import) = import {
715                        self.use_imports_macros.insert(
716                            import,
717                            prev.map_or_else(|| ImportOrDef::Def(fld.def.into()), Into::into),
718                        );
719                    }
720                    entry.insert(fld);
721                    changed = true;
722                }
723                _ => {}
724            }
725        }
726
727        if def.is_none() && self.unresolved.insert(lookup.1) {
728            changed = true;
729        }
730
731        changed
732    }

734    /// Marks everything that is not a procedural macro as private to `this_module`.
735    pub(crate) fn censor_non_proc_macros(&mut self, krate: Crate) {
736        self.types
737            .values_mut()
738            .map(|def| &mut def.vis)
739            .chain(self.values.values_mut().map(|def| &mut def.vis))
740            .chain(self.unnamed_trait_imports.iter_mut().map(|(_, def)| &mut def.vis))
741            .for_each(|vis| *vis = Visibility::PubCrate(krate));
742
743        for mac in self.macros.values_mut() {
744            if matches!(mac.def, MacroId::ProcMacroId(_) if mac.import.is_none()) {
745                continue;
746            }
747            mac.vis = Visibility::PubCrate(krate)
748        }
749    }

751    pub(crate) fn dump(&self, db: &dyn SourceDatabase, buf: &mut String) {
752        let mut entries: Vec<_> = self.resolutions().collect();
753        entries.sort_by_key(|(name, _)| name.clone());
754
755        let print_macro_sub_ns = |buf: &mut String, macro_id: MacroId| {
756            let styles = crate::nameres::macro_styles_from_id(db, macro_id);
757            if styles.contains(MacroCallStyles::FN_LIKE) {
758                buf.push('!');
759            }
760            if styles.contains(MacroCallStyles::ATTR) || styles.contains(MacroCallStyles::DERIVE) {
761                buf.push('#');
762            }
763        };
764
765        for (name, def) in entries {
766            let display_name: &dyn fmt::Display = match &name {
767                Some(name) => &name.display(db, Edition::LATEST),
768                None => &"_",
769            };
770            format_to!(buf, "- {display_name} :");
771
772            if let Some(Item { import, .. }) = def.types {
773                buf.push_str(" type");
774                match import {
775                    Some(ImportOrExternCrate::Import(_)) => buf.push_str(" (import)"),
776                    Some(ImportOrExternCrate::Glob(_)) => buf.push_str(" (glob)"),
777                    Some(ImportOrExternCrate::ExternCrate(_)) => buf.push_str(" (extern)"),
778                    None => (),
779                }
780            }
781            if let Some(Item { import, .. }) = def.values {
782                buf.push_str(" value");
783                match import {
784                    Some(ImportOrGlob::Import(_)) => buf.push_str(" (import)"),
785                    Some(ImportOrGlob::Glob(_)) => buf.push_str(" (glob)"),
786                    None => (),
787                }
788            }
789            if let Some(Item { def: macro_id, import, .. }) = def.macros {
790                buf.push_str(" macro");
791                print_macro_sub_ns(buf, macro_id);
792                match import {
793                    Some(ImportOrExternCrate::Import(_)) => buf.push_str(" (import)"),
794                    Some(ImportOrExternCrate::Glob(_)) => buf.push_str(" (glob)"),
795                    Some(ImportOrExternCrate::ExternCrate(_)) => buf.push_str(" (extern)"),
796                    None => (),
797                }
798            }
799            if def.is_none() {
800                buf.push_str(" _");
801            }
802
803            buf.push('\n');
804        }
805
806        // Also dump legacy-textual-scope macros visible at the _end_ of the scope.
807        //
808        // For tests involving a cursor position, this might include macros that
809        // are _not_ visible at the cursor position.
810        let mut legacy_macros = self.legacy_macros().collect::<Vec<_>>();
811        legacy_macros.sort_by(|(a, _), (b, _)| Ord::cmp(a, b));
812        for (name, macros) in legacy_macros {
813            format_to!(buf, "- (legacy) {} :", name.display(db, Edition::LATEST));
814            for &macro_id in macros {
815                buf.push_str(" macro");
816                print_macro_sub_ns(buf, macro_id);
817            }
818            buf.push('\n');
819        }
820    }

824        let Self {

829            declarations,

831            builtin_derive_impls,

906impl PerNs {
907    pub(crate) fn from_def(
908        def: ModuleDefId,
909        vis: Visibility,
910        value_ns_ctor_vis: Option<Visibility>,
911        import: Option<ImportOrExternCrate>,
912    ) -> PerNs {
913        match def {
914            ModuleDefId::ModuleId(_) => PerNs::types(def, vis, import),
915            ModuleDefId::FunctionId(_) => {
916                PerNs::values(def, vis, import.and_then(ImportOrExternCrate::import_or_glob))
917            }
918            ModuleDefId::AdtId(adt) => match adt {
919                AdtId::UnionId(_) => PerNs::types(def, vis, import),
920                AdtId::EnumId(_) => PerNs::types(def, vis, import),
921                AdtId::StructId(_) => match value_ns_ctor_vis {
922                    Some(value_ns_ctor_vis) => PerNs {
923                        types: Some(Item { def, vis, import }),
924                        values: Some(Item {
925                            def,
926                            vis: value_ns_ctor_vis,
927                            import: import.and_then(ImportOrExternCrate::import_or_glob),
928                        }),
929                        macros: None,
930                    },
931                    None => PerNs::types(def, vis, import),
932                },
933            },
934            ModuleDefId::EnumVariantId(_) => match value_ns_ctor_vis {
935                Some(value_ns_ctor_vis) => PerNs {
936                    types: Some(Item { def, vis, import }),
937                    values: Some(Item {
938                        def,
939                        vis: value_ns_ctor_vis,
940                        import: import.and_then(ImportOrExternCrate::import_or_glob),
941                    }),
942                    macros: None,
943                },
944                None => PerNs::types(def, vis, import),
945            },
946            ModuleDefId::ConstId(_) | ModuleDefId::StaticId(_) => {
947                PerNs::values(def, vis, import.and_then(ImportOrExternCrate::import_or_glob))
948            }
949            ModuleDefId::TraitId(_) => PerNs::types(def, vis, import),
950            ModuleDefId::TypeAliasId(_) => PerNs::types(def, vis, import),
951            ModuleDefId::BuiltinType(_) => PerNs::types(def, vis, import),
952            ModuleDefId::MacroId(mac) => PerNs::macros(mac, vis, import),
953        }
954    }
955}

957#[derive(Clone, Copy, PartialEq, Eq, Debug, Hash)]
958pub enum ItemInNs {
959    Types(ModuleDefId),
960    Values(ModuleDefId),
961    Macros(MacroId),
962}
963
964impl ItemInNs {
965    pub fn as_module_def_id(self) -> Option<ModuleDefId> {
966        match self {
967            ItemInNs::Types(id) | ItemInNs::Values(id) => Some(id),
968            ItemInNs::Macros(_) => None,
969        }
970    }
971
972    /// Returns the crate defining this item (or `None` if `self` is built-in).
973    pub fn krate(&self, db: &dyn SourceDatabase) -> Option<Crate> {
974        self.module(db).map(|module_id| module_id.krate(db))
975    }
976
977    pub fn module(&self, db: &dyn SourceDatabase) -> Option<ModuleId> {
978        match self {
979            ItemInNs::Types(id) | ItemInNs::Values(id) => id.module(db),
980            ItemInNs::Macros(id) => Some(id.module(db)),
981        }
982    }
