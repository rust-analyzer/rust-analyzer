39// test or_pattern
40// fn main() {
41//     match () {
42//         (_ | _) => (),
43//         &(_ | _) => (),
44//         (_ | _,) => (),
45//         [_ | _,] => (),
46//     }
47// }
48/// Parses a pattern list separated by pipes `|`, with no leading `|`,using the
49/// given `recovery_set`.
50fn pattern_r(p: &mut Parser<'_>, recovery_set: TokenSet) {
51    let m = p.start();
52    let has_leading_pipe = p.eat(T![|]);
53
54    pattern_single_r(p, recovery_set);
55
56    if !p.at(T![|]) && !has_leading_pipe {
57        m.abandon(p);
58        return;
59    }
60    while p.eat(T![|]) {
61        pattern_single_r(p, recovery_set);
62    }
63    m.complete(p, OR_PAT);
64}

66fn pattern_single_r(p: &mut Parser<'_>, recovery_set: TokenSet) {
67    // test range_pat
68    // fn main() {
69    //     match () {
70    //         (..1) => (),
71    //         (..=3) => (),
72    //         (..2 | 4) => (),
73    //     }
74    //
75    //     match 92 {
76    //         0 ... 100 => (),
77    //         101 ..= 200 => (),
78    //         200 .. 301 => (),
79    //         302 .. => (),
80    //         ..= 303 => (),
81    //     }
82    //
83    //     match Some(10 as u8) {
84    //         Some(0) | None => (),
85    //         Some(1..) => (),
86    //         Some(..=2) => (),
87    //     }
88    //
89    //     match () {
90    //         S { a: 0 } => (),
91    //         S { a: 1.. } => (),
92    //         S { a: ..=2 } => (),
93    //     }
94    //
95    //     match () {
96    //         [0] => (),
97    //         [1..] => (),
98    //         [..=2] => (),
99    //     }
100    //
101    //     match (10 as u8, 5 as u8) {
102    //         (0, _) => (),
103    //         (1.., _) => (),
104    //         (..=2, _) => (),
105    //     }
106    //
107    // }
108
109    if p.at(T![..=]) {
110        let m = p.start();
111        p.bump(T![..=]);
112        atom_pat(p, recovery_set);
113        m.complete(p, RANGE_PAT);
114        return;
115    }
116
117    // test exclusive_range_pat
118    // fn main() {
119    //     match 42 {
120    //         ..0 => {}
121    //         1..2 => {}
122    //     }
123    // }
124
125    // test dot_dot_pat
126    // fn main() {
127    //     let .. = ();
128    //     //
129    //     // Tuples
130    //     //
131    //     let (a, ..) = ();
132    //     let (a, ..,) = ();
133    //     let Tuple(a, ..) = ();
134    //     let Tuple(a, ..,) = ();
135    //     let (.., ..) = ();
136    //     let Tuple(.., ..) = ();
137    //     let (.., a, ..) = ();
138    //     let Tuple(.., a, ..) = ();
139    //     //
140    //     // Slices
141    //     //
142    //     let [..] = ();
143    //     let [head, ..] = ();
144    //     let [head, tail @ ..] = ();
145    //     let [head, .., cons] = ();
146    //     let [head, mid @ .., cons] = ();
147    //     let [head, .., .., cons] = ();
148    //     let [head, .., mid, tail @ ..] = ();
149    //     let [head, .., mid, .., cons] = ();
150    // }
151    if p.at(T![..]) {
152        let m = p.start();
153        p.bump(T![..]);
154        if p.at_ts(RANGE_PAT_END_FIRST) {
155            atom_pat(p, recovery_set);
156            m.complete(p, RANGE_PAT);
157        } else {
158            m.complete(p, REST_PAT);
159        }
160        return;
161    }
162
163    if let Some(lhs) = atom_pat(p, recovery_set) {
164        for range_op in [T![...], T![..=], T![..]] {
165            if p.at(range_op) {
166                let m = lhs.precede(p);
167                p.bump(range_op);
168
169                // testing if we're at one of the following positions:
170                // `0 .. =>`
171                //       ^
172                // `let 0 .. =`
173                //           ^
174                // `let 0..: _ =`
175                //         ^
176                // (1.., _)
177                //     ^
178                // `Some(0 .. )`
179                //            ^
180                // `S { t: 0.. }`
181                //             ^
182                // `[0..]`
183                //      ^
184                // `0 .. if`
185                //       ^
186                // `0 .. | _`
187                //       ^
188                if matches!(
189                    p.current(),
190                    T![=] | T![,] | T![:] | T![')'] | T!['}'] | T![']'] | T![if] | T![|] | EOF
191                ) {
192                    // test half_open_range_pat
193                    // fn f() {
194                    //     let 0 .. = 1u32;
195                    //     let 0..: _ = 1u32;
196                    //
197                    //     match 42 {
198                    //         0 .. if true => (),
199                    //         _ => (),
200                    //     }
201                    //
202                    //     let &(0.. | _) = &0_i32;
203                    // }
204                } else {
205                    atom_pat(p, recovery_set);
206                }
207                m.complete(p, RANGE_PAT);
208                return;
209            }
210        }
211    }
212}

214fn builtin_pat(p: &mut Parser<'_>) -> Option<CompletedMarker> {
215    let m = p.start();
216    p.bump_remap(T![builtin]);
217    p.bump(T![#]);
218    if p.eat_contextual_kw(T![deref]) {
219        // test deref_pat
220        // fn foo() { let builtin # deref(_) = 1; }
221        p.expect(T!['(']);
222        pattern(p);
223        p.expect(T![')']);
224        Some(m.complete(p, DEREF_PAT))
225    } else if p.eat_contextual_kw(T![cfg]) {
226        // test cfg_pred_pat
227        // fn foo() { let builtin # cfg(any(true, x = "y")) = true; }
228        p.expect(T!['(']);
229        attributes::cfg_predicate(p);
230        p.eat(T![,]);
231        p.expect(T![')']);
232        Some(m.complete(p, CFG_PRED_PAT))
233    } else {
234        m.abandon(p);
235        None
236    }
237}

239const PAT_RECOVERY_SET: TokenSet = TokenSet::new(&[
240    T![let],
241    T![if],
242    T![while],
243    T![loop],
244    T![match],
245    T![')'],
246    T![']'],
247    T!['}'],
248    T![,],
249    T![=],
250    T![&],
251]);
252
253fn atom_pat(p: &mut Parser<'_>, recovery_set: TokenSet) -> Option<CompletedMarker> {
254    if p.at_contextual_kw(T![builtin]) && p.nth_at(1, T![#]) {
255        return builtin_pat(p);
256    }
257
258    let m = match p.current() {
259        T![box] => box_pat(p),
260        T![ref] | T![mut] => ident_pat(p, true),
261        T![const] => const_block_pat(p),
262        IDENT => match p.nth(1) {
263            // Checks the token after an IDENT to see if a pattern is a path (Struct { .. }) or macro
264            // (T![x]).
265            T!['('] | T!['{'] | T![!] => path_or_macro_pat(p),
266            T![:] if p.nth_at(1, T![::]) => path_or_macro_pat(p),
267            _ => ident_pat(p, true),
268        },
269
270        // test type_path_in_pattern
271        // fn main() { let <_>::Foo = (); }
272        _ if paths::is_path_start(p) => path_or_macro_pat(p),
273        _ if is_literal_pat_start(p) => literal_pat(p),
274
275        T![_] => wildcard_pat(p),
276        T![&] => ref_pat(p),
277        T!['('] => tuple_pat(p),
278        T!['['] => slice_pat(p),
279        T![!] => not_null_pat(p),
280
281        _ => {
282            p.err_recover("expected pattern", recovery_set);
283            return None;
284        }
285    };
286
287    Some(m)
288}

290fn is_literal_pat_start(p: &Parser<'_>) -> bool {
291    p.at(T![-]) && (p.nth(1) == INT_NUMBER || p.nth(1) == FLOAT_NUMBER)
292        || p.at_ts(expressions::LITERAL_FIRST)
293}

312// test path_part
313// fn foo() {
314//     let foo::Bar = ();
315//     let ::Bar = ();
316//     let Bar { .. } = ();
317//     let Bar(..) = ();
318// }
319fn path_or_macro_pat(p: &mut Parser<'_>) -> CompletedMarker {
320    assert!(paths::is_path_start(p));
321    let m = p.start();
322    paths::expr_path(p);
323    let kind = match p.current() {
324        T!['('] => {
325            tuple_pat_fields(p);
326            TUPLE_STRUCT_PAT
327        }
328        T!['{'] => {
329            record_pat_field_list(p);
330            RECORD_PAT
331        }
332        // test marco_pat
333        // fn main() {
334        //     let m!(x) = 0;
335        // }
336        T![!] => {
337            items::macro_call_after_excl(p);
338            return m.complete(p, MACRO_CALL).precede(p).complete(p, MACRO_PAT);
339        }
340        _ => PATH_PAT,
341    };
342    m.complete(p, kind)
343}

360// test record_pat_field
361// fn foo() {
362//     let S { 0: 1 } = ();
363//     let S { x: 1 } = ();
364//     let S { #[cfg(any())] x: 1 } = ();
365// }
366fn record_pat_field(p: &mut Parser<'_>) {
367    match p.current() {
368        IDENT | INT_NUMBER if p.nth(1) == T![:] => {
369            name_ref_or_index(p);
370            p.bump(T![:]);
371            // test record_field_pat_leading_or
372            // fn foo() { let R { a: | 1 | 2 } = 0; }
373            pattern(p);
374        }
375        // test_err record_pat_field_eq_recovery
376        // fn main() {
377        //     let S { field = foo };
378        // }
379        IDENT | INT_NUMBER if p.nth(1) == T![=] => {
380            name_ref_or_index(p);
381            p.err_and_bump("expected `:`");
382            pattern(p);
383        }
384        T![box] => {
385            // FIXME: not all box patterns should be allowed
386            box_pat(p);
387        }
388        T![ref] | T![mut] | IDENT => {
389            ident_pat(p, false);
390        }
391        _ => {
392            p.err_and_bump("expected identifier");
393        }
394    }
395}

397// test record_pat_field_list
398// fn foo() {
399//     let S {} = ();
400//     let S { f, ref mut g } = ();
401//     let S { h: _, ..} = ();
402//     let S { h: _, } = ();
403//     let S { #[cfg(any())] .. } = ();
404// }
405fn record_pat_field_list(p: &mut Parser<'_>) {
406    assert!(p.at(T!['{']));
407    let m = p.start();
408    p.bump(T!['{']);
409    while !p.at(EOF) && !p.at(T!['}']) {
410        let m = p.start();
411        attributes::outer_attrs(p);
412
413        match p.current() {
414            // A trailing `..` is *not* treated as a REST_PAT.
415            T![.] if p.at(T![..]) => {
416                p.bump(T![..]);
417                m.complete(p, REST_PAT);
418            }
419            T!['{'] => {
420                error_block(p, "expected ident");
421                m.abandon(p);
422            }
423            _ => {
424                record_pat_field(p);
425                m.complete(p, RECORD_PAT_FIELD);
426            }
427        }
428        if !p.at(T!['}']) {
429            p.expect(T![,]);
430        }
431    }
432    p.expect(T!['}']);
433    m.complete(p, RECORD_PAT_FIELD_LIST);
434}

471// test tuple_pat
472// fn main() {
473//     let (a, b, ..) = ();
474//     let (a,) = ();
475//     let (..) = ();
476//     let () = ();
477//     let (| a | a, | b) = ((),());
478// }
479fn tuple_pat(p: &mut Parser<'_>) -> CompletedMarker {
480    assert!(p.at(T!['(']));
481    let m = p.start();
482    p.bump(T!['(']);
483    let mut has_comma = false;
484    let mut has_pat = false;
485    let mut has_rest = false;
486
487    // test_err tuple_pat_leading_comma
488    // fn foo() {
489    //     let (,);
490    // }
491    if p.eat(T![,]) {
492        p.error("expected pattern");
493        has_comma = true;
494    }
495
496    while !p.at(EOF) && !p.at(T![')']) {
497        has_pat = true;
498        if !p.at_ts(PAT_TOP_FIRST) {
499            p.error("expected a pattern");
500            break;
501        }
502        has_rest |= !p.at(T![..=]) && p.at(T![..]) && !RANGE_PAT_END_FIRST.contains(p.nth(2));
503        pattern(p);
504        if !p.at(T![')']) {
505            has_comma = true;
506            p.expect(T![,]);
507        }
508    }
509    p.expect(T![')']);
510
511    m.complete(p, if !has_comma && !has_rest && has_pat { PAREN_PAT } else { TUPLE_PAT })
512}

528fn pat_list(p: &mut Parser<'_>, ket: SyntaxKind) {
529    while !p.at(EOF) && !p.at(ket) {
530        pattern(p);
531        if !p.eat(T![,]) {
532            if p.at_ts(PAT_TOP_FIRST) {
533                p.error(format!("expected {:?}, got {:?}", T![,], p.current()));
534            } else {
535                break;
536            }
537        }
538    }
539}

543//     let a = ();
544//     let mut b = ();
545//     let ref c = ();

547//     let e @ _ = ();
548//     let ref mut f @ g @ _ = ();

550fn ident_pat(p: &mut Parser<'_>, with_at: bool) -> CompletedMarker {
551    assert!(matches!(p.current(), T![ref] | T![mut] | IDENT));
552    let m = p.start();
553    p.eat(T![ref]);
554    p.eat(T![mut]);
555    name_r(p, PAT_RECOVERY_SET);
556    if with_at && p.eat(T![@]) {
557        pattern_single(p);
558    }
559    m.complete(p, IDENT_PAT)
560}
