373    #[returns(copy)]

377impl<'db> InternedClosureId<'db> {
378    #[inline]
379    pub fn new(db: &'db dyn HirDatabase, loc: InternedClosure<'db>) -> Self {
380        if cfg!(debug_assertions) {
381            let store = ExpressionStore::of(db, loc.owner.expression_store_owner(db));
382            let expr = &store[loc.expr];
383            assert!(
384                matches!(
385                    expr,
386                    hir_def::hir::Expr::Closure {
387                        closure_kind: hir_def::hir::ClosureKind::Closure,
388                        ..
389                    }
390                ),
391                "expected a closure, found {expr:?}"
392            );
393        }
394
395        Self::new_impl(db, loc)
396    }
397}

458/// An anonymous const expression that appears in a type position (e.g., array lengths,
459/// const generic arguments like `{ N + 1 }`, or const param defaults). Unlike named constants,
460/// these don't have their own `Body` — their expressions live in the parent's signature `ExpressionStore`.
461#[derive(Debug, Hash, PartialEq, Eq, Clone)]
462pub struct AnonConstLoc {
463    /// The owner store containing this expression.
464    pub owner: ExpressionStoreOwnerId,
465    /// The ExprId within the owner's ExpressionStore that is the root
466    /// of this anonymous const expression.
467    pub expr: ExprId,
468    pub ty: StoredEarlyBinder<StoredTy>,
469    /// Whether to allow using generic params from the owner.
470    /// true for array repeats, false for everything else.
471    pub(crate) allow_using_generic_params: bool,
472}

504pub fn signature_anon_consts_and_diagnostics<'db>(
505    db: &'db dyn HirDatabase,
506    def: GenericDefId,
507) -> ArrayVec<(&'db [AnonConstId<'db>], &'db [TyLoweringDiagnostic]), 5> {
508    let mut result = ArrayVec::new();
509
510    // Queries common to all generic defs:
511    push(&mut result, db.generic_defaults_with_diagnostics(def));
512    push(&mut result, GenericPredicates::query_with_diagnostics(db, def));
513    push(&mut result, db.const_param_types_with_diagnostics(def));
514
515    match def {
516        GenericDefId::ImplId(id) => {
517            push(&mut result, db.impl_self_ty_with_diagnostics(id));
518            if let Some(trait_ref) = db.impl_trait_with_diagnostics(id) {
519                push(&mut result, trait_ref);
520            }
521        }
522        GenericDefId::TypeAliasId(id) => {
523            push(&mut result, db.type_for_type_alias_with_diagnostics(id));
524            push(&mut result, db.type_alias_bounds_with_diagnostics(id));
525        }
526        GenericDefId::FunctionId(id) => push(&mut result, db.fn_sig_for_fn_with_diagnostics(id)),
527        GenericDefId::ConstId(def) => push(&mut result, db.type_for_const_with_diagnostics(def)),
528        GenericDefId::StaticId(def) => push(&mut result, db.type_for_static_with_diagnostics(def)),
529        GenericDefId::TraitId(_) | GenericDefId::AdtId(_) => {}
530    }
531
532    return result;
533
534    fn push<'db, T>(
535        result: &mut ArrayVec<(&'db [AnonConstId<'db>], &'db [TyLoweringDiagnostic]), 5>,
536        item: &'db TyLoweringResult<'db, T>,
537    ) {
538        result.push((item.defined_anon_consts(), item.diagnostics()));
539    }
540}

554/// A constant, which might appears as a const item, an anonymous const block in expressions
555/// or patterns, or as a constant in types with const generics.
556#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, salsa::Supertype)]
557pub enum GeneralConstId<'db> {
558    ConstId(ConstId),
559    StaticId(StaticId),
560    AnonConstId(AnonConstId<'db>),
561}
562
563impl_from!(impl<'db> ConstId, StaticId, AnonConstId<'db> for GeneralConstId<'db>);
564
565impl<'db> GeneralConstId<'db> {
566    pub fn generic_def(self, db: &'db dyn HirDatabase) -> Option<GenericDefId> {
567        match self {
568            GeneralConstId::ConstId(it) => Some(it.into()),
569            GeneralConstId::StaticId(it) => Some(it.into()),
570            GeneralConstId::AnonConstId(it) => Some(it.loc(db).owner.generic_def(db)),
571        }
572    }
573
574    pub fn name(self, db: &'db dyn SourceDatabase) -> String {
575        match self {
576            GeneralConstId::StaticId(it) => {
577                StaticSignature::of(db, it).name.display(db, Edition::CURRENT).to_string()
578            }
579            GeneralConstId::ConstId(const_id) => {
580                ConstSignature::of(db, const_id).name.as_ref().map_or_else(
581                    || "_".to_owned(),
582                    |name| name.display(db, Edition::CURRENT).to_string(),
583                )
584            }
585            GeneralConstId::AnonConstId(_) => "{const}".to_owned(),
586        }
587    }
