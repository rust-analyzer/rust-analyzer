131    fn add_item_bounds_for_hidden_type(
132        &self,
133        opaque_id: OpaqueTyIdWrapper<'_>,
134        args: GenericArgs<'db>,
135        param_env: ParamEnv<'db>,
136        hidden_ty: Ty<'db>,
137        goals: &mut Vec<Goal<'db, Predicate<'db>>>,
138    ) {
139        let interner = self.interner;
140        // Require that the hidden type is well-formed. We have to
141        // make sure we wf-check the hidden type to fix #114728.
142        //
143        // However, we don't check that all types are well-formed.
144        // We only do so for types provided by the user or if they are
145        // "used", e.g. for method selection.
146        //
147        // This means we never check the wf requirements of the hidden
148        // type during MIR borrowck, causing us to infer the wrong
149        // lifetime for its member constraints which then results in
150        // unexpected region errors.
151        goals.push(Goal::new(interner, param_env, ClauseKind::WellFormed(hidden_ty.into())));
152
153        let replace_opaques_in = |clause: Clause<'db>| {
154            fold_tys(interner, clause, |ty| match ty.kind() {
155                // Replace all other mentions of the same opaque type with the hidden type,
156                // as the bounds must hold on the hidden type after all.
157                TyKind::Alias(AliasTy {
158                    kind: AliasTyKind::Opaque { def_id: def_id2 },
159                    args: args2,
160                    ..
161                }) if opaque_id == def_id2 && args == args2 => hidden_ty,
162                _ => ty,
163            })
164        };
165
166        let item_bounds = opaque_id.0.predicates(interner.db);
167        for predicate in item_bounds.iter_instantiated_copied(interner, args.as_slice()) {
168            let predicate = replace_opaques_in(predicate.skip_norm_wip());
169
170            // Require that the predicate holds for the concrete type.
171            debug!(?predicate);
172            goals.push(Goal::new(interner, param_env, predicate));
173        }
174    }

176    fn fetch_eligible_assoc_item(
177        &self,
178        _goal_trait_ref: rustc_type_ir::TraitRef<Self::Interner>,
179        trait_assoc_def_id: TraitAssocTermId,
180        impl_id: AnyImplId,
181    ) -> FetchEligibleAssocItemResponse<Self::Interner> {
182        let AnyImplId::ImplId(impl_id) = impl_id else {
183            // Builtin derive traits don't have type/consts assoc items.
184            return FetchEligibleAssocItemResponse::Err(ErrorGuaranteed);
185        };
186        let impl_items = impl_id.impl_items(self.0.interner.db());
187        let id = match trait_assoc_def_id.0 {
188            TermId::TypeAliasId(trait_assoc_id) => {
189                let trait_assoc_data = TypeAliasSignature::of(self.0.interner.db, trait_assoc_id);
190                impl_items
191                    .items
192                    .iter()
193                    .find_map(|(impl_assoc_name, impl_assoc_id)| {
194                        if let AssocItemId::TypeAliasId(impl_assoc_id) = *impl_assoc_id
195                            && *impl_assoc_name == trait_assoc_data.name
196                        {
197                            Some(impl_assoc_id)
198                        } else {
199                            None
200                        }
201                    })
202                    .or_else(|| {
203                        if trait_assoc_data.ty.is_some() { Some(trait_assoc_id) } else { None }
204                    })
205                    .map(|def| ImplOrTraitAssocTermId(TermId::TypeAliasId(def)))
206            }
207            TermId::ConstId(trait_assoc_id) => {
208                let trait_assoc_data = ConstSignature::of(self.0.interner.db, trait_assoc_id);
209                let trait_assoc_name = trait_assoc_data
210                    .name
211                    .as_ref()
212                    .expect("unnamed consts should not get passed to the solver");
213                impl_items
214                    .items
215                    .iter()
216                    .find_map(|(impl_assoc_name, impl_assoc_id)| {
217                        if let AssocItemId::ConstId(impl_assoc_id) = *impl_assoc_id
218                            && impl_assoc_name == trait_assoc_name
219                        {
220                            Some(impl_assoc_id)
221                        } else {
222                            None
223                        }
224                    })
225                    .or_else(
226                        || {
227                            if trait_assoc_data.has_body() { Some(trait_assoc_id) } else { None }
228                        },
229                    )
230                    .map(|def| ImplOrTraitAssocTermId(TermId::ConstId(def)))
231            }
232        };
233        match id {
234            Some(id) => FetchEligibleAssocItemResponse::Found(id),
235            None => match self.typing_mode_raw() {
236                TypingMode::ErasedNotCoherence(_) => {
237                    FetchEligibleAssocItemResponse::NotFoundBecauseErased
238                }
239                typing_mode => {
240                    FetchEligibleAssocItemResponse::NotFound(typing_mode.assert_not_erased())
241                }
242            },
243        }
244    }

257    fn evaluate_const(
258        &self,
259        param_env: ParamEnv<'db>,
260        uv: UnevaluatedConst<'db>,
261    ) -> Option<Const<'db>> {
262        let ec = match uv.def.0 {
263            GeneralConstId::ConstId(c) => {
264                let subst = uv.args;
265                self.cx().db.const_eval(c, subst, None)
266            }
267            GeneralConstId::StaticId(c) => self.cx().db.const_eval_static(c),
268            GeneralConstId::AnonConstId(c) => {
269                let subst = uv.args;
270                self.cx().db.anon_const_eval(c, subst, None)
271            }
272        };
273        let konst = match ec {
274            Ok(ec) => Const::new_from_allocation(
275                self.interner,
276                &ec,
277                ParamEnvAndCrate { param_env, krate: self.interner.expect_crate() },
278            ),
279            // FIXME: Not all evaluation errors should return an error const, rustc returns None on `EvaluateConstErr::{HasGenericsOrInfers, InvalidConstParamTy}`.
280            Err(_) => Const::new_error(self.interner, ErrorGuaranteed),
281        };
282        Some(konst)
283    }

285    fn compute_goal_fast_path(
286        &self,
287        goal: rustc_type_ir::solve::Goal<
288            Self::Interner,
289            <Self::Interner as rustc_type_ir::Interner>::Predicate,
290        >,
291        _span: <Self::Interner as rustc_type_ir::Interner>::Span,
292    ) -> Option<Certainty> {
293        if let Some(trait_pred) = goal.predicate.as_trait_clause() {
294            if self.shallow_resolve(trait_pred.self_ty().skip_binder()).is_ty_var()
295                // We don't do this fast path when opaques are defined since we may
296                // eventually use opaques to incompletely guide inference via ty var
297                // self types.
298                // FIXME: Properly consider opaques here.
299                && self.inner.borrow_mut().opaque_types().is_empty()
300            {
301                return Some(Certainty::AMBIGUOUS);
302            }
303
304            if trait_pred.polarity() == PredicatePolarity::Positive {
305                match self.0.interner.as_trait_lang_item(trait_pred.def_id()) {
306                    Some(SolverTraitLangItem::Sized) | Some(SolverTraitLangItem::MetaSized) => {
307                        let predicate = self.resolve_vars_if_possible(goal.predicate);
308                        if sizedness_fast_path(self.interner, predicate, goal.param_env) {
309                            return Some(Certainty::Yes);
310                        }
311                    }
312                    Some(SolverTraitLangItem::Copy | SolverTraitLangItem::Clone) => {
313                        let self_ty =
314                            self.resolve_vars_if_possible(trait_pred.self_ty().skip_binder());
315                        // Unlike `Sized` traits, which always prefer the built-in impl,
316                        // `Copy`/`Clone` may be shadowed by a param-env candidate which
317                        // could force a lifetime error or guide inference. While that's
318                        // not generally desirable, it is observable, so for now let's
319                        // ignore this fast path for types that have regions or infer.
320                        if !self_ty
321                            .has_type_flags(TypeFlags::HAS_FREE_REGIONS | TypeFlags::HAS_INFER)
322                            && self_ty.is_trivially_pure_clone_copy()
323                        {
324                            return Some(Certainty::Yes);
325                        }
326                    }
327                    _ => {}
328                }
329            }
330        }
331
332        let pred = goal.predicate.kind();
333        match pred.no_bound_vars()? {
334            PredicateKind::DynCompatible(def_id)
335                if self.0.interner.trait_is_dyn_compatible(def_id) =>
336            {
337                Some(Certainty::Yes)
338            }
339            PredicateKind::Clause(ClauseKind::RegionOutlives(outlives)) => {
340                self.0.sub_regions(outlives.1, outlives.0);
341                Some(Certainty::Yes)
342            }
343            PredicateKind::Clause(ClauseKind::TypeOutlives(outlives)) => {
344                self.0.register_type_outlives_constraint(outlives.0, outlives.1);
345
346                Some(Certainty::Yes)
347            }
348            PredicateKind::Subtype(SubtypePredicate { a, b, .. })
349            | PredicateKind::Coerce(CoercePredicate { a, b }) => {
350                match (self.shallow_resolve(a).kind(), self.shallow_resolve(b).kind()) {
351                    (
352                        TyKind::Infer(InferTy::TyVar(a_vid)),
353                        TyKind::Infer(InferTy::TyVar(b_vid)),
354                    ) => {
355                        self.sub_unify_ty_vids_raw(a_vid, b_vid);
356                        Some(Certainty::AMBIGUOUS)
357                    }
358                    _ => None,
359                }
360            }
361            PredicateKind::Clause(ClauseKind::ConstArgHasType(ct, _)) => {
362                if self.shallow_resolve_const(ct).is_ct_infer() {
363                    Some(Certainty::AMBIGUOUS)
364                } else {
365                    None
366                }
367            }
368            PredicateKind::Clause(ClauseKind::WellFormed(arg)) => {
369                let arg = self.shallow_resolve_term(arg);
370                if arg.is_trivially_wf(self.interner) {
371                    Some(Certainty::Yes)
372                } else if arg.is_infer() {
373                    Some(Certainty::AMBIGUOUS)
374                } else {
375                    None
376                }
377            }
378            _ => None,
379        }
380    }
