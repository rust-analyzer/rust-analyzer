144#[inline]
145pub fn default_types<'db>() -> &'db DefaultAny<'db> {
146    static TYPES: OnceLock<DefaultAny<'static>> = OnceLock::new();
147
148    TYPES.get_or_init(|| {
149        let create_ty = |kind| {
150            let ty = Ty::new_without_interner(kind);
151            // We need to increase the refcount (forever), so that the types won't be freed.
152            let ty = ManuallyDrop::new(ty.store());
153            ty.as_ref()
154        };
155        let create_const = |kind| {
156            let ty = Const::new_without_interner(kind);
157            // We need to increase the refcount (forever), so that the types won't be freed.
158            let ty = ManuallyDrop::new(ty.store());
159            ty.as_ref()
160        };
161        let create_region = |kind| {
162            let ty = Region::new_without_interner(kind);
163            // We need to increase the refcount (forever), so that the types won't be freed.
164            let ty = ManuallyDrop::new(ty.store());
165            ty.as_ref()
166        };
167        let create_generic_args = |slice| {
168            let ty = GenericArgs::new_from_slice_no_empty_check(slice);
169            // We need to increase the refcount (forever), so that the types won't be freed.
170            let ty = ManuallyDrop::new(ty.store());
171            ty.as_ref()
172        };
173        let create_bound_var_kinds = |slice| {
174            let ty = BoundVarKinds::new_from_slice_no_empty_check(slice);
175            // We need to increase the refcount (forever), so that the types won't be freed.
176            let ty = ManuallyDrop::new(ty.store());
177            ty.as_ref()
178        };
179        let create_canonical_vars = |slice| {
180            let ty = CanonicalVarKinds::new_from_slice_no_empty_check(slice);
181            // We need to increase the refcount (forever), so that the types won't be freed.
182            let ty = ManuallyDrop::new(ty.store());
183            ty.as_ref()
184        };
185        let create_variances_of = |slice| {
186            let ty = VariancesOf::new_from_slice_no_empty_check(slice);
187            // We need to increase the refcount (forever), so that the types won't be freed.
188            let ty = ManuallyDrop::new(ty.store());
189            ty.as_ref()
190        };
191        let create_pat_list = |slice| {
192            let ty = PatList::new_from_slice_no_empty_check(slice);
193            // We need to increase the refcount (forever), so that the types won't be freed.
194            let ty = ManuallyDrop::new(ty.store());
195            ty.as_ref()
196        };
197        let create_predefined_opaques = |slice| {
198            let ty = PredefinedOpaques::new_from_slice_no_empty_check(slice);
199            // We need to increase the refcount (forever), so that the types won't be freed.
200            let ty = ManuallyDrop::new(ty.store());
201            ty.as_ref()
202        };
203        let create_solver_def_ids = |slice| {
204            let ty = SolverDefIds::new_from_slice_no_empty_check(slice);
205            // We need to increase the refcount (forever), so that the types won't be freed.
206            let ty = ManuallyDrop::new(ty.store());
207            ty.as_ref()
208        };
209        let create_bound_existential_predicates = |slice| {
210            let ty = BoundExistentialPredicates::new_from_slice_no_empty_check(slice);
211            // We need to increase the refcount (forever), so that the types won't be freed.
212            let ty = ManuallyDrop::new(ty.store());
213            ty.as_ref()
214        };
215        let create_clauses = |slice| {
216            let ty = Clauses::new_from_slice_no_empty_check(slice);
217            // We need to increase the refcount (forever), so that the types won't be freed.
218            let ty = ManuallyDrop::new(ty.store());
219            ty.as_ref()
220        };
221        let create_region_assumptions = |slice| {
222            let ty = RegionAssumptions::new_from_slice_no_empty_check(slice);
223            // We need to increase the refcount (forever), so that the types won't be freed.
224            let ty = ManuallyDrop::new(ty.store());
225            ty.as_ref()
226        };
227        let create_tys = |slice| {
228            let ty = Tys::new_from_slice_no_empty_check(slice);
229            // We need to increase the refcount (forever), so that the types won't be freed.
230            let ty = ManuallyDrop::new(ty.store());
231            ty.as_ref()
232        };
233        let create_consts = |slice| {
234            let ty = Consts::new_from_slice_no_empty_check(slice);
235            // We need to increase the refcount (forever), so that the types won't be freed.
236            let ty = ManuallyDrop::new(ty.store());
237            ty.as_ref()
238        };
239        let create_projection = |slice| {
240            let it = crate::mir::Projection::new_from_slice_no_empty_check(slice);
241            // We need to increase the refcount (forever), so that the types won't be freed.
242            let it = ManuallyDrop::new(it.store());
243            it.as_ref()
244        };
245
246        let str = create_ty(TyKind::Str);
247        let statik = create_region(RegionKind::ReStatic);
248        let empty_tys = create_tys(&[]);
249        let unit = create_ty(TyKind::Tuple(empty_tys));
250        let u8 = create_ty(TyKind::Uint(rustc_ast_ir::UintTy::U8));
251        let u8_slice = create_ty(TyKind::Slice(u8));
252        let static_u8_slice = create_ty(TyKind::Ref(statik, u8_slice, Mutability::Not));
253        DefaultAny {
254            types: DefaultTypes {
255                usize: create_ty(TyKind::Uint(rustc_ast_ir::UintTy::Usize)),
256                u8,
257                u16: create_ty(TyKind::Uint(rustc_ast_ir::UintTy::U16)),
258                u32: create_ty(TyKind::Uint(rustc_ast_ir::UintTy::U32)),
259                u64: create_ty(TyKind::Uint(rustc_ast_ir::UintTy::U64)),
260                u128: create_ty(TyKind::Uint(rustc_ast_ir::UintTy::U128)),
261                isize: create_ty(TyKind::Int(rustc_ast_ir::IntTy::Isize)),
262                i8: create_ty(TyKind::Int(rustc_ast_ir::IntTy::I8)),
263                i16: create_ty(TyKind::Int(rustc_ast_ir::IntTy::I16)),
264                i32: create_ty(TyKind::Int(rustc_ast_ir::IntTy::I32)),
265                i64: create_ty(TyKind::Int(rustc_ast_ir::IntTy::I64)),
266                i128: create_ty(TyKind::Int(rustc_ast_ir::IntTy::I128)),
267                f16: create_ty(TyKind::Float(rustc_ast_ir::FloatTy::F16)),
268                f32: create_ty(TyKind::Float(rustc_ast_ir::FloatTy::F32)),
269                f64: create_ty(TyKind::Float(rustc_ast_ir::FloatTy::F64)),
270                f128: create_ty(TyKind::Float(rustc_ast_ir::FloatTy::F128)),
271                unit,
272                bool: create_ty(TyKind::Bool),
273                char: create_ty(TyKind::Char),
274                str,
275                never: create_ty(TyKind::Never),
276                error: create_ty(TyKind::Error(ErrorGuaranteed)),
277                static_str_ref: create_ty(TyKind::Ref(statik, str, rustc_ast_ir::Mutability::Not)),
278                u8_slice,
279                static_u8_slice,
280                mut_unit_ptr: create_ty(TyKind::RawPtr(unit, rustc_ast_ir::Mutability::Mut)),
281                // This type must not appear anywhere except here.
282                dyn_trait_dummy_self: create_ty(TyKind::Infer(rustc_type_ir::InferTy::FreshTy(0))),
283            },
284            consts: DefaultConsts {
285                error: create_const(ConstKind::Error(ErrorGuaranteed)),
286                u8_values: std::array::from_fn(|u8_value| {
287                    create_const(ConstKind::Value(ValueConst {
288                        ty: u8,
289                        value: ValTree::new(ValTreeKind::Leaf(ScalarInt::from(u8_value as u8))),
290                    }))
291                }),
292            },
293            regions: DefaultRegions {
294                error: create_region(RegionKind::ReError(ErrorGuaranteed)),
295                statik,
296                erased: create_region(RegionKind::ReErased),
297            },
298            empty: DefaultEmpty {
299                tys: empty_tys,
300                generic_args: create_generic_args(&[]),
301                bound_var_kinds: create_bound_var_kinds(&[]),
302                canonical_vars: create_canonical_vars(&[]),
303                variances: create_variances_of(&[]),
304                pat_list: create_pat_list(&[]),
305                predefined_opaques: create_predefined_opaques(&[]),
306                def_ids: create_solver_def_ids(&[]),
307                bound_existential_predicates: create_bound_existential_predicates(&[]),
308                clauses: create_clauses(&[]),
309                region_assumptions: create_region_assumptions(&[]),
310                consts: create_consts(&[]),
311                projection: create_projection(&[]),
312            },
313            one_invariant: create_variances_of(&[rustc_type_ir::Variance::Invariant]),
314            one_covariant: create_variances_of(&[rustc_type_ir::Variance::Covariant]),
315            coroutine_captures_by_ref_bound_var_kinds: create_bound_var_kinds(&[
316                BoundVariableKind::Region(BoundRegionKind::ClosureEnv),
317            ]),
318        }
319    })
