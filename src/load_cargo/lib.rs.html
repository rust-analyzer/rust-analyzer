55pub fn load_workspace_at(
56    root: &Path,
57    cargo_config: &CargoConfig,
58    load_config: &LoadCargoConfig,
59    progress: &(dyn Fn(String) + Sync),
60) -> anyhow::Result<(RootDatabase, vfs::Vfs, Option<ProcMacroClient>)> {
61    let root = AbsPathBuf::assert_utf8(std::env::current_dir()?.join(root));
62    let root = ProjectManifest::discover_single(&root)?;
63    let manifest_path = root.manifest_path().clone();
64    let mut workspace = ProjectWorkspace::load(root, cargo_config, progress)?;
65
66    if load_config.load_out_dirs_from_check {
67        let build_scripts = workspace.run_build_scripts(cargo_config, progress)?;
68        if let Some(error) = build_scripts.error() {
69            tracing::debug!(
70                "Errors occurred while running build scripts for {}: {}",
71                manifest_path,
72                error
73            );
74        }
75        workspace.set_build_scripts(build_scripts)
76    }
77
78    load_workspace(workspace, &cargo_config.extra_env, load_config)
79}

81pub fn load_workspace(
82    ws: ProjectWorkspace,
83    extra_env: &FxHashMap<String, Option<String>>,
84    load_config: &LoadCargoConfig,
85) -> anyhow::Result<(RootDatabase, vfs::Vfs, Option<ProcMacroClient>)> {
86    let lru_cap = std::env::var("RA_LRU_CAP").ok().and_then(|it| it.parse::<u16>().ok());
87    let mut db = RootDatabase::new(lru_cap);
88
89    let (vfs, proc_macro_server) = load_workspace_into_db(ws, extra_env, load_config, &mut db)?;
90
91    Ok((db, vfs, proc_macro_server))
92}

94// This variant of `load_workspace` allows deferring the loading of rust-analyzer
95// into an existing database, which is useful in certain third-party scenarios,
96// now that `salsa` supports extending foreign databases (e.g. `RootDatabase`).
97pub fn load_workspace_into_db(
98    ws: ProjectWorkspace,
99    extra_env: &FxHashMap<String, Option<String>>,
100    load_config: &LoadCargoConfig,
101    db: &mut RootDatabase,
102) -> anyhow::Result<(vfs::Vfs, Option<ProcMacroClient>)> {
103    let (sender, receiver) = unbounded();
104    let mut vfs = vfs::Vfs::default();
105    let mut loader = {
106        let loader = vfs_notify::NotifyHandle::spawn(sender);
107        Box::new(loader)
108    };
109
110    tracing::debug!(?load_config, "LoadCargoConfig");
111    let proc_macro_server = match &load_config.with_proc_macro_server {
112        ProcMacroServerChoice::Sysroot => ws.find_sysroot_proc_macro_srv().map(|it| {
113            it.and_then(|it| {
114                ProcMacroClient::spawn(
115                    &it,
116                    extra_env,
117                    ws.toolchain.as_ref(),
118                    load_config.proc_macro_processes,
119                )
120                .map_err(Into::into)
121            })
122            .map_err(|e| ProcMacroLoadingError::ProcMacroSrvError(e.to_string().into_boxed_str()))
123        }),
124        ProcMacroServerChoice::Explicit(path) => Some(
125            ProcMacroClient::spawn(
126                path,
127                extra_env,
128                ws.toolchain.as_ref(),
129                load_config.proc_macro_processes,
130            )
131            .map_err(|e| ProcMacroLoadingError::ProcMacroSrvError(e.to_string().into_boxed_str())),
132        ),
133        ProcMacroServerChoice::None => Some(Err(ProcMacroLoadingError::Disabled)),
134    };
135    match &proc_macro_server {
136        Some(Ok(server)) => {
137            tracing::info!(manifest=%ws.manifest_or_root(), path=%server.server_path(), "Proc-macro server started")
138        }
139        Some(Err(e)) => {
140            tracing::info!(manifest=%ws.manifest_or_root(), %e, "Failed to start proc-macro server")
141        }
142        None => {
143            tracing::info!(manifest=%ws.manifest_or_root(), "No proc-macro server started")
144        }
145    }
146
147    let (crate_graph, proc_macros) = ws.to_crate_graph(
148        &mut |path: &AbsPath| {
149            let contents = loader.load_sync(path);
150            let path = vfs::VfsPath::from(path.to_path_buf());
151            vfs.set_file_contents(path.clone(), contents);
152            vfs.file_id(&path).and_then(|(file_id, excluded)| {
153                (excluded == vfs::FileExcluded::No).then_some(file_id)
154            })
155        },
156        extra_env,
157    );
158    let proc_macros = {
159        let proc_macro_server = match &proc_macro_server {
160            Some(Ok(it)) => Ok(it),
161            Some(Err(e)) => {
162                Err(ProcMacroLoadingError::ProcMacroSrvError(e.to_string().into_boxed_str()))
163            }
164            None => Err(ProcMacroLoadingError::ProcMacroSrvError(
165                "proc-macro-srv is not running, workspace is missing a sysroot".into(),
166            )),
167        };
168        proc_macros
169            .into_iter()
170            .map(|(crate_id, path)| {
171                (
172                    crate_id,
173                    path.and_then(|(_, path)| {
174                        proc_macro_server.as_ref().map_err(Clone::clone).and_then(
175                            |proc_macro_server| load_proc_macro(proc_macro_server, &path, &[]),
176                        )
177                    }),
178                )
179            })
180            .collect()
181    };
182
183    let project_folders = ProjectFolders::new(std::slice::from_ref(&ws), &[], None);
184    loader.set_config(vfs::loader::Config {
185        load: project_folders.load,
186        watch: vec![],
187        version: 0,
188    });
189
190    load_crate_graph_into_db(
191        crate_graph,
192        proc_macros,
193        project_folders.source_root_config,
194        &mut vfs,
195        &receiver,
196        db,
197    );
198
199    if load_config.prefill_caches {
200        let all = ide_db::base_db::all_crates(db);
201        prime_caches::parallel_prime_caches(db, &all, load_config.num_worker_threads, &|_| ());
202    }
203
204    Ok((vfs, proc_macro_server.and_then(Result::ok)))
205}

210    pub watch: Vec<usize>,
211    pub source_root_config: SourceRootConfig,

214impl ProjectFolders {
215    pub fn new(
216        workspaces: &[ProjectWorkspace],
217        global_excludes: &[AbsPathBuf],
218        user_config_dir_path: Option<&AbsPath>,
219    ) -> ProjectFolders {
220        let mut res = ProjectFolders::default();
221        let mut fsc = FileSetConfig::builder();
222        let mut local_filesets = vec![];
223
224        // Dedup source roots
225        // Depending on the project setup, we can have duplicated source roots, or for example in
226        // the case of the rustc workspace, we can end up with two source roots that are almost the
227        // same but not quite, like:
228        // PackageRoot { is_local: false, include: [AbsPathBuf(".../rust/src/tools/miri/cargo-miri")], exclude: [] }
229        // PackageRoot {
230        //     is_local: true,
231        //     include: [AbsPathBuf(".../rust/src/tools/miri/cargo-miri"), AbsPathBuf(".../rust/build/x86_64-pc-windows-msvc/stage0-tools/x86_64-pc-windows-msvc/release/build/cargo-miri-85801cd3d2d1dae4/out")],
232        //     exclude: [AbsPathBuf(".../rust/src/tools/miri/cargo-miri/.git"), AbsPathBuf(".../rust/src/tools/miri/cargo-miri/target")]
233        // }
234        //
235        // The first one comes from the explicit rustc workspace which points to the rustc workspace itself
236        // The second comes from the rustc workspace that we load as the actual project workspace
237        // These `is_local` differing in this kind of way gives us problems, especially when trying to filter diagnostics as we don't report diagnostics for external libraries.
238        // So we need to deduplicate these, usually it would be enough to deduplicate by `include`, but as the rustc example shows here that doesn't work,
239        // so we need to also coalesce the includes if they overlap.
240
241        let mut roots: Vec<_> = workspaces
242            .iter()
243            .flat_map(|ws| ws.to_roots())
244            .update(|root| root.include.sort())
245            .sorted_by(|a, b| a.include.cmp(&b.include))
246            .collect();
247
248        // map that tracks indices of overlapping roots
249        let mut overlap_map = FxHashMap::<_, Vec<_>>::default();
250        let mut done = false;
251
252        while !mem::replace(&mut done, true) {
253            // maps include paths to indices of the corresponding root
254            let mut include_to_idx = FxHashMap::default();
255            // Find and note down the indices of overlapping roots
256            for (idx, root) in roots.iter().enumerate().filter(|(_, it)| !it.include.is_empty()) {
257                for include in &root.include {
258                    match include_to_idx.entry(include) {
259                        Entry::Occupied(e) => {
260                            overlap_map.entry(*e.get()).or_default().push(idx);
261                        }
262                        Entry::Vacant(e) => {
263                            e.insert(idx);
264                        }
265                    }
266                }
267            }
268            for (k, v) in overlap_map.drain() {
269                done = false;
270                for v in v {
271                    let r = mem::replace(
272                        &mut roots[v],
273                        PackageRoot { is_local: false, include: vec![], exclude: vec![] },
274                    );
275                    roots[k].is_local |= r.is_local;
276                    roots[k].include.extend(r.include);
277                    roots[k].exclude.extend(r.exclude);
278                }
279                roots[k].include.sort();
280                roots[k].exclude.sort();
281                roots[k].include.dedup();
282                roots[k].exclude.dedup();
283            }
284        }
285
286        // Collect workspace roots not already covered by a local PackageRoot
287        // (e.g. virtual workspaces where no package lives at the workspace root).
288        // We need these to load workspace-root rust-analyzer.toml into a local source root.
289        let uncovered_ws_roots: Vec<AbsPathBuf> = workspaces
290            .iter()
291            .filter_map(|ws| {
292                let ws_root = ws.workspace_root().to_path_buf();
293                let dominated =
294                    roots.iter().any(|root| root.is_local && root.include.contains(&ws_root));
295                (!dominated).then_some(ws_root)
296            })
297            .collect();
298
299        for root in roots.into_iter().filter(|it| !it.include.is_empty()) {
300            let file_set_roots: Vec<VfsPath> =
301                root.include.iter().cloned().map(VfsPath::from).collect();
302
303            let entry = {
304                let mut dirs = vfs::loader::Directories::default();
305                dirs.extensions.push("rs".into());
306                dirs.extensions.push("toml".into());
307                dirs.extensions.push("md".into());
308                dirs.include.extend(root.include);
309                dirs.exclude.extend(root.exclude);
310                for excl in global_excludes {
311                    if dirs
312                        .include
313                        .iter()
314                        .any(|incl| incl.starts_with(excl) || excl.starts_with(incl))
315                    {
316                        dirs.exclude.push(excl.clone());
317                    }
318                }
319
320                vfs::loader::Entry::Directories(dirs)
321            };
322
323            if root.is_local {
324                res.watch.push(res.load.len());
325            }
326            res.load.push(entry);
327
328            if root.is_local {
329                local_filesets.push(fsc.len() as u64);
330            }
331            fsc.add_file_set(file_set_roots)
332        }
333
334        for ws in workspaces.iter() {
335            let mut file_set_roots: Vec<VfsPath> = vec![];
336            let mut entries = vec![];
337
338            for buildfile in ws.buildfiles() {
339                file_set_roots.push(VfsPath::from(buildfile.to_owned()));
340                entries.push(buildfile.to_owned());
341            }
342
343            if !file_set_roots.is_empty() {
344                let entry = vfs::loader::Entry::Files(entries);
345                res.watch.push(res.load.len());
346                res.load.push(entry);
347                local_filesets.push(fsc.len() as u64);
348                fsc.add_file_set(file_set_roots)
349            }
350        }
351
352        // For virtual workspaces, the workspace root has no local PackageRoot, so
353        // rust-analyzer.toml there would fall into a library source root and be
354        // ignored. Load it explicitly via Entry::Files and register the workspace
355        // root as a local file-set root so the file is classified as local.
356        for ws_root in &uncovered_ws_roots {
357            let ratoml_path = ws_root.join("rust-analyzer.toml");
358            let file_set_roots = vec![VfsPath::from(ws_root.clone())];
359            let entry = vfs::loader::Entry::Files(vec![ratoml_path]);
360            res.watch.push(res.load.len());
361            res.load.push(entry);
362            local_filesets.push(fsc.len() as u64);
363            fsc.add_file_set(file_set_roots);
364        }
365
366        if let Some(user_config_path) = user_config_dir_path {
367            let ratoml_path = {
368                let mut p = user_config_path.to_path_buf();
369                p.push("rust-analyzer.toml");
370                p
371            };
372
373            let file_set_roots = vec![VfsPath::from(ratoml_path.to_owned())];
374            let entry = vfs::loader::Entry::Files(vec![ratoml_path]);
375
376            res.watch.push(res.load.len());
377            res.load.push(entry);
378            local_filesets.push(fsc.len() as u64);
379            fsc.add_file_set(file_set_roots)
380        }
381
382        let fsc = fsc.build();
383        res.source_root_config = SourceRootConfig { fsc, local_filesets };
384
385        res
386    }
387}

389#[derive(Default, Debug)]
390pub struct SourceRootConfig {
391    pub fsc: FileSetConfig,
392    pub local_filesets: Vec<u64>,
393}
394
395impl SourceRootConfig {
396    pub fn partition(&self, vfs: &vfs::Vfs) -> Vec<SourceRoot> {
397        self.fsc
398            .partition(vfs)
399            .into_iter()
400            .enumerate()
401            .map(|(idx, file_set)| {
402                let is_local = self.local_filesets.contains(&(idx as u64));
403                if is_local {
404                    SourceRoot::new_local(file_set)
405                } else {
406                    SourceRoot::new_library(file_set)
407                }
408            })
409            .collect()
410    }
411
412    /// Returns whether `path` belongs to a library (non-local) source root, such as the
413    /// sysroot sources or a cargo registry dependency.
414    ///
415    /// Paths that belong to no configured file set are *not* considered library files, as
416    /// files outside of any loaded workspace (for example scratch files) fall into the
417    /// catch-all file set despite being client-editable.
418    pub fn path_is_library(&self, path: &VfsPath) -> bool {
419        match self.fsc.classify_path(path) {
420            Some(idx) => !self.local_filesets.contains(&(idx as u64)),
421            None => false,
422        }
423    }
424
425    /// Maps local source roots to their parent source roots by bytewise comparing of root paths .
426    /// If a `SourceRoot` doesn't have a parent and is local then it is not contained in this mapping but it can be asserted that it is a root `SourceRoot`.
427    pub fn source_root_parent_map(&self) -> FxHashMap<SourceRootId, SourceRootId> {
428        let roots = self.fsc.roots();
429
430        let mut map = FxHashMap::default();
431
432        // See https://github.com/rust-lang/rust-analyzer/issues/17409
433        //
434        // We can view the connections between roots as a graph. The problem is
435        // that this graph may contain cycles, so when adding edges, it is necessary
436        // to check whether it will lead to a cycle.
437        //
438        // Since we ensure that each node has at most one outgoing edge (because
439        // each SourceRoot can have only one parent), we can use a disjoint-set to
440        // maintain the connectivity between nodes. If an edge’s two nodes belong
441        // to the same set, they are already connected.
442        let mut dsu = FxHashMap::default();
443        fn find_parent(dsu: &mut FxHashMap<u64, u64>, id: u64) -> u64 {
444            if let Some(&parent) = dsu.get(&id) {
445                let parent = find_parent(dsu, parent);
446                dsu.insert(id, parent);
447                parent
448            } else {
449                id
450            }
451        }
452
453        for (idx, (root, root_id)) in roots.iter().enumerate() {
454            if !self.local_filesets.contains(root_id)
455                || map.contains_key(&SourceRootId(*root_id as u32))
456            {
457                continue;
458            }
459
460            for (root2, root2_id) in roots[..idx].iter().rev() {
461                if self.local_filesets.contains(root2_id)
462                    && root_id != root2_id
463                    && root.starts_with(root2)
464                {
465                    // check if the edge will create a cycle
466                    if find_parent(&mut dsu, *root_id) != find_parent(&mut dsu, *root2_id) {
467                        map.insert(SourceRootId(*root_id as u32), SourceRootId(*root2_id as u32));
468                        dsu.insert(*root_id, *root2_id);
469                    }
470
471                    break;
472                }
473            }
474        }
475
476        map
477    }
478}

480/// Load the proc-macros for the given lib path, disabling all expanders whose names are in `ignored_macros`.
481pub fn load_proc_macro(
482    server: &ProcMacroClient,
483    path: &AbsPath,
484    ignored_macros: &[Box<str>],
485) -> ProcMacroLoadResult {
486    let res: Result<Vec<_>, _> = (|| {
487        let dylib = MacroDylib::new(path.to_path_buf());
488        let vec = server.load_dylib(dylib).map_err(|e| {
489            ProcMacroLoadingError::ProcMacroSrvError(format!("{e}").into_boxed_str())
490        })?;
491        if vec.is_empty() {
492            return Err(ProcMacroLoadingError::NoProcMacros);
493        }
494        Ok(vec
495            .into_iter()
496            .map(|expander| expander_to_proc_macro(expander, ignored_macros))
497            .collect())
498    })();
499    match res {
500        Ok(proc_macros) => {
501            tracing::info!(
502                "Loaded proc-macros for {path}: {:?}",
503                proc_macros.iter().map(|it| it.name.clone()).collect::<Vec<_>>()
504            );
505            Ok(proc_macros)
506        }
507        Err(e) => {
508            tracing::warn!("proc-macro loading for {path} failed: {e}");
509            Err(e)
510        }
511    }
512}

514fn load_crate_graph_into_db(
515    crate_graph: CrateGraphBuilder,
516    proc_macros: ProcMacrosBuilder,
517    source_root_config: SourceRootConfig,
518    vfs: &mut vfs::Vfs,
519    receiver: &Receiver<vfs::loader::Message>,
520    db: &mut RootDatabase,
521) {
522    let mut analysis_change = ChangeWithProcMacros::default();
523
524    db.enable_proc_attr_macros();
525
526    // wait until Vfs has loaded all roots
527    for task in receiver {
528        match task {
529            vfs::loader::Message::Progress { n_done, .. } => {
530                if n_done == LoadingProgress::Finished {
531                    break;
532                }
533            }
534            vfs::loader::Message::Loaded { files } | vfs::loader::Message::Changed { files } => {
535                let _p =
536                    tracing::info_span!("load_cargo::load_crate_craph/LoadedChanged").entered();
537                for (path, contents) in files {
538                    vfs.set_file_contents(path.into(), contents);
539                }
540            }
541        }
542    }
543    let changes = vfs.take_changes();
544    for (_, file) in changes {
545        if let vfs::Change::Create(v, _) | vfs::Change::Modify(v, _) = file.change
546            && let Ok(text) = String::from_utf8(v)
547        {
548            analysis_change.change_file(file.file_id, Some(text))
549        }
550    }
551    let source_roots = source_root_config.partition(vfs);
552    analysis_change.set_roots(source_roots);
553
554    analysis_change.set_crate_graph(crate_graph);
555    analysis_change.set_proc_macros(proc_macros);
556
557    db.apply_change(analysis_change);
558}

560fn expander_to_proc_macro(
561    expander: proc_macro_api::ProcMacro,
562    ignored_macros: &[Box<str>],
563) -> ProcMacro {
564    let name = expander.name();
565    let kind = match expander.kind() {
566        proc_macro_api::ProcMacroKind::CustomDerive => ProcMacroKind::CustomDerive,
567        proc_macro_api::ProcMacroKind::Bang => ProcMacroKind::Bang,
568        proc_macro_api::ProcMacroKind::Attr => ProcMacroKind::Attr,
569    };
570    let disabled = ignored_macros.iter().any(|replace| **replace == *name);
571    ProcMacro {
572        name: intern::Symbol::intern(name),
573        kind,
574        expander: sync::Arc::new(Expander(expander)),
575        disabled,
576    }
577}

579#[derive(Debug, PartialEq, Eq)]
580struct Expander(proc_macro_api::ProcMacro);
581
582impl ProcMacroExpander for Expander {
583    fn expand(
584        &self,
585        db: &dyn SourceDatabase,
586        subtree: &tt::TopSubtree,
587        attrs: Option<&tt::TopSubtree>,
588        env: &Env,
589        def_site: Span,
590        call_site: Span,
591        mixed_site: Span,
592        current_dir: String,
593    ) -> Result<tt::TopSubtree, ProcMacroExpansionError> {
594        let cb = |req| match req {
595            SubRequest::LocalFilePath { file_id } => {
596                let file_id = FileId::from_raw(file_id);
597                let source_root_id = db.file_source_root(file_id).source_root_id(db);
598                let source_root = db.source_root(source_root_id).source_root(db);
599                let name = source_root
600                    .path_for_file(&file_id)
601                    .and_then(|path| path.as_path())
602                    .map(|path| path.to_string());
603
604                Ok(SubResponse::LocalFilePathResult { name })
605            }
606            // Not incremental: requires full file text.
607            SubRequest::SourceText { file_id, ast_id, start, end } => {
608                let range = resolve_sub_span(
609                    db,
610                    file_id,
611                    ast_id,
612                    TextRange::new(TextSize::from(start), TextSize::from(end)),
613                );
614                let source = db.file_text(range.file_id.file_id(db)).text(db);
615                let text = source
616                    .get(usize::from(range.range.start())..usize::from(range.range.end()))
617                    .map(ToOwned::to_owned);
618
619                Ok(SubResponse::SourceTextResult { text })
620            }
621            // Not incremental: requires building line index.
622            SubRequest::LineColumn { file_id, ast_id, offset } => {
623                let range =
624                    resolve_sub_span(db, file_id, ast_id, TextRange::empty(TextSize::from(offset)));
625                let (line, column) = db
626                    .line_column(range.file_id.file_id(db), range.range.start())
627                    .map(|(line, col)| (line + 1, col + 1))
628                    .unwrap_or((1, 1));
629                // proc_macro::Span line/column are 1-based
630                Ok(SubResponse::LineColumnResult { line, column })
631            }
632            SubRequest::FilePath { file_id } => {
633                let file_id = FileId::from_raw(file_id);
634                let source_root_id = db.file_source_root(file_id).source_root_id(db);
635                let source_root = db.source_root(source_root_id).source_root(db);
636                let name = source_root
637                    .path_for_file(&file_id)
638                    .and_then(|path| path.as_path())
639                    .map(|path| path.to_string())
640                    .unwrap_or_default();
641
642                Ok(SubResponse::FilePathResult { name })
643            }
644            // Not incremental: requires global span resolution.
645            SubRequest::ByteRange { file_id, ast_id, start, end } => {
646                let range = resolve_sub_span(
647                    db,
648                    file_id,
649                    ast_id,
650                    TextRange::new(TextSize::from(start), TextSize::from(end)),
651                );
652
653                Ok(SubResponse::ByteRangeResult { range: range.range.into() })
654            }
655            SubRequest::SpanSource { file_id, ast_id, start, end, ctx } => {
656                let span = Span {
657                    range: TextRange::new(TextSize::from(start), TextSize::from(end)),
658                    anchor: SpanAnchor {
659                        file_id: span::EditionedFileId::from_raw(file_id),
660                        ast_id: span::ErasedFileAstId::from_raw(ast_id),
661                    },
662                    // SAFETY: We only receive spans from the server. If someone mess up the communication UB can happen,
663                    // but that will be their problem.
664                    ctx: unsafe { SyntaxContext::from_u32(ctx) },
665                };
666
667                let mut current_span = span;
668                let mut current_ctx = span.ctx;
669
670                while let Some(macro_call_id) = current_ctx.outer_expn(db) {
671                    let macro_call_loc = hir_expand::MacroCallId::from(macro_call_id).loc(db);
672
673                    let call_site_file = macro_call_loc.kind.file_id();
674
675                    let resolved = hir_expand::resolve_span(db, current_span);
676
677                    current_ctx = macro_call_loc.ctxt;
678                    current_span = Span {
679                        range: resolved.range,
680                        anchor: SpanAnchor {
681                            file_id: resolved.file_id.span_file_id(db),
682                            ast_id: span::ROOT_ERASED_FILE_AST_ID,
683                        },
684                        ctx: current_ctx,
685                    };
686
687                    if call_site_file.file_id().is_some() {
688                        break;
689                    }
690                }
691
692                let resolved = hir_expand::resolve_span(db, current_span);
693
694                Ok(SubResponse::SpanSourceResult {
695                    file_id: resolved.file_id.span_file_id(db).as_u32(),
696                    ast_id: span::ROOT_ERASED_FILE_AST_ID.into_raw(),
697                    start: u32::from(resolved.range.start()),
698                    end: u32::from(resolved.range.end()),
699                    ctx: current_span.ctx.into_u32(),
700                })
701            }
702            SubRequest::SpanParent { file_id, ast_id, start, end, ctx } => {
703                let span = Span {
704                    range: TextRange::new(TextSize::from(start), TextSize::from(end)),
705                    anchor: SpanAnchor {
706                        file_id: span::EditionedFileId::from_raw(file_id),
707                        ast_id: span::ErasedFileAstId::from_raw(ast_id),
708                    },
709                    // SAFETY: We only receive spans from the server. If someone mess up the communication UB can happen,
710                    // but that will be their problem.
711                    ctx: unsafe { SyntaxContext::from_u32(ctx) },
712                };
713
714                if let Some(macro_call_id) = span.ctx.outer_expn(db) {
715                    let macro_call_loc = hir_expand::MacroCallId::from(macro_call_id).loc(db);
716
717                    let call_site_file = macro_call_loc.kind.file_id();
718                    let call_site_ast_id = macro_call_loc.kind.erased_ast_id();
719
720                    if let Some(editioned_file_id) = call_site_file.file_id() {
721                        let range = hir_expand::HirFileId::from(editioned_file_id)
722                            .ast_id_map(db)
723                            .get_erased(call_site_ast_id)
724                            .text_range();
725
726                        let parent_span = Some(ParentSpan {
727                            file_id: editioned_file_id.span_file_id(db).as_u32(),
728                            ast_id: span::ROOT_ERASED_FILE_AST_ID.into_raw(),
729                            start: u32::from(range.start()),
730                            end: u32::from(range.end()),
731                            ctx: macro_call_loc.ctxt.into_u32(),
732                        });
733
734                        return Ok(SubResponse::SpanParentResult { parent_span });
735                    }
736                }
737
738                Ok(SubResponse::SpanParentResult { parent_span: None })
739            }
740            // FIXME: implement this
741            SubRequest::SpanJoin { .. } => Ok(SubResponse::SpanJoinResult { span: None }),
742        };
743        match self.0.expand(
744            subtree.view(),
745            attrs.map(|attrs| attrs.view()),
746            env.clone().into(),
747            def_site,
748            call_site,
749            mixed_site,
750            current_dir,
751            Some(&cb),
752        ) {
753            Ok(Ok(subtree)) => Ok(subtree),
754            Ok(Err(err)) => Err(ProcMacroExpansionError::Panic(err)),
755            Err(err) => Err(ProcMacroExpansionError::System(err.to_string())),
756        }
757    }
758
759    fn eq_dyn(&self, other: &dyn ProcMacroExpander) -> bool {
760        (other as &dyn Any).downcast_ref::<Self>() == Some(self)
761    }
762}
763
764fn resolve_sub_span(
765    db: &dyn SourceDatabase,
766    file_id: u32,
767    ast_id: u32,
768    range: TextRange,
769) -> hir_expand::FileRange {
770    let ast_id = span::ErasedFileAstId::from_raw(ast_id);
771    let editioned_file_id = span::EditionedFileId::from_raw(file_id);
772    let span = Span {
773        range,
774        anchor: SpanAnchor { file_id: editioned_file_id, ast_id },
775        ctx: SyntaxContext::root(editioned_file_id.edition()),
776    };
777    hir_expand::resolve_span(db, span)
778}
