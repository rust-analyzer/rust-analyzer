48// Feature: Inlay Hints
49//
50// rust-analyzer shows additional information inline with the source code.
51// Editors usually render this using read-only virtual text snippets interspersed with code.
52//
53// rust-analyzer by default shows hints for
54//
55// * types of local variables
56// * names of function arguments
57// * names of const generic parameters
58// * types of chained expressions
59//
60// Optionally, one can enable additional hints for
61//
62// * return types of closure expressions
63// * elided lifetimes
64// * compiler inserted reborrows
65// * names of generic type and lifetime parameters
66//
67// Note: inlay hints for function argument names are heuristically omitted to reduce noise and will not appear if
68// any of the
69// [following criteria](https://github.com/rust-lang/rust-analyzer/blob/6b8b8ff4c56118ddee6c531cde06add1aad4a6af/crates/ide/src/inlay_hints/param_name.rs#L92-L99)
70// are met:
71//
72// * the parameter name is a suffix of the function's name
73// * the argument is a qualified constructing or call expression where the qualifier is an ADT
74// * exact argument<->parameter match(ignoring leading underscore) or parameter is a prefix/suffix
75//   of argument with _ splitting it off
76// * the parameter name starts with `ra_fixture`
77// * the parameter name is a
78// [well known name](https://github.com/rust-lang/rust-analyzer/blob/6b8b8ff4c56118ddee6c531cde06add1aad4a6af/crates/ide/src/inlay_hints/param_name.rs#L200)
79// in a unary function
80// * the parameter name is a
81// [single character](https://github.com/rust-lang/rust-analyzer/blob/6b8b8ff4c56118ddee6c531cde06add1aad4a6af/crates/ide/src/inlay_hints/param_name.rs#L201)
82// in a unary function
83//
84// ![Inlay hints](https://user-images.githubusercontent.com/48062697/113020660-b5f98b80-917a-11eb-8d70-3be3fd558cdd.png)
85pub(crate) fn inlay_hints(
86    db: &RootDatabase,
87    file_id: FileId,
88    range_limit: Option<TextRange>,
89    config: &InlayHintsConfig<'_>,
90) -> Vec<InlayHint> {
91    let _p = tracing::info_span!("inlay_hints").entered();
92    let sema = Semantics::new(db);
93    let file_id = sema.attach_first_edition(file_id);
94    let file = sema.parse(file_id);
95    let file = file.syntax();
96
97    let mut acc = Vec::new();
98
99    let Some(scope) = sema.scope(file) else {
100        return acc;
101    };
102    let famous_defs = FamousDefs(&sema, scope.krate());
103    let display_target = famous_defs.1.to_display_target(sema.db);
104
105    let ctx = &mut InlayHintCtx::default();
106    let mut hints = |event| {
107        if let Some(node) = handle_event(ctx, event) {
108            hints(&mut acc, ctx, &famous_defs, config, file_id, display_target, node);
109        }
110    };
111    let mut preorder = file.preorder();
112    while let Some(event) = preorder.next() {
113        if matches!((&event, range_limit), (WalkEvent::Enter(node), Some(range)) if range.intersect(node.text_range()).is_none())
114        {
115            preorder.skip_subtree();
116            continue;
117        }
118        hints(event);
119    }
120    if let Some(range_limit) = range_limit {
121        acc.retain(|hint| range_limit.contains_range(hint.range));
122    }
123    acc
124}

132pub(crate) fn inlay_hints_resolve(
133    db: &RootDatabase,
134    file_id: FileId,
135    resolve_range: TextRange,
136    hash: u64,
137    config: &InlayHintsConfig<'_>,
138    hasher: impl Fn(&InlayHint) -> u64,
139) -> Option<InlayHint> {
140    let _p = tracing::info_span!("inlay_hints_resolve").entered();
141    let sema = Semantics::new(db);
142    let file_id = sema.attach_first_edition(file_id);
143    let file = sema.parse(file_id);
144    let file = file.syntax();
145
146    let scope = sema.scope(file)?;
147    let famous_defs = FamousDefs(&sema, scope.krate());
148    let mut acc = Vec::new();
149
150    let display_target = famous_defs.1.to_display_target(sema.db);
151
152    let ctx = &mut InlayHintCtx::default();
153    let mut hints = |event| {
154        if let Some(node) = handle_event(ctx, event) {
155            hints(&mut acc, ctx, &famous_defs, config, file_id, display_target, node);
156        }
157    };
158
159    let mut preorder = file.preorder();
160    while let Some(event) = preorder.next() {
161        // FIXME: This can miss some hints that require the parent of the range to calculate
162        if matches!(&event, WalkEvent::Enter(node) if resolve_range.intersect(node.text_range()).is_none())
163        {
164            preorder.skip_subtree();
165            continue;
166        }
167        hints(event);
168    }
169    acc.into_iter().find(|hint| hasher(hint) == hash)
170}

172fn handle_event(ctx: &mut InlayHintCtx, node: WalkEvent<SyntaxNode>) -> Option<SyntaxNode> {
173    match node {
174        WalkEvent::Enter(node) => {
175            if let Some(node) = ast::AnyHasGenericParams::cast(node.clone()) {
176                let params = node
177                    .generic_param_list()
178                    .map(|it| {
179                        it.lifetime_params()
180                            .filter_map(|it| {
181                                it.lifetime().map(|it| format_smolstr!("{}", &it.text()[1..]))
182                            })
183                            .collect()
184                    })
185                    .unwrap_or_default();
186                ctx.lifetime_stacks.push(params);
187            }
188            if let Some(node) = ast::ExternBlock::cast(node.clone()) {
189                ctx.extern_block_parent = Some(node);
190            }
191            Some(node)
192        }
193        WalkEvent::Leave(n) => {
194            if ast::AnyHasGenericParams::can_cast(n.kind()) {
195                ctx.lifetime_stacks.pop();
196            }
197            if ast::ExternBlock::can_cast(n.kind()) {
198                ctx.extern_block_parent = None;
199            }
200            None
201        }
202    }
203}

205// FIXME: At some point when our hir infra is fleshed out enough we should flip this and traverse the
206// HIR instead of the syntax tree.
207fn hints(
208    hints: &mut Vec<InlayHint>,
209    ctx: &mut InlayHintCtx,
210    famous_defs @ FamousDefs(sema, _krate): &FamousDefs<'_, '_>,
211    config: &InlayHintsConfig<'_>,
212    file_id: EditionedFileId,
213    display_target: DisplayTarget,
214    node: SyntaxNode,
215) {
216    closing_brace::hints(
217        hints,
218        sema,
219        config,
220        display_target,
221        InRealFile { file_id, value: node.clone() },
222    );
223    if let Some(any_has_generic_args) = ast::AnyHasGenericArgs::cast(node.clone()) {
224        generic_param::hints(hints, famous_defs, config, any_has_generic_args);
225    }
226
227    match_ast! {
228        match node {
229            ast::Expr(expr) => {
230                chaining::hints(hints, famous_defs, config, display_target, &expr);
231                adjustment::hints(hints, famous_defs, config, display_target, &expr);
232                match expr {
233                    ast::Expr::CallExpr(it) => param_name::hints(hints, famous_defs, config, file_id, ast::Expr::from(it)),
234                    ast::Expr::MethodCallExpr(it) => {
235                        param_name::hints(hints, famous_defs, config, file_id, ast::Expr::from(it))
236                    }
237                    ast::Expr::ClosureExpr(it) => {
238                        closure_captures::hints(
239                            hints,
240                            famous_defs,
241                            config,
242                            Either::Left(it.clone()),
243                            file_id.edition(sema.db),
244                        );
245                        closure_ret::hints(hints, famous_defs, config, display_target, it)
246                    },
247                    ast::Expr::BlockExpr(it) => closure_captures::hints(
248                        hints,
249                        famous_defs,
250                        config,
251                        Either::Right(it),
252                        file_id.edition(sema.db),
253                    ),
254                    ast::Expr::RangeExpr(it) => range_exclusive::hints(hints, famous_defs, config, it),
255                    ast::Expr::Literal(it) => ra_fixture::hints(hints, famous_defs.0, file_id, config, it),
256                    _ => Some(()),
257                }
258            },
259            ast::Pat(it) => {
260                binding_mode::hints(hints, famous_defs, config, &it);
261                match it {
262                    ast::Pat::IdentPat(it) => {
263                        bind_pat::hints(hints, famous_defs, config, display_target, &it);
264                    }
265                    ast::Pat::RangePat(it) => {
266                        range_exclusive::hints(hints, famous_defs, config, it);
267                    }
268                    _ => {}
269                }
270                Some(())
271            },
272            ast::Item(it) => match it {
273                ast::Item::Fn(it) => {
274                    implicit_drop::hints(hints, famous_defs, config, display_target, &it);
275                    if let Some(extern_block) = &ctx.extern_block_parent {
276                        extern_block::fn_hints(hints, famous_defs, config, &it, extern_block);
277                    }
278                    lifetime::fn_hints(hints, ctx, famous_defs, config,  it)
279                },
280                ast::Item::Static(it) => {
281                    if let Some(extern_block) = &ctx.extern_block_parent {
282                        extern_block::static_hints(hints, famous_defs, config, &it, extern_block);
283                    }
284                    implicit_static::hints(hints, famous_defs, config,  Either::Left(it))
285                },
286                ast::Item::Const(it) => implicit_static::hints(hints, famous_defs, config, Either::Right(it)),
287                ast::Item::Enum(it) => discriminant::enum_hints(hints, famous_defs, config, it),
288                ast::Item::ExternBlock(it) => extern_block::extern_block_hints(hints, famous_defs, config, it),
289                _ => None,
290            },
291            // FIXME: trait object type elisions
292            ast::Type(ty) => match ty {
293                ast::Type::FnPtrType(ptr) => lifetime::fn_ptr_hints(hints, ctx, famous_defs, config,  ptr),
294                ast::Type::PathType(path) => {
295                    lifetime::fn_path_hints(hints, ctx, famous_defs, config, &path);
296                    implied_dyn_trait::hints(hints, famous_defs, config, Either::Left(path));
297                    Some(())
298                },
299                ast::Type::DynTraitType(dyn_) => {
300                    implied_dyn_trait::hints(hints, famous_defs, config, Either::Right(dyn_));
301                    Some(())
302                },
303                ast::Type::InferType(placeholder) => {
304                    placeholders::type_hints(hints, famous_defs, config, display_target, placeholder);
305                    Some(())
306                },
307                _ => Some(()),
308            },
309            ast::GenericParamList(it) => bounds::hints(hints, famous_defs, config,  it),
310            _ => Some(()),
311        }
312    };
313}

355impl InlayHintsConfig<'_> {
356    fn lazy_text_edit(&self, finish: impl FnOnce() -> TextEdit) -> LazyProperty<TextEdit> {
357        if self.fields_to_resolve.resolve_text_edits {
358            LazyProperty::Lazy
359        } else {
360            let edit = finish();
361            never!(edit.is_empty(), "inlay hint produced an empty text edit");
362            LazyProperty::Computed(edit)
363        }
364    }
365
366    fn lazy_tooltip(&self, finish: impl FnOnce() -> InlayTooltip) -> LazyProperty<InlayTooltip> {
367        if self.fields_to_resolve.resolve_hint_tooltip
368            && self.fields_to_resolve.resolve_label_tooltip
369        {
370            LazyProperty::Lazy
371        } else {
372            let tooltip = finish();
373            never!(
374                match &tooltip {
375                    InlayTooltip::String(s) => s,
376                    InlayTooltip::Markdown(s) => s,
377                }
378                .is_empty(),
379                "inlay hint produced an empty tooltip"
380            );
381            LazyProperty::Computed(tooltip)
382        }
383    }
384
385    /// This always reports a resolvable location, so only use this when it is very likely for a
386    /// location link to actually resolve but where computing `finish` would be costly.
387    fn lazy_location_opt(
388        &self,
389        finish: impl FnOnce() -> Option<FileRange>,
390    ) -> Option<LazyProperty<FileRange>> {
391        if self.fields_to_resolve.resolve_label_location {
392            Some(LazyProperty::Lazy)
393        } else {
394            finish().map(LazyProperty::Computed)
395        }
396    }
397}

582impl InlayHintLabel {
583    pub fn simple(
584        s: impl Into<String>,
585        tooltip: Option<LazyProperty<InlayTooltip>>,
586        linked_location: Option<LazyProperty<FileRange>>,
587    ) -> InlayHintLabel {
588        InlayHintLabel {
589            parts: smallvec![InlayHintLabelPart { text: s.into(), linked_location, tooltip }],

592
593    pub fn prepend_str(&mut self, s: &str) {
594        match &mut *self.parts {
595            [InlayHintLabelPart { text, linked_location: None, tooltip: None }, ..] => {
596                text.insert_str(0, s)
597            }
598            _ => self.parts.insert(
599                0,
600                InlayHintLabelPart { text: s.into(), linked_location: None, tooltip: None },
601            ),
602        }
603    }
604
605    pub fn append_str(&mut self, s: &str) {
606        match &mut *self.parts {
607            [.., InlayHintLabelPart { text, linked_location: None, tooltip: None }] => {
608                text.push_str(s)
609            }
610            _ => self.parts.push(InlayHintLabelPart {
611                text: s.into(),
612                linked_location: None,
613                tooltip: None,
614            }),
615        }
616    }
617
618    pub fn append_part(&mut self, part: InlayHintLabelPart) {
619        if part.linked_location.is_none()
620            && part.tooltip.is_none()
621            && let Some(InlayHintLabelPart { text, linked_location: None, tooltip: None }) =
622                self.parts.last_mut()
623        {
624            text.push_str(&part.text);
625            return;
626        }
627        self.parts.push(part);
628    }
629}

666    /// Source location represented by this label part. The client will use this to fetch the part's
667    /// hover tooltip, and Ctrl+Clicking the label part will navigate to the definition the location
668    /// refers to (not necessarily the location itself).
669    /// When setting this, no tooltip must be set on the containing hint, or VS Code will display
670    /// them both.
671    pub linked_location: Option<LazyProperty<FileRange>>,
672    /// The tooltip to show when hovering over the inlay hint, this may invoke other actions like
673    /// hover requests to show.
674    pub tooltip: Option<LazyProperty<InlayTooltip>>,
675}
676
677impl std::hash::Hash for InlayHintLabelPart {
678    fn hash<H: std::hash::Hasher>(&self, state: &mut H) {
679        self.text.hash(state);
680        self.linked_location.is_some().hash(state);
681        self.tooltip.is_some().hash(state);
682    }
683}
684
685impl fmt::Debug for InlayHintLabelPart {
686    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
687        match self {
688            Self { text, linked_location: None, tooltip: None | Some(LazyProperty::Lazy) } => {
689                text.fmt(f)
690            }
691            Self { text, linked_location, tooltip } => f
692                .debug_struct("InlayHintLabelPart")
693                .field("text", text)
694                .field("linked_location", linked_location)
695                .field(
696                    "tooltip",
697                    &tooltip.as_ref().map_or("", |it| match it {
698                        LazyProperty::Computed(
699                            InlayTooltip::String(it) | InlayTooltip::Markdown(it),
700                        ) => it,
701                        LazyProperty::Lazy => "",
702                    }),
703                )
704                .finish(),
705        }
706    }
707}

709#[derive(Debug)]
710struct InlayHintLabelBuilder<'a, 'db> {
711    sema: &'a Semantics<'db, RootDatabase>,
712    result: InlayHintLabel,
713    last_part: String,
714    resolve: bool,
715    location: Option<LazyProperty<FileRange>>,
716}

718impl fmt::Write for InlayHintLabelBuilder<'_, '_> {
719    fn write_str(&mut self, s: &str) -> fmt::Result {
720        self.last_part.write_str(s)
721    }
722}

724impl HirWrite for InlayHintLabelBuilder<'_, '_> {
725    fn start_location_link(&mut self, def: ModuleDefId) {
726        never!(self.location.is_some(), "location link is already started");
727        self.make_new_part();
728
729        self.location = Some(if self.resolve {
730            LazyProperty::Lazy
731        } else {
732            LazyProperty::Computed({
733                let Some(location) = ModuleDef::from(def).try_to_nav(self.sema) else { return };
734                let location = location.call_site();
735                FileRange { file_id: location.file_id, range: location.focus_or_full_range() }
736            })
737        });
738    }
739
740    fn start_location_link_generic(&mut self, def: GenericParamId) {
741        never!(self.location.is_some(), "location link is already started");
742        self.make_new_part();
743
744        self.location = Some(if self.resolve {
745            LazyProperty::Lazy
746        } else {
747            LazyProperty::Computed({
748                let Some(location) = GenericParam::from(def).try_to_nav(self.sema) else { return };
749                let location = location.call_site();
750                FileRange { file_id: location.file_id, range: location.focus_or_full_range() }
751            })
752        });
753    }
754
755    fn end_location_link(&mut self) {
756        self.make_new_part();
757    }
758}
759
760impl InlayHintLabelBuilder<'_, '_> {
761    fn make_new_part(&mut self) {
762        let text = take(&mut self.last_part);
763        if !text.is_empty() {
764            self.result.parts.push(InlayHintLabelPart {
765                text,
766                linked_location: self.location.take(),
767                tooltip: None,
768            });
769        }
770    }
771
772    fn finish(mut self) -> InlayHintLabel {
773        self.make_new_part();
774        self.result
775    }
776}

778fn label_of_ty<'db>(
779    famous_defs @ FamousDefs(sema, _): &FamousDefs<'_, 'db>,
780    config: &InlayHintsConfig<'_>,
781    ty: &hir::Type<'db>,
782    display_target: DisplayTarget,
783) -> Option<InlayHintLabel> {
784    fn rec<'db>(
785        sema: &Semantics<'db, RootDatabase>,
786        famous_defs: &FamousDefs<'_, 'db>,
787        mut max_length: Option<usize>,
788        ty: &hir::Type<'db>,
789        label_builder: &mut InlayHintLabelBuilder<'_, '_>,
790        config: &InlayHintsConfig<'_>,
791        display_target: DisplayTarget,
792    ) -> Result<(), HirDisplayError> {
793        let iter_item_type = hint_iterator(sema, famous_defs, ty);
794        match iter_item_type {
795            Some((iter_trait, item, ty)) => {
796                const LABEL_START: &str = "impl ";
797                const LABEL_ITERATOR: &str = "Iterator";
798                const LABEL_MIDDLE: &str = "<";
799                const LABEL_ITEM: &str = "Item";
800                const LABEL_MIDDLE2: &str = " = ";
801                const LABEL_END: &str = ">";
802
803                max_length = max_length.map(|len| {
804                    len.saturating_sub(
805                        LABEL_START.len()
806                            + LABEL_ITERATOR.len()
807                            + LABEL_MIDDLE.len()
808                            + LABEL_MIDDLE2.len()
809                            + LABEL_END.len(),
810                    )
811                });
812
813                let module_def_location = |label_builder: &mut InlayHintLabelBuilder<'_, '_>,
814                                           def: ModuleDef,
815                                           name| {
816                    let def = def.try_into();
817                    if let Ok(def) = def {
818                        label_builder.start_location_link(def);
819                    }
820                    #[expect(
821                        clippy::question_mark,
822                        reason = "false positive; replacing with `?` leads to 'type annotations needed' error"
823                    )]
824                    if let Err(err) = label_builder.write_str(name) {
825                        return Err(err);
826                    }
827                    if def.is_ok() {
828                        label_builder.end_location_link();
829                    }
830                    Ok(())
831                };
832
833                label_builder.write_str(LABEL_START)?;
834                module_def_location(label_builder, ModuleDef::from(iter_trait), LABEL_ITERATOR)?;
835                label_builder.write_str(LABEL_MIDDLE)?;
836                module_def_location(label_builder, ModuleDef::from(item), LABEL_ITEM)?;
837                label_builder.write_str(LABEL_MIDDLE2)?;
838                rec(sema, famous_defs, max_length, &ty, label_builder, config, display_target)?;
839                label_builder.write_str(LABEL_END)?;
840                Ok(())
841            }
842            None => ty
843                .display_truncated(sema.db, max_length, display_target)
844                .with_closure_style(config.closure_style)
845                .write_to(label_builder),
846        }
847    }
848
849    let mut label_builder = InlayHintLabelBuilder {
850        sema,
851        last_part: String::new(),
852        location: None,
853        result: InlayHintLabel::default(),
854        resolve: config.fields_to_resolve.resolve_label_location,
855    };
856    let _ =
857        rec(sema, famous_defs, config.max_length, ty, &mut label_builder, config, display_target);
858    let r = label_builder.finish();
859    Some(r)
860}

862/// Checks if the type is an Iterator from std::iter and returns the iterator trait and the item type of the concrete iterator.
863fn hint_iterator<'db>(
864    sema: &Semantics<'db, RootDatabase>,
865    famous_defs: &FamousDefs<'_, 'db>,
866    ty: &hir::Type<'db>,
867) -> Option<(hir::Trait, hir::TypeAlias, hir::Type<'db>)> {
868    let db = sema.db;
869    let strukt = ty.strip_references().as_adt()?;
870    let krate = strukt.module(db).krate(db);
871    if krate != famous_defs.core()? {
872        return None;
873    }
874    let iter_trait = famous_defs.core_iter_Iterator()?;
875    let iter_mod = famous_defs.core_iter()?;
876
877    // Assert that this struct comes from `core::iter`.
878    if !(strukt.visibility(db) == hir::Visibility::Public
879        && strukt.module(db).path_to_root(db).contains(&iter_mod))
880    {
881        return None;
882    }
883
884    if ty.impls_trait(db, iter_trait, &[]) {
885        let assoc_type_item = iter_trait.items(db).into_iter().find_map(|item| match item {
886            hir::AssocItem::TypeAlias(alias) if alias.name(db) == sym::Item => Some(alias),
887            _ => None,
888        })?;
889        if let Some(ty) = ty.normalize_trait_assoc_type(db, &[], assoc_type_item) {
890            return Some((iter_trait, assoc_type_item, ty));
891        }
892    }
893
894    None
895}

897fn ty_to_text_edit(
898    sema: &Semantics<'_, RootDatabase>,
899    config: &InlayHintsConfig<'_>,
900    node_for_hint: &SyntaxNode,
901    ty: &hir::Type<'_>,
902    offset_to_insert_ty: TextSize,
903    additional_edits: &dyn Fn(&mut TextEditBuilder),
904    prefix: impl Into<String>,
905) -> Option<LazyProperty<TextEdit>> {
906    // FIXME: Limit the length and bail out on excess somehow?
907    let rendered = sema
908        .scope(node_for_hint)
909        .and_then(|scope| ty.display_source_code(scope.db, scope.module().into(), false).ok())?;
910    Some(config.lazy_text_edit(|| {
911        let mut builder = TextEdit::builder();
912        builder.insert(offset_to_insert_ty, prefix.into());
913        builder.insert(offset_to_insert_ty, rendered);
914
915        additional_edits(&mut builder);
916
917        builder.finish()
918    }))
919}
