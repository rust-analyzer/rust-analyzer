175impl ExpandErrorKind {
176    pub fn render_to_string(&self, db: &dyn SourceDatabase) -> RenderedExpandError {
177        match self {
178            ExpandErrorKind::ProcMacroAttrExpansionDisabled => RenderedExpandError {
179                message: "procedural attribute macro expansion is disabled".to_owned(),
180                error: false,
181                kind: RenderedExpandError::ATTR_EXP_DISABLED,
182            },
183            ExpandErrorKind::MacroDisabled => RenderedExpandError {
184                message: "proc-macro is explicitly disabled".to_owned(),
185                error: false,
186                kind: RenderedExpandError::DISABLED,
187            },
188            &ExpandErrorKind::MissingProcMacroExpander(def_crate) => {
189                match ProcMacros::get_for_crate(db, def_crate).and_then(|it| it.get_error()) {
190                    Some(e) => RenderedExpandError {
191                        message: e.to_string(),
192                        error: e.is_hard_error(),
193                        kind: RenderedExpandError::GENERAL_KIND,
194                    },
195                    None => RenderedExpandError {
196                        message: format!(
197                            "internal error: proc-macro map is missing error entry for crate {def_crate:?}"
198                        ),
199                        error: true,
200                        kind: RenderedExpandError::GENERAL_KIND,
201                    },
202                }
203            }
204            ExpandErrorKind::MacroDefinition => RenderedExpandError {
205                message: "macro definition has parse errors".to_owned(),
206                error: true,
207                kind: RenderedExpandError::GENERAL_KIND,
208            },
209            ExpandErrorKind::Mbe(e) => RenderedExpandError {
210                message: e.to_string(),
211                error: true,
212                kind: RenderedExpandError::GENERAL_KIND,
213            },
214            ExpandErrorKind::RecursionOverflow => RenderedExpandError {
215                message: "overflow expanding the original macro".to_owned(),
216                error: true,
217                kind: RenderedExpandError::GENERAL_KIND,
218            },
219            ExpandErrorKind::Other(e) => RenderedExpandError {
220                message: (**e).to_owned(),
221                error: true,
222                kind: RenderedExpandError::GENERAL_KIND,
223            },
224            ExpandErrorKind::ProcMacroPanic(e) => RenderedExpandError {
225                message: format!("proc-macro panicked: {e}"),
226                error: true,
227                kind: RenderedExpandError::GENERAL_KIND,
228            },
229        }
230    }

486#[salsa::tracked]
487impl MacroCallId {
488    /// Implementation of [`HirFileId::parse_or_expand`] for the macro case.
489    // FIXME: We should verify that the parsed node is one of the many macro node variants we expect
490    // instead of having it be untyped
491    #[salsa::tracked(returns(ref), lru = 512)]
492    pub fn parse_macro_expansion(
493        self,
494        db: &dyn SourceDatabase,
495    ) -> ExpandResult<(Parse<SyntaxNode>, ExpansionSpanMap)> {
496        let _p = tracing::info_span!("parse_macro_expansion").entered();
497        let loc = self.loc(db);
498        let expand_to = loc.expand_to();
499        let mbe::ValueResult { value: (tt, matched_arm), err } = self.macro_expand(db, loc);
500
501        let (parse, mut rev_token_map) = token_tree_to_syntax_node(db, &tt, expand_to);
502        rev_token_map.matched_arm = matched_arm;
503
504        ExpandResult { value: (parse, rev_token_map), err }
505    }
506
507    pub fn parse_macro_expansion_error(
508        self,
509        db: &dyn SourceDatabase,
510    ) -> Option<ExpandResult<Arc<[SyntaxError]>>> {
511        let e: ExpandResult<Arc<[SyntaxError]>> =
512            self.parse_macro_expansion(db).as_ref().map(|it| Arc::from(it.0.errors()));
513        if e.value.is_empty() && e.err.is_none() { None } else { Some(e) }
514    }
515
516    /// This resolves the [MacroCallId] to check if it is a derive macro if so get the [macro_arg] for the derive.
517    /// Other wise return the [macro_arg] for the macro_call_id.
518    ///
519    /// This is not connected to the database so it does not cache the result. However, the inner [macro_arg] query is
520    ///
521    /// [macro_arg]: Self::macro_arg
522    #[expect(deprecated, reason = "we are `macro_arg_considering_derives`")]
523    pub fn macro_arg_considering_derives<'db>(
524        self,
525        db: &'db dyn SourceDatabase,
526        kind: &MacroCallKind,
527    ) -> &'db MacroArgResult {
528        match kind {
529            // Get the macro arg for the derive macro
530            MacroCallKind::Derive { derive_macro_id, .. } => derive_macro_id.macro_arg(db),
531            // Normal macro arg
532            _ => self.macro_arg(db),
533        }
534    }
535
536    /// Lowers syntactic macro call to a token tree representation. That's a firewall
537    /// query, only typing in the macro call itself changes the returned
538    /// subtree.
539    #[salsa::tracked(returns(ref))]
540    #[deprecated = "calling this is incorrect, call `macro_arg_considering_derives` instead"]
541    fn macro_arg(self, db: &dyn SourceDatabase) -> MacroArgResult {
542        let loc = self.loc(db);
543
544        if let MacroCallLoc {
545            def: MacroDefId { kind: MacroDefKind::BuiltInEager(..), .. },
546            kind: MacroCallKind::FnLike { eager: Some(eager), .. },
547            ..
548        } = &loc
549        {
550            return (eager.arg.clone(), SyntaxFixupUndoInfo::NONE, eager.span);
551        }
552
553        let (parse, map) = loc.kind.file_id().parse_with_map(db);
554        let root = parse.syntax_node();
555
556        let (is_derive, censor_item_tree_attr_ids, item_node, span) = match &loc.kind {
557            MacroCallKind::FnLike { ast_id, .. } => {
558                let node = &ast_id.to_ptr(db).to_node(&root);
559                let path_range = node
560                    .path()
561                    .map_or_else(|| node.syntax().text_range(), |path| path.syntax().text_range());
562                let span = map.span_for_range(path_range);
563
564                let dummy_tt = |kind| {
565                    (
566                        tt::TopSubtree::from_token_trees(
567                            tt::Delimiter { open: span, close: span, kind },
568                            tt::TokenTreesView::empty(),
569                        ),
570                        SyntaxFixupUndoInfo::default(),
571                        span,
572                    )
573                };
574
575                let Some(tt) = node.token_tree() else {
576                    return dummy_tt(tt::DelimiterKind::Invisible);
577                };
578                let first = tt.left_delimiter_token().map(|it| it.kind()).unwrap_or(T!['(']);
579                let last = tt.right_delimiter_token().map(|it| it.kind()).unwrap_or(T![.]);
580
581                let mismatched_delimiters = !matches!(
582                    (first, last),
583                    (T!['('], T![')']) | (T!['['], T![']']) | (T!['{'], T!['}'])
584                );
585                if mismatched_delimiters {
586                    // Don't expand malformed (unbalanced) macro invocations. This is
587                    // less than ideal, but trying to expand unbalanced  macro calls
588                    // sometimes produces pathological, deeply nested code which breaks
589                    // all kinds of things.
590                    //
591                    // So instead, we'll return an empty subtree here
592                    cov_mark::hit!(issue9358_bad_macro_stack_overflow);
593
594                    let kind = match first {
595                        _ if loc.def.is_proc_macro() => tt::DelimiterKind::Invisible,
596                        T!['('] => tt::DelimiterKind::Parenthesis,
597                        T!['['] => tt::DelimiterKind::Bracket,
598                        T!['{'] => tt::DelimiterKind::Brace,
599                        _ => tt::DelimiterKind::Invisible,
600                    };
601                    return dummy_tt(kind);
602                }
603
604                let mut tt = syntax_bridge::syntax_node_to_token_tree(
605                    tt.syntax(),
606                    map,
607                    span,
608                    if loc.def.is_proc_macro() {
609                        DocCommentDesugarMode::ProcMacro
610                    } else {
611                        DocCommentDesugarMode::Mbe
612                    },
613                );
614                if loc.def.is_proc_macro() {
615                    // proc macros expect their inputs without parentheses, MBEs expect it with them included
616                    tt.set_top_subtree_delimiter_kind(tt::DelimiterKind::Invisible);
617                }
618                return (tt, SyntaxFixupUndoInfo::NONE, span);
619            }
620            // MacroCallKind::Derive should not be here. As we are getting the argument for the derive macro
621            MacroCallKind::Derive { .. } => {
622                unreachable!("`MacroCallId::macro_arg` called with `MacroCallKind::Derive`")
623            }
624            MacroCallKind::Attr { ast_id, censored_attr_ids: attr_ids, .. } => {
625                let node = ast_id.to_ptr(db).to_node(&root);
626                let (_, attr) =
627                    attr_ids.invoc_attr().find_attr_range_with_source(db, loc.krate, &node);
628                let range = attr
629                    .path()
630                    .map(|path| path.syntax().text_range())
631                    .unwrap_or_else(|| attr.syntax().text_range());
632                let span = map.span_for_range(range);
633
634                let is_derive = matches!(loc.def.kind, MacroDefKind::BuiltInAttr(_, expander) if expander.is_derive());
635                (is_derive, &**attr_ids, node, span)
636            }
637        };
638
639        let (mut tt, undo_info) = attr_macro_input_to_token_tree(
640            db,
641            item_node.syntax(),
642            map,
643            span,
644            is_derive,
645            censor_item_tree_attr_ids,
646            loc.krate,
647        );
648
649        if loc.def.is_proc_macro() {
650            // proc macros expect their inputs without parentheses, MBEs expect it with them included
651            tt.set_top_subtree_delimiter_kind(tt::DelimiterKind::Invisible);
652        }
653
654        (tt, undo_info, span)
655    }

657    fn macro_expand<'db>(
658        self,
659        db: &'db dyn SourceDatabase,
660        loc: &MacroCallLoc,
661    ) -> ExpandResult<(Cow<'db, tt::TopSubtree>, MatchedArmIndex)> {
662        let _p = tracing::info_span!("macro_expand").entered();
663
664        let (ExpandResult { value: (tt, matched_arm), err }, span) = match loc.def.kind {
665            MacroDefKind::ProcMacro(..) => {
666                return self.expand_proc_macro(db).as_ref().map(|it| (Cow::Borrowed(it), None));
667            }
668            _ => {
669                let (macro_arg, undo_info, span) =
670                    self.macro_arg_considering_derives(db, &loc.kind);
671                let span = *span;
672
673                let arg = macro_arg;
674                let res = match loc.def.kind {
675                    MacroDefKind::Declarative(id, _) => {
676                        id.decl_macro_expander(db, loc.def.krate).expand(db, arg, self, span)
677                    }
678                    MacroDefKind::BuiltIn(_, it) => {
679                        it.expand(db, self, arg, span).map_err(Into::into).zip_val(None)
680                    }
681                    MacroDefKind::BuiltInDerive(_, it) => {
682                        it.expand(db, self, arg, span).map_err(Into::into).zip_val(None)
683                    }
684                    MacroDefKind::UnimplementedBuiltIn(_) => {
685                        expand_unimplemented_builtin_macro(span).zip_val(None)
686                    }
687                    MacroDefKind::BuiltInEager(_, it) => {
688                        // This might look a bit odd, but we do not expand the inputs to eager macros here.
689                        // Eager macros inputs are expanded, well, eagerly when we collect the macro calls.
690                        // That kind of expansion uses the ast id map of an eager macros input though which goes through
691                        // the HirFileId machinery. As eager macro inputs are assigned a macro file id that query
692                        // will end up going through here again, whereas we want to just want to inspect the raw input.
693                        // As such we just return the input subtree here.
694                        let eager = match &loc.kind {
695                            MacroCallKind::FnLike { eager: None, .. } => {
696                                return ExpandResult::ok(Cow::Borrowed(macro_arg)).zip_val(None);
697                            }
698                            MacroCallKind::FnLike { eager: Some(eager), .. } => Some(&**eager),
699                            _ => None,
700                        };
701
702                        let mut res = it.expand(db, self, arg, span).map_err(Into::into);
703
704                        if let Some(EagerCallInfo { error, .. }) = eager {
705                            // FIXME: We should report both errors!
706                            res.err = error.clone().or(res.err);
707                        }
708                        res.zip_val(None)
709                    }
710                    MacroDefKind::BuiltInAttr(_, it) => {
711                        let mut res = it.expand(db, self, arg, span);
712                        fixup::reverse_fixups(&mut res.value, undo_info);
713                        res.zip_val(None)
714                    }
715                    MacroDefKind::ProcMacro(_, _, _) => unreachable!(),
716                };
717                (res, span)
718            }
719        };
720
721        // Skip checking token tree limit for include! macro call
722        if !loc.def.is_include() {
723            // Set a hard limit for the expanded tt
724            if let Err(value) = check_tt_count(&tt) {
725                return value
726                    .map(|()| Cow::Owned(tt::TopSubtree::empty(tt::DelimSpan::from_single(span))))
727                    .zip_val(matched_arm);
728            }
729        }
730
731        ExpandResult { value: (Cow::Owned(tt), matched_arm), err }
732    }
733
734    /// Special case of [`Self::macro_expand`] for procedural macros. We can't LRU
735    /// proc macros, since they are not deterministic in general, and
736    /// non-determinism breaks salsa in a very, very, very bad way.
737    /// @edwin0cheng heroically debugged this once! See #4315 for details
738    #[salsa::tracked(returns(ref))]
739    fn expand_proc_macro(self, db: &dyn SourceDatabase) -> ExpandResult<tt::TopSubtree> {
740        let loc = self.loc(db);
741        let (macro_arg, undo_info, span) = self.macro_arg_considering_derives(db, &loc.kind);
742
743        let (ast, expander) = match loc.def.kind {
744            MacroDefKind::ProcMacro(ast, expander, _) => (ast, expander),
745            _ => unreachable!(),
746        };
747
748        let attr_arg = match &loc.kind {
749            MacroCallKind::Attr { attr_args: Some(attr_args), .. } => Some(&**attr_args),
750            _ => None,
751        };
752
753        let ExpandResult { value: mut tt, err } = {
754            let span = proc_macro_span(db, ast);
755            expander.expand(
756                db,
757                loc.def.krate,
758                loc.krate,
759                macro_arg,
760                attr_arg,
761                span_with_def_site_ctxt(db, span, self.into(), loc.def.edition),
762                span_with_call_site_ctxt(db, span, self.into(), loc.def.edition),
763                span_with_mixed_site_ctxt(db, span, self.into(), loc.def.edition),
764            )
765        };
766
767        // Set a hard limit for the expanded tt
768        if let Err(value) = check_tt_count(&tt) {
769            return value.map(|()| tt::TopSubtree::empty(tt::DelimSpan::from_single(*span)));
770        }
771
772        fixup::reverse_fixups(&mut tt, undo_info);
773
774        ExpandResult { value: tt, err }
775    }
776}

778impl MacroCallId {
779    /// This expands the given macro call, but with different arguments. This is
780    /// used for completion, where we want to see what 'would happen' if we insert a
781    /// token. The `token_to_map` mapped down into the expansion, with the mapped
782    /// token(s) returned with their priority.
783    pub fn expand_speculative(
784        self,
785        db: &dyn SourceDatabase,
786        speculative_args: &SyntaxNode,
787        token_to_map: SyntaxToken,
788    ) -> Option<(SyntaxNode, Vec<(SyntaxToken, u8)>)> {
789        let loc = self.loc(db);
790        let (_, _, span) = *self.macro_arg_considering_derives(db, &loc.kind);
791
792        let span_map = RealSpanMap::absolute(span.anchor.file_id);
793        let span_map = SpanMap::RealSpanMap(&span_map);
794
795        // Build the subtree and token mapping for the speculative args
796        let (mut tt, undo_info) = match &loc.kind {
797            MacroCallKind::FnLike { .. } => (
798                syntax_bridge::syntax_node_to_token_tree(
799                    speculative_args,
800                    span_map,
801                    span,
802                    if loc.def.is_proc_macro() {
803                        DocCommentDesugarMode::ProcMacro
804                    } else {
805                        DocCommentDesugarMode::Mbe
806                    },
807                ),
808                SyntaxFixupUndoInfo::NONE,
809            ),
810            MacroCallKind::Attr { .. } if loc.def.is_attribute_derive() => (
811                syntax_bridge::syntax_node_to_token_tree(
812                    speculative_args,
813                    span_map,
814                    span,
815                    DocCommentDesugarMode::ProcMacro,
816                ),
817                SyntaxFixupUndoInfo::NONE,
818            ),
819            MacroCallKind::Derive { derive_macro_id, .. } => {
820                let MacroCallKind::Attr { censored_attr_ids: attr_ids, .. } =
821                    &derive_macro_id.loc(db).kind
822                else {
823                    unreachable!("`derive_macro_id` should be `MacroCallKind::Attr`");
824                };
825                attr_macro_input_to_token_tree(
826                    db,
827                    speculative_args,
828                    span_map,
829                    span,
830                    true,
831                    attr_ids,
832                    loc.krate,
833                )
834            }
835            MacroCallKind::Attr { censored_attr_ids: attr_ids, .. } => {
836                attr_macro_input_to_token_tree(
837                    db,
838                    speculative_args,
839                    span_map,
840                    span,
841                    false,
842                    attr_ids,
843                    loc.krate,
844                )
845            }
846        };
847
848        let attr_arg = match &loc.kind {
849            MacroCallKind::Attr { censored_attr_ids: attr_ids, .. } => {
850                if loc.def.is_attribute_derive() {
851                    // for pseudo-derive expansion we actually pass the attribute itself only
852                    ast::Attr::cast(speculative_args.clone())
853                        .and_then(|attr| {
854                            if let ast::Meta::TokenTreeMeta(meta) = attr.meta()? {
855                                meta.token_tree()
856                            } else {
857                                None
858                            }
859                        })
860                        .map(|token_tree| {
861                            let mut tree = syntax_node_to_token_tree(
862                                token_tree.syntax(),
863                                span_map,
864                                span,
865                                DocCommentDesugarMode::ProcMacro,
866                            );
867                            tree.set_top_subtree_delimiter_kind(tt::DelimiterKind::Invisible);
868                            tree.set_top_subtree_delimiter_span(tt::DelimSpan::from_single(span));
869                            tree
870                        })
871                } else {
872                    // Attributes may have an input token tree, build the subtree and map for this as well
873                    // then try finding a token id for our token if it is inside this input subtree.
874                    let item = ast::Item::cast(speculative_args.clone())?;
875                    let (_, meta) = attr_ids
876                        .invoc_attr()
877                        .find_attr_range_with_source_opt(db, loc.krate, &item)?;
878                    if let ast::Meta::TokenTreeMeta(meta) = meta
879                        && let Some(tt) = meta.token_tree()
880                    {
881                        let mut attr_arg = syntax_bridge::syntax_node_to_token_tree(
882                            tt.syntax(),
883                            span_map,
884                            span,
885                            DocCommentDesugarMode::ProcMacro,
886                        );
887                        attr_arg.set_top_subtree_delimiter_kind(tt::DelimiterKind::Invisible);
888                        Some(attr_arg)
889                    } else {
890                        None
891                    }
892                }
893            }
894            _ => None,
895        };
896
897        // Do the actual expansion, we need to directly expand the proc macro due to the attribute args
898        // Otherwise the expand query will fetch the non speculative attribute args and pass those instead.
899        let mut speculative_expansion = match loc.def.kind {
900            MacroDefKind::ProcMacro(ast, expander, _) => {
901                let span = proc_macro_span(db, ast);
902                tt.set_top_subtree_delimiter_kind(tt::DelimiterKind::Invisible);
903                tt.set_top_subtree_delimiter_span(tt::DelimSpan::from_single(span));
904                expander.expand(
905                    db,
906                    loc.def.krate,
907                    loc.krate,
908                    &tt,
909                    attr_arg.as_ref(),
910                    span_with_def_site_ctxt(db, span, self.into(), loc.def.edition),
911                    span_with_call_site_ctxt(db, span, self.into(), loc.def.edition),
912                    span_with_mixed_site_ctxt(db, span, self.into(), loc.def.edition),
913                )
914            }
915            MacroDefKind::BuiltInAttr(_, it) if it.is_derive() => {
916                pseudo_derive_attr_expansion(&tt, attr_arg.as_ref()?, span)
917            }
918            MacroDefKind::Declarative(it, _) => it
919                .decl_macro_expander(db, loc.krate)
920                .expand_unhygienic(db, &tt, loc.kind.call_style(), span),
921            MacroDefKind::BuiltIn(_, it) => it.expand(db, self, &tt, span).map_err(Into::into),
922            MacroDefKind::BuiltInDerive(_, it) => {
923                it.expand(db, self, &tt, span).map_err(Into::into)
924            }
925            MacroDefKind::BuiltInEager(_, it) => it.expand(db, self, &tt, span).map_err(Into::into),
926            MacroDefKind::BuiltInAttr(_, it) => it.expand(db, self, &tt, span),
927            MacroDefKind::UnimplementedBuiltIn(_) => expand_unimplemented_builtin_macro(span),
928        };
929
930        let expand_to = loc.expand_to();
931
932        fixup::reverse_fixups(&mut speculative_expansion.value, &undo_info);
933        let (node, rev_tmap) =
934            token_tree_to_syntax_node(db, &speculative_expansion.value, expand_to);
935
936        let syntax_node = node.syntax_node();
937        let token = rev_tmap
938            .ranges_with_span(span_map.span_for_range(token_to_map.text_range()))
939            .filter_map(|(range, ctx)| {
940                syntax_node.covering_element(range).into_token().zip(Some(ctx))
941            })
942            .map(|(t, ctx)| {
943                // prefer tokens of the same kind and text, as well as non opaque marked ones
944                // Note the inversion of the score here, as we want to prefer the first token in case
945                // of all tokens having the same score
946                let ranking = ctx.is_opaque(db) as u8
947                    + 2 * (t.kind() != token_to_map.kind()) as u8
948                    + 4 * ((t.text() != token_to_map.text()) as u8);
949                (t, ranking)
950            })
951            .collect();
952        Some((node.syntax_node(), token))
953    }
954}

963/// Retrieves the span to be used for a proc-macro expansions spans.
964/// This is a firewall query as it requires parsing the file, which we don't want proc-macros to
965/// directly depend on as that would cause to frequent invalidations, mainly because of the
966/// parse queries being LRU cached. If they weren't the invalidations would only happen if the
967/// user wrote in the file that defines the proc-macro.
968fn proc_macro_span(db: &dyn SourceDatabase, ast: AstId<ast::Fn>) -> Span {
969    #[salsa::tracked(returns(copy))]
970    fn proc_macro_span(db: &dyn SourceDatabase, ast: AstId<ast::Fn>, _: ()) -> Span {
971        let (parse, span_map) = ast.file_id.parse_with_map(db);
972        let root = parse.syntax_node();
973        let ast_id_map = ast.file_id.ast_id_map(db);
974
975        let node = ast_id_map.get(ast.value).to_node(&root);
976        let range = ast::HasName::name(&node)
977            .map_or_else(|| node.syntax().text_range(), |name| name.syntax().text_range());
978        span_map.span_for_range(range)
979    }
980    proc_macro_span(db, ast, ())
981}

984    db: &dyn SourceDatabase,
985    tt: &tt::TopSubtree,

998fn check_tt_count(tt: &tt::TopSubtree) -> Result<(), ExpandResult<()>> {
999    let tt = tt.top_subtree();
1000    let count = tt.count();
1001    if count <= TOKEN_LIMIT {
1002        Ok(())
1003    } else {
1004        Err(ExpandResult {
1005            value: (),
1006            err: Some(ExpandError::other(
1007                tt.delimiter.open,
1008                format!(
1009                    "macro invocation exceeds token limit: produced {count} tokens, limit is {TOKEN_LIMIT}",
1010                ),
1011            )),
1012        })
1013    }

1189            MacroCallKind::FnLike { ast_id: InFile { value, .. }, .. } => value.erase(),
1190            MacroCallKind::Derive { ast_id: InFile { value, .. }, .. } => value.erase(),
1191            MacroCallKind::Attr { ast_id: InFile { value, .. }, .. } => value.erase(),
1192        }

1195    /// Returns the original file range that best describes the location of this macro call.
1196    ///
1197    /// This spans the entire macro call, including its input. That is for
1198    /// - fn_like! {}, it spans the path and token tree
1199    /// - #\[derive], it spans the `#[derive(...)]` attribute and the annotated item
1200    /// - #\[attr], it spans the `#[attr(...)]` attribute and the annotated item
1201    pub fn original_call_range_with_input(&self, db: &dyn SourceDatabase) -> FileRange {
1202        let get_range = |kind: &_| match kind {
1203            MacroCallKind::FnLike { ast_id, .. } => ast_id.erase(),
1204            MacroCallKind::Derive { ast_id, .. } => ast_id.erase(),
1205            MacroCallKind::Attr { ast_id, .. } => ast_id.erase(),
1206        };
1207
1208        let mut ast_id = get_range(self);
1209        let mut file_id = self.file_id();
1210        let file_id = loop {
1211            match file_id {
1212                HirFileId::MacroFile(file) => {
1213                    let kind = &file.loc(db).kind;
1214                    ast_id = get_range(kind);
1215                    file_id = kind.file_id();
1216                }
1217                HirFileId::FileId(file_id) => break file_id,
1218            }
1219        };
1220
1221        FileRange { range: ast_id.to_ptr(db).text_range(), file_id }
1222    }
1223
1224    /// Returns the original file range that best describes the location of this macro call.
1225    ///
1226    /// Here we try to roughly match what rustc does to improve diagnostics: fn-like macros
1227    /// get the macro path (rustc shows the whole `ast::MacroCall`), attribute macros get the
1228    /// attribute's range, and derives get only the specific derive that is being referred to.
1229    pub fn original_call_range(&self, db: &dyn SourceDatabase, krate: Crate) -> FileRange {
1230        let get_range = |kind: &_| match kind {
1231            MacroCallKind::FnLike { ast_id, .. } => {
1232                let node = ast_id.to_node(db);
1233                node.path()
1234                    .unwrap()
1235                    .syntax()
1236                    .text_range()
1237                    .cover(node.excl_token().unwrap().text_range())
1238            }
1239            MacroCallKind::Derive { ast_id, derive_attr_index, .. } => {
1240                // FIXME: should be the range of the macro name, not the whole derive
1241                derive_attr_index.find_attr_range(db, krate, *ast_id).1.syntax().text_range()
1242            }
1243            // FIXME: handle `cfg_attr`
1244            MacroCallKind::Attr { ast_id, censored_attr_ids: attr_ids, .. } => {
1245                attr_ids.invoc_attr().find_attr_range(db, krate, *ast_id).1.syntax().text_range()
1246            }
1247        };
1248
1249        let mut range = get_range(self);
1250        let mut file_id = self.file_id();
1251        let file_id = loop {
1252            match file_id {
1253                HirFileId::MacroFile(file) => {
1254                    let kind = &file.loc(db).kind;
1255                    range = get_range(kind);
1256                    file_id = kind.file_id();
1257                }
1258                HirFileId::FileId(file_id) => break file_id,
1259            }
1260        };
1261
1262        FileRange { range, file_id }
1263    }

1265    fn arg(&self, db: &dyn SourceDatabase) -> InFile<Option<SyntaxNode>> {
1266        match self {
1267            MacroCallKind::FnLike { ast_id, .. } => {
1268                ast_id.to_in_file_node(db).map(|it| Some(it.token_tree()?.syntax().clone()))
1269            }
1270            MacroCallKind::Derive { ast_id, .. } => {
1271                ast_id.to_in_file_node(db).syntax().cloned().map(Some)
1272            }
1273            MacroCallKind::Attr { ast_id, .. } => {
1274                ast_id.to_in_file_node(db).syntax().cloned().map(Some)
1275            }
1276        }
1277    }

1419/// Maps up the text range out of the expansion hierarchy back into the original file its from only
1420/// considering the root spans contained.
1421/// Unlike [`map_node_range_up`], this will not return `None` if any anchors or syntax contexts differ.
1422pub fn map_node_range_up_rooted(
1423    db: &dyn SourceDatabase,
1424    exp_map: &ExpansionSpanMap,
1425    range: TextRange,
1426) -> Option<FileRange> {
1427    let mut spans = exp_map.spans_for_range(range).filter(|span| span.ctx.is_root());
1428    let Span { range, anchor, ctx } = spans.next()?;
1429    let mut start = range.start();
1430    let mut end = range.end();
1431
1432    for span in spans {
1433        if span.anchor != anchor {
1434            return None;
1435        }
1436        start = start.min(span.range.start());
1437        end = end.max(span.range.end());
1438    }
1439    Some(resolve_span(db, Span { range: TextRange::new(start, end), anchor, ctx }))
1440}

1447    exp_map: &ExpansionSpanMap,

1485/// In Rust, macros expand token trees to token trees. When we want to turn a
1486/// token tree into an AST node, we need to figure out what kind of AST node we
1487/// want: something like `foo` can be a type, an expression, or a pattern.
1488///
1489/// Naively, one would think that "what this expands to" is a property of a
1490/// particular macro: macro `m1` returns an item, while macro `m2` returns an
1491/// expression, etc. That's not the case -- macros are polymorphic in the
1492/// result, and can expand to any type of the AST node.
1493///
1494/// What defines the actual AST node is the syntactic context of the macro
1495/// invocation. As a contrived example, in `let T![*] = T![*];` the first `T`
1496/// expands to a pattern, while the second one expands to an expression.
1497///
1498/// `ExpandTo` captures this bit of information about a particular macro call
1499/// site.
1500#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
1501pub enum ExpandTo {
1502    Statements,
1503    Items,
1504    Pattern,
1505    Type,
1506    Expr,
1507}
1508
1509impl ExpandTo {
1510    pub fn from_call_site(call: &ast::MacroCall) -> ExpandTo {
1511        use syntax::SyntaxKind::*;
1512
1513        let syn = call.syntax();
1514
1515        let parent = match syn.parent() {
1516            Some(it) => it,
1517            None => return ExpandTo::Statements,
1518        };
1519
1520        // FIXME: macros in statement position are treated as expression statements, they should
1521        // probably be their own statement kind. The *grand*parent indicates what's valid.
1522        if parent.kind() == MACRO_EXPR
1523            && parent
1524                .parent()
1525                .is_some_and(|p| matches!(p.kind(), EXPR_STMT | STMT_LIST | MACRO_STMTS))
1526        {
1527            return ExpandTo::Statements;
1528        }
1529
1530        match parent.kind() {
1531            MACRO_ITEMS | SOURCE_FILE | ITEM_LIST => ExpandTo::Items,
1532            MACRO_STMTS | EXPR_STMT | STMT_LIST => ExpandTo::Statements,
1533            MACRO_PAT => ExpandTo::Pattern,
1534            MACRO_TYPE => ExpandTo::Type,
1535
1536            ARG_LIST | ARRAY_EXPR | AWAIT_EXPR | BIN_EXPR | BREAK_EXPR | CALL_EXPR | CAST_EXPR
1537            | CLOSURE_EXPR | FIELD_EXPR | FOR_EXPR | IF_EXPR | INDEX_EXPR | LET_EXPR
1538            | MATCH_ARM | MATCH_EXPR | MATCH_GUARD | METHOD_CALL_EXPR | PAREN_EXPR | PATH_EXPR
1539            | PREFIX_EXPR | RANGE_EXPR | RECORD_EXPR_FIELD | REF_EXPR | RETURN_EXPR | TRY_EXPR
1540            | TUPLE_EXPR | WHILE_EXPR | MACRO_EXPR => ExpandTo::Expr,
1541            _ => {
1542                // Unknown , Just guess it is `Items`
1543                ExpandTo::Items
1544            }
1545        }
1546    }
1547}

1733    /// How many macro expansions separate this file from the real file it originates from.
1734    ///
1735    /// `Expander` and `DefCollector` seed their expansion depth counters from this. Both otherwise
1736    /// start at zero for every body and every block, so a macro expanding to an item that invokes
1737    /// the macro again advances neither counter past one and never reaches the recursion limit.
1738    ///
1739    /// Written recursively rather than as a loop so that salsa memoizes each file's depth. Callers
1740    /// ask for this once per body, and the bodies of one expansion chain sit at depths 1..n, so a
1741    /// loop would walk 1 + 2 + ... + n steps for that chain. Reusing the parent's memoized depth
1742    /// makes each file O(1) once computed, and the chain linear.
1743    #[inline]
1744    pub fn macro_expansion_depth(self, db: &dyn SourceDatabase) -> u32 {
1745        match self {
