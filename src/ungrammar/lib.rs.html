1//! Ungrammar -- a DSL for specifying concrete syntax tree grammar.
2//!
3//! Producing a parser is an explicit non-goal -- it's ok for this grammar to be
4//! ambiguous, non LL, non LR, etc.
5//!
6//! See this
7//! [introductory post](https://rust-analyzer.github.io/blog/2020/10/24/introducing-ungrammar.html)
8//! for details.

25/// A node, like `A = 'b' | 'c'`.
26///
27/// Indexing into a [`Grammar`] with a [`Node`] returns a reference to a
28/// [`NodeData`].
29#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
30pub struct Node(usize);
31
32/// A token, denoted with single quotes, like `'+'` or `'struct'`.
33///
34/// Indexing into a [`Grammar`] with a [`Token`] returns a reference to a
35/// [`TokenData`].
36#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
37pub struct Token(usize);
38
39/// An Ungrammar grammar.
40#[derive(Default, Debug)]
41pub struct Grammar {
42    nodes: Vec<NodeData>,
43    tokens: Vec<TokenData>,
44}
45
46impl FromStr for Grammar {
47    type Err = Error;
48    fn from_str(s: &str) -> Result<Self> {
49        let tokens = lexer::tokenize(s)?;
50        parser::parse(tokens)
51    }
52}
53
54impl Grammar {
55    /// Returns an iterator over all nodes in the grammar.
56    pub fn iter(&self) -> impl Iterator<Item = Node> + '_ {
57        (0..self.nodes.len()).map(Node)
58    }
59
60    /// Returns an iterator over all tokens in the grammar.
61    pub fn tokens(&self) -> impl Iterator<Item = Token> + '_ {
62        (0..self.tokens.len()).map(Token)
63    }
64}
65
66impl ops::Index<Node> for Grammar {
67    type Output = NodeData;
68    fn index(&self, Node(index): Node) -> &NodeData {
69        &self.nodes[index]
70    }
71}
72
73impl ops::Index<Token> for Grammar {
74    type Output = TokenData;
75    fn index(&self, Token(index): Token) -> &TokenData {
76        &self.tokens[index]
77    }
78}

80/// Data about a node.
81#[derive(Debug)]
82pub struct NodeData {
83    /// The name of the node.
84    ///
85    /// In the rule `A = 'b' | 'c'`, this is `"A"`.
86    pub name: String,
87    /// The rule for this node.
88    ///
89    /// In the rule `A = 'b' | 'c'`, this represents `'b' | 'c'`.
90    pub rule: Rule,
91}
92
93/// Data about a token.
94#[derive(Debug)]
95pub struct TokenData {
96    /// The name of the token.
97    pub name: String,
98}
99
100/// A production rule.
101#[derive(Debug, Clone, Eq, PartialEq)]
102pub enum Rule {
103    /// A labeled rule, like `a:B` (`"a"` is the label, `B` is the rule).
104    Labeled {
105        /// The label.
106        label: String,
107        /// The rule.
108        rule: Box<Rule>,
109    },
110    /// A node, like `A`.
111    Node(Node),
112    /// A token, like `'struct'`.
113    Token(Token),
114    /// A sequence of rules, like `'while' '(' Expr ')' Stmt`.
115    Seq(Vec<Rule>),
116    /// An alternative between many rules, like `'+' | '-' | '*' | '/'`.
117    Alt(Vec<Rule>),
118    /// An optional rule, like `A?`.
119    Opt(Box<Rule>),
120    /// A repeated rule, like `A*`.
121    Rep(Box<Rule>),
122}
