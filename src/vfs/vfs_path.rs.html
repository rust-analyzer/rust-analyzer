6/// Path in [`Vfs`].
7///
8/// Long-term, we want to support files which do not reside in the file-system,
9/// so we treat `VfsPath`s as opaque identifiers.
10///
11/// [`Vfs`]: crate::Vfs
12#[derive(Clone, Ord, PartialOrd, Eq, PartialEq, Hash)]
13pub struct VfsPath(VfsPathRepr);
14
15impl VfsPath {
16    /// Creates an "in-memory" path from `/`-separated string.
17    ///
18    /// This is most useful for testing, to avoid windows/linux differences
19    ///
20    /// # Panics
21    ///
22    /// Panics if `path` does not start with `'/'`.
23    pub fn new_virtual_path(path: String) -> VfsPath {
24        assert!(path.starts_with('/'));
25        VfsPath(VfsPathRepr::VirtualPath(VirtualPath(path)))
26    }

28    /// Create a path from string. Input should be a string representation of
29    /// an absolute path inside filesystem
30    pub fn new_real_path(path: String) -> VfsPath {
31        VfsPath::from(AbsPathBuf::assert(path.into()))
32    }

34    /// Returns the `AbsPath` representation of `self` if `self` is on the file system.
35    pub fn as_path(&self) -> Option<&AbsPath> {
36        match &self.0 {
37            VfsPathRepr::PathBuf(it) => Some(it.as_path()),
38            VfsPathRepr::VirtualPath(_) => None,
39        }
40    }

42    pub fn into_abs_path(self) -> Option<AbsPathBuf> {
43        match self.0 {
44            VfsPathRepr::PathBuf(it) => Some(it),
45            VfsPathRepr::VirtualPath(_) => None,
46        }
47    }

49    /// Creates a new `VfsPath` with `path` adjoined to `self`.
50    pub fn join(&self, path: &str) -> Option<VfsPath> {
51        match &self.0 {
52            VfsPathRepr::PathBuf(it) => {
53                let res = it.join(path).normalize();
54                Some(VfsPath(VfsPathRepr::PathBuf(res)))
55            }
56            VfsPathRepr::VirtualPath(it) => {
57                let res = it.join(path)?;
58                Some(VfsPath(VfsPathRepr::VirtualPath(res)))
59            }
60        }
61    }

63    /// Remove the last component of `self` if there is one.
64    ///
65    /// If `self` has no component, returns `false`; else returns `true`.
66    ///
67    /// # Example
68    ///
69    /// ```ignore
70    /// # use vfs::{AbsPathBuf, VfsPath};
71    /// let mut path = VfsPath::from(AbsPathBuf::assert("/foo/bar".into()));
72    /// assert!(path.pop());
73    /// assert_eq!(path, VfsPath::from(AbsPathBuf::assert("/foo".into())));
74    /// assert!(path.pop());
75    /// assert_eq!(path, VfsPath::from(AbsPathBuf::assert("/".into())));
76    /// assert!(!path.pop());
77    /// ```
78    pub fn pop(&mut self) -> bool {
79        match &mut self.0 {
80            VfsPathRepr::PathBuf(it) => it.pop(),
81            VfsPathRepr::VirtualPath(it) => it.pop(),
82        }
83    }
84
85    /// Returns `true` if `other` is a prefix of `self`.
86    pub fn starts_with(&self, other: &VfsPath) -> bool {
87        match (&self.0, &other.0) {
88            (VfsPathRepr::PathBuf(lhs), VfsPathRepr::PathBuf(rhs)) => lhs.starts_with(rhs),
89            (VfsPathRepr::VirtualPath(lhs), VfsPathRepr::VirtualPath(rhs)) => lhs.starts_with(rhs),
90            (VfsPathRepr::PathBuf(_) | VfsPathRepr::VirtualPath(_), _) => false,
91        }
92    }
93
94    pub fn strip_prefix(&self, other: &VfsPath) -> Option<&RelPath> {
95        match (&self.0, &other.0) {
96            (VfsPathRepr::PathBuf(lhs), VfsPathRepr::PathBuf(rhs)) => lhs.strip_prefix(rhs),
97            (VfsPathRepr::VirtualPath(lhs), VfsPathRepr::VirtualPath(rhs)) => lhs.strip_prefix(rhs),
98            (VfsPathRepr::PathBuf(_) | VfsPathRepr::VirtualPath(_), _) => None,
99        }
100    }

102    /// Returns the `VfsPath` without its final component, if there is one.
103    ///
104    /// Returns [`None`] if the path is a root or prefix.
105    pub fn parent(&self) -> Option<VfsPath> {
106        let mut parent = self.clone();
107        if parent.pop() { Some(parent) } else { None }
108    }

110    /// Returns `self`'s base name and file extension.
111    pub fn name_and_extension(&self) -> Option<(&str, Option<&str>)> {
112        match &self.0 {
113            VfsPathRepr::PathBuf(p) => p.name_and_extension(),
114            VfsPathRepr::VirtualPath(p) => p.name_and_extension(),
115        }
116    }

118    /// **Don't make this `pub`**
119    ///
120    /// Encode the path in the given buffer.
121    ///
122    /// The encoding will be `0` if [`AbsPathBuf`], `1` if [`VirtualPath`], followed
123    /// by `self`'s representation.
124    ///
125    /// Note that this encoding is dependent on the operating system.
126    pub(crate) fn encode(&self, buf: &mut Vec<u8>) {
127        let tag = match &self.0 {
128            VfsPathRepr::PathBuf(_) => 0,
129            VfsPathRepr::VirtualPath(_) => 1,
130        };
131        buf.push(tag);
132        match &self.0 {
133            VfsPathRepr::PathBuf(path) => {
134                #[cfg(windows)]
135                {
136                    use windows_paths::Encode;
137                    let components = path.components();
138                    let mut add_sep = false;
139                    for component in components {
140                        if add_sep {
141                            windows_paths::SEP.encode(buf);
142                        }
143                        let len_before = buf.len();
144                        match component {
145                            paths::Utf8Component::Prefix(prefix) => {
146                                // kind() returns a normalized and comparable path prefix.
147                                prefix.kind().encode(buf);
148                            }
149                            paths::Utf8Component::RootDir => {
150                                if !add_sep {
151                                    component.as_str().encode(buf);
152                                }
153                            }
154                            _ => component.as_str().encode(buf),
155                        }
156
157                        // some components may be encoded empty
158                        add_sep = len_before != buf.len();
159                    }
160                }
161                #[cfg(not(windows))]
162                {
163                    buf.extend(path.as_str().as_bytes());
164                }
165            }
166            VfsPathRepr::VirtualPath(VirtualPath(s)) => buf.extend(s.as_bytes()),
167        }
168    }
169}

173    pub(crate) trait Encode {

177    impl Encode for u8 {
178        fn encode(&self, buf: &mut Vec<u8>) {
179            buf.push(*self);
180        }
181    }

183    impl Encode for &str {
184        fn encode(&self, buf: &mut Vec<u8>) {
185            buf.extend(self.as_bytes());
186        }
187    }

189    pub(crate) const SEP: &str = "\\";
190    const VERBATIM: &str = "\\\\?\\";
191    const UNC: &str = "UNC";
192    const DEVICE: &str = "\\\\.\\";
193    const COLON: &str = ":";
194
195    impl Encode for paths::Utf8Prefix<'_> {
196        fn encode(&self, buf: &mut Vec<u8>) {
197            match self {
198                paths::Utf8Prefix::Verbatim(c) => {
199                    VERBATIM.encode(buf);
200                    c.encode(buf);
201                }
202                paths::Utf8Prefix::VerbatimUNC(server, share) => {
203                    VERBATIM.encode(buf);
204                    UNC.encode(buf);
205                    SEP.encode(buf);
206                    server.encode(buf);
207                    SEP.encode(buf);
208                    share.encode(buf);
209                }
210                paths::Utf8Prefix::VerbatimDisk(d) => {
211                    VERBATIM.encode(buf);
212                    d.encode(buf);
213                    COLON.encode(buf);
214                }
215                paths::Utf8Prefix::DeviceNS(device) => {
216                    DEVICE.encode(buf);
217                    device.encode(buf);
218                }
219                paths::Utf8Prefix::UNC(server, share) => {
220                    SEP.encode(buf);
221                    SEP.encode(buf);
222                    server.encode(buf);
223                    SEP.encode(buf);
224                    share.encode(buf);
225                }
226                paths::Utf8Prefix::Disk(d) => {
227                    d.encode(buf);
228                    COLON.encode(buf);
229                }
230            }
231        }
232    }

314/// `/`-separated virtual path.
315///
316/// This is used to describe files that do not reside on the file system.
317#[derive(Debug, Clone, Ord, PartialOrd, Eq, PartialEq, Hash)]
318struct VirtualPath(String);
319
320impl VirtualPath {
321    /// Returns `true` if `other` is a prefix of `self`.
322    fn starts_with(&self, other: &VirtualPath) -> bool {
323        <_ as AsRef<paths::Utf8Path>>::as_ref(&self.0).starts_with(&other.0)
324    }
325
326    fn strip_prefix(&self, base: &VirtualPath) -> Option<&RelPath> {
327        <_ as AsRef<paths::Utf8Path>>::as_ref(&self.0)
328            .strip_prefix(&base.0)
329            .ok()
330            .map(RelPath::new_unchecked)
331    }
332
333    /// Remove the last component of `self`.
334    ///
335    /// This will find the last `'/'` in `self`, and remove everything after it,
336    /// including the `'/'`.
337    ///
338    /// If `self` contains no `'/'`, returns `false`; else returns `true`.
339    ///
340    /// # Example
341    ///
342    /// ```rust,ignore
343    /// let mut path = VirtualPath("/foo/bar".to_string());
344    /// path.pop();
345    /// assert_eq!(path.0, "/foo");
346    /// path.pop();
347    /// assert_eq!(path.0, "");
348    /// ```
349    fn pop(&mut self) -> bool {
350        let pos = match self.0.rfind('/') {
351            Some(pos) => pos,
352            None => return false,
353        };
354        self.0 = self.0[..pos].to_string();
355        true
356    }
357
358    /// Append the given *relative* path `path` to `self`.
359    ///
360    /// This will resolve any leading `"../"` in `path` before appending it.
361    ///
362    /// Returns [`None`] if `path` has more leading `"../"` than the number of
363    /// components in `self`.
364    ///
365    /// # Notes
366    ///
367    /// In practice, appending here means `self/path` as strings.
368    fn join(&self, mut path: &str) -> Option<VirtualPath> {
369        let mut res = self.clone();
370        while path.starts_with("../") {
371            if !res.pop() {
372                return None;
373            }
374            path = &path["../".len()..];
375        }
376        path = path.trim_start_matches("./");
377        res.0 = format!("{}/{path}", res.0);
378        Some(res)
379    }
380
381    /// Returns `self`'s base name and file extension.
382    ///
383    /// # Returns
384    /// - `None` if `self` ends with `"//"`.
385    /// - `Some((name, None))` if `self`'s base contains no `.`, or only one `.` at the start.
386    /// - `Some((name, Some(extension))` else.
387    ///
388    /// # Note
389    /// The extension will not contains `.`. This means `"/foo/bar.baz.rs"` will
390    /// return `Some(("bar.baz", Some("rs"))`.
391    fn name_and_extension(&self) -> Option<(&str, Option<&str>)> {
392        let file_path = if self.0.ends_with('/') { &self.0[..&self.0.len() - 1] } else { &self.0 };
393        let file_name = match file_path.rfind('/') {
394            Some(position) => &file_path[position + 1..],
395            None => file_path,
396        };
397
398        if file_name.is_empty() {
399            None
400        } else {
401            let mut file_stem_and_extension = file_name.rsplitn(2, '.');
402            let extension = file_stem_and_extension.next();
403            let file_stem = file_stem_and_extension.next();
404
405            match (file_stem, extension) {
406                (None, None) => None,
407                (None | Some(""), Some(_)) => Some((file_name, None)),
408                (Some(file_stem), extension) => Some((file_stem, extension)),
409            }
410        }
411    }
